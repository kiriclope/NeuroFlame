import torch
import torch.nn as nn
import torch.nn.functional as F

def safe_mean(tensor):
    if tensor.numel() == 0:
        return torch.tensor(0.0, device=tensor.device, dtype=tensor.dtype)
    return tensor.mean()

class BCEOneClassLoss(nn.Module):
    def __init__(self):
        super().__init__()
        self.criterion = nn.BCEWithLogitsLoss(reduction='none')

    def forward(self, logits, targets, class_bal=1):
        bce = self.criterion(logits, targets.float())

        if class_bal == 0:
            mask1 = (targets == 1)
            pos_loss = safe_mean(bce[mask1])

            mask0 = (targets == 0)
            neutral_targets = torch.full_like(logits[mask0], 0.5)
            neutral_loss = safe_mean(self.criterion(logits[mask0], neutral_targets))
            return pos_loss + 0.01 * neutral_loss

        return safe_mean(bce)

class SignBCELoss(nn.Module):
    def __init__(self, alpha=0.5, thresh=1.0, class_bal=0):
        super().__init__()
        self.alpha = alpha
        self.thresh = thresh
        self.class_bal = class_bal
        self.bce_with_logits = BCEOneClassLoss()

    def forward(self, readout, targets):
        bce_loss = 0.0
        if self.alpha != 1.0:
            bce_loss = self.bce_with_logits(readout, targets, self.class_bal)

        sign_overlap = torch.sign(2 * targets - 1) * readout
        sign_loss = torch.zeros_like(sign_overlap)

        if self.alpha != 0:
            if self.class_bal == 0:
                mask0 = (targets == 0)
                if mask0.sum() > 0:
                    sign_loss[mask0] = 0.1 * torch.abs(sign_overlap[mask0])

                mask1 = (targets == 1)
                if mask1.sum() > 0:
                    sign_loss[mask1] = F.relu(self.thresh - sign_overlap[mask1])
            else:
                sign_loss = F.relu(self.thresh - sign_overlap)

        return (1 - self.alpha) * bce_loss + self.alpha * safe_mean(sign_loss)

class DualLoss(nn.Module):
    def __init__(
        self,
        device,
        alpha=1.0,
        thresh=5.0,
        stim_idx=None,
        gng_idx=None,
        cue_idx=None,
        test_idx=None,
        rwd_idx=None,
        zero_idx=None,
        read_idx=None,
        class_bal=None,
    ):
        super().__init__()
        self.alpha = alpha
        self.thresh = thresh
        self.class_bal = class_bal or [0]

        self.zero_idx = zero_idx if zero_idx is not None else []
        self.stim_idx = torch.tensor(stim_idx or [], dtype=torch.int, device=device)
        self.gng_idx  = torch.tensor(gng_idx  or [], dtype=torch.int, device=device)
        self.cue_idx  = torch.tensor(cue_idx  or [], dtype=torch.int, device=device)
        self.test_idx = torch.tensor(test_idx or [], dtype=torch.int, device=device)
        self.rwd_idx  = torch.tensor(rwd_idx  or [], dtype=torch.int, device=device)

        self.read_idx = read_idx or [-1]
        self.loss = SignBCELoss(self.alpha, self.thresh)
        self.l1loss = nn.SmoothL1Loss()

    def forward(self, readout, targets):
        zeros = torch.zeros_like(readout[:, self.zero_idx, 0])
        loss = self.l1loss(readout[:, self.zero_idx, self.read_idx[0]], zeros)

        if len(self.read_idx) > 1 and self.stim_idx.numel() > 0:
            loss += self.l1loss(
                readout[:, :self.stim_idx[0]-1, self.read_idx[1]],
                zeros[:, :self.stim_idx[0]-1]
            )

        if self.test_idx.numel() != 0:
            self.loss.class_bal = 1
            loss += self.loss(
                readout[:, self.test_idx, self.read_idx[-1]],
                targets[:, -1, :self.test_idx.shape[0]]
            )

        if self.cue_idx.numel() != 0:
            self.loss.class_bal = self.class_bal[3]
            loss += self.loss(
                readout[:, self.cue_idx, self.read_idx[3]],
                targets[:, 2, :self.cue_idx.shape[0]]
            )

        if self.gng_idx.numel() != 0:
            self.loss.class_bal = 1
            loss += self.loss(
                readout[:, self.gng_idx, self.read_idx[2]],
                targets[:, 2, :self.gng_idx.shape[0]]
            )

        if self.stim_idx.numel() != 0:
            self.loss.class_bal = 1
            loss += self.loss(
                readout[:, self.stim_idx, self.read_idx[1]],
                targets[:, 1, :self.stim_idx.shape[0]]
            )

        if self.rwd_idx.numel() != 0:
            self.loss.class_bal = self.class_bal[0]
            loss += self.loss(
                readout[:, self.rwd_idx, self.read_idx[0]],
                targets[:, 0, :self.rwd_idx.shape[0]]
            )

        return loss

import numpy as np
import torch
from torch.utils.data import TensorDataset, DataLoader
from sklearn.model_selection import train_test_split

def split_data(X, Y, train_perc=0.8, batch_size=32):
    if Y.ndim == 3:
        stratify = Y[:, 0, 0].cpu().numpy()
    else:
        stratify = Y[:, 0].cpu().numpy()

    X_train, X_val, Y_train, Y_val = train_test_split(
        X, Y, train_size=train_perc, stratify=stratify, shuffle=True
    )

    train_loader = DataLoader(TensorDataset(X_train, Y_train), batch_size=batch_size, shuffle=True)
    val_loader = DataLoader(TensorDataset(X_val, Y_val), batch_size=batch_size, shuffle=False)
    return train_loader, val_loader

def get_steps(model):
    return np.arange(0, model.N_STEPS - model.N_STEADY, model.N_WINDOW)

def make_dpa_dataset(model, device, n_batch, rank):
    steps = get_steps(model)

    mask = steps >= (model.N_STIM_ON[-1].cpu().numpy() - model.N_STEADY)
    rwd_idx = np.where(mask)[0]

    test_idx = rwd_idx if rank == 3 else []

    stim_mask = (
        (steps >= (model.N_STIM_ON[0].cpu().numpy() - model.N_STEADY)) &
        (steps <  (model.N_STIM_ON[-1].cpu().numpy() - model.N_STEADY))
    )
    stim_idx = np.where(stim_mask)[0]
    zero_idx = np.where(~mask)[0]

    model.N_BATCH = n_batch
    model.lr_eval_win = max(len(rwd_idx), len(stim_idx))

    ff_input = []
    labels = np.zeros((3, 4, model.N_BATCH, model.lr_eval_win))

    l = 0
    for i in [-1, 1]:
        for k in [-1, 1]:
            model.I0[0] = i
            model.I0[4] = k

            if k == 1:
                labels[2, l] = 1
            if i == 1:
                labels[1, l] = 1
            if i == k:
                labels[0, l] = 1

            ff_input.append(model.init_ff_input())
            l += 1

    labels = torch.tensor(labels, dtype=torch.float, device=device).reshape(3, -1, model.lr_eval_win).transpose(0, 1)
    ff_input = torch.vstack(ff_input)

    idx = dict(rwd_idx=rwd_idx, stim_idx=stim_idx, test_idx=test_idx, zero_idx=zero_idx)
    return ff_input, labels, idx

def make_gonogo_dataset(model, device, n_batch):
    steps = get_steps(model)

    rwd_mask = (
        (steps >= (model.N_STIM_ON[2].cpu().numpy() - model.N_STEADY)) &
        (steps <  (model.N_STIM_ON[4].cpu().numpy() - model.N_STEADY))
    )
    rwd_idx = np.where(rwd_mask)[0]

    stim_mask = (
        (steps >= (model.N_STIM_ON[1].cpu().numpy() - model.N_STEADY)) &
        (steps <  (model.N_STIM_ON[2].cpu().numpy() - model.N_STEADY))
    )
    stim_idx = np.where(stim_mask)[0]
    zero_idx = np.where(steps < (model.N_STIM_ON[1].cpu().numpy() - model.N_STEADY))[0]

    model.N_BATCH = n_batch
    model.lr_eval_win = max(len(rwd_idx), len(stim_idx))

    model.I0[0] = 0
    model.I0[1] = 1
    model.I0[2] = 1
    model.I0[3] = 0
    model.I0[4] = 0
    go = model.init_ff_input()

    model.I0[0] = 0
    model.I0[1] = -1
    model.I0[2] = 1
    model.I0[3] = 0
    model.I0[4] = 0
    nogo = model.init_ff_input()

    ff_input = torch.cat((go, nogo))

    labels_go = torch.ones((model.N_BATCH, model.lr_eval_win), device=device)
    labels_nogo = torch.zeros((model.N_BATCH, model.lr_eval_win), device=device)
    labels = torch.cat((labels_go, labels_nogo))
    labels = labels.repeat((2, 1, 1)).transpose(0, 1)

    idx = dict(rwd_idx=rwd_idx, stim_idx=stim_idx, zero_idx=zero_idx)
    return ff_input, labels, idx

def make_dual_dataset(model, device, n_batch):
    steps = get_steps(model)

    rwd_idx = np.where(steps >= (model.N_STIM_ON[-1].cpu().numpy() - model.N_STEADY))[0]
    test_idx = rwd_idx

    cue_mask = (
        (steps >= (model.N_STIM_ON[2].cpu().numpy() - model.N_STEADY)) &
        (steps <= (model.N_STIM_OFF[3].cpu().numpy() - model.N_STEADY))
    )
    cue_idx = np.where(cue_mask)[0]

    gng_mask = (
        (steps >= (model.N_STIM_ON[1].cpu().numpy() - model.N_STEADY)) &
        (steps <= (model.N_STIM_ON[2].cpu().numpy() - model.N_STEADY))
    )
    gng_idx = np.where(gng_mask)[0]

    stim_mask = (
        (steps >= (model.N_STIM_ON[0].cpu().numpy() - model.N_STEADY)) &
        (steps <= (model.N_STIM_ON[-1].cpu().numpy() - model.N_STEADY))
    )
    stim_idx = np.where(stim_mask)[0]

    zero_idx = np.where(~(steps >= (model.N_STIM_ON[-1].cpu().numpy() - model.N_STEADY)) & ~cue_mask & ~stim_mask)[0]

    model.N_BATCH = n_batch
    model.lr_eval_win = max(len(rwd_idx), len(cue_idx), len(stim_idx), len(gng_idx))

    ff_input = []
    labels = np.zeros((4, 12, model.N_BATCH, model.lr_eval_win))

    l = 0
    for i in [-1, 1]:
        for j in [-1, 0, 1]:
            for k in [-1, 1]:
                model.I0[0] = i
                model.I0[1] = j
                model.I0[4] = k

                if k == 1:
                    labels[3, l] = 1
                if i == 1:
                    labels[1, l] = 1
                if i == k:
                    labels[0, l] = 1

                if j == 1:
                    model.I0[2] = 1.0
                    model.I0[3] = 0.0 * model.IF_RL
                    labels[2, l] = 1
                elif j == -1:
                    model.I0[2] = 1.0
                    model.I0[3] = 0.0
                else:
                    model.I0[2] = 0.0
                    model.I0[3] = 0.0

                ff_input.append(model.init_ff_input())
                l += 1

    labels = torch.tensor(labels, dtype=torch.float, device=device).reshape(4, -1, model.lr_eval_win).transpose(0, 1)
    ff_input = torch.vstack(ff_input)

    idx = dict(
        rwd_idx=rwd_idx,
        test_idx=test_idx,
        cue_idx=cue_idx,
        gng_idx=gng_idx,
        stim_idx=stim_idx,
        zero_idx=zero_idx,
    )
    return ff_input, labels, idx

#+begin_src python
import os
import json
import torch
import torch.optim as optim

def training_step(dataloader, model, loss_fn, optimizer, zero_grad=None):
    model.train()
    total_loss = 0.0

    for X, y in dataloader:
        X, y = X.to(model.device), y.to(model.device)

        optimizer.zero_grad()
        model(X)
        y_pred = model.readout
        loss = loss_fn(y_pred, y)
        loss.backward()

        if zero_grad is not None:
            try:
                if zero_grad == 'all':
                    model.low_rank.U.grad[:, :] = 0
                    model.low_rank.V.grad[:, :] = 0
                else:
                    model.low_rank.U.grad[:, zero_grad] = 0
                    model.low_rank.V.grad[:, zero_grad] = 0
            except Exception:
                pass

        optimizer.step()
        total_loss += loss.item()

    return total_loss / len(dataloader)

def validation_step(dataloader, model, loss_fn):
    model.eval()
    val_loss = 0.0
    size = len(dataloader.dataset)

    with torch.no_grad():
        for X, y in dataloader:
            X, y = X.to(model.device), y.to(model.device)
            model(X)
            y_pred = model.readout
            batch_loss = loss_fn(y_pred, y)
            val_loss += batch_loss.item() * X.size(0)

    return val_loss / size

def fit(model, train_loader, val_loader, criterion, optimizer, num_epochs=100, thresh=0.15, zero_grad=None):
    scheduler = optim.lr_scheduler.ExponentialLR(optimizer, gamma=0.9)

    history = {"train_loss": [], "val_loss": []}

    for epoch in range(num_epochs):
        train_loss = training_step(train_loader, model, criterion, optimizer, zero_grad=zero_grad)
        val_loss = validation_step(val_loader, model, criterion)

        scheduler.step()

        history["train_loss"].append(train_loss)
        history["val_loss"].append(val_loss)

        print(f"Epoch {epoch+1}/{num_epochs}, train={train_loss:.4f}, val={val_loss:.4f}")

        if train_loss < thresh and val_loss < thresh:
            print("Early stop: below threshold")
            break
        if val_loss > 300:
            print("Early stop: diverged")
            break
        if torch.isnan(torch.tensor(train_loss)):
            print("Early stop: NaN")
            break

    return history

def freeze_for_dpa(model):
    model.J_STP.requires_grad = True
    if model.LR_READOUT:
        for param in model.low_rank.linear.parameters():
            param.requires_grad = False
        model.low_rank.linear.bias.requires_grad = False

def freeze_for_gonogo(model):
    model.J_STP.requires_grad = False

def freeze_for_dual(model):
    model.J_STP.requires_grad = False

def save_run(model, save_path, history, args_dict):
    os.makedirs(os.path.dirname(save_path), exist_ok=True)
    torch.save(model.state_dict(), save_path)

    meta_path = save_path.replace(".pth", ".json")
    with open(meta_path, "w") as f:
        json.dump(
            {
                "args": args_dict,
                "history": history,
            },
            f,
            indent=2,
        )

#+begin_src python
import torch
import torch.nn as nn
import torch.nn.functional as F


def safe_mean(tensor):
    if tensor.numel() == 0:
        return torch.tensor(0.0, device=tensor.device, dtype=tensor.dtype)
    return tensor.mean()


class BCEOneClassLoss(nn.Module):
    def __init__(self):
        super().__init__()
        self.criterion = nn.BCEWithLogitsLoss(reduction="none")

    def forward(self, logits, targets, class_bal=1):
        bce = self.criterion(logits, targets.float())

        if class_bal == 0:
            mask1 = targets == 1
            pos_loss = safe_mean(bce[mask1])

            mask0 = targets == 0
            neutral_targets = torch.full_like(logits[mask0], 0.5)
            neutral_loss = safe_mean(self.criterion(logits[mask0], neutral_targets))
            return pos_loss + 0.01 * neutral_loss

        return safe_mean(bce)


class SignBCELoss(nn.Module):
    def __init__(self, alpha=0.5, thresh=1.0, class_bal=0):
        super().__init__()
        self.alpha = alpha
        self.thresh = thresh
        self.class_bal = class_bal
        self.bce_with_logits = BCEOneClassLoss()

    def forward(self, readout, targets):
        bce_loss = 0.0
        if self.alpha != 1.0:
            bce_loss = self.bce_with_logits(readout, targets, self.class_bal)

        sign_overlap = torch.sign(2 * targets - 1) * readout
        sign_loss = torch.zeros_like(sign_overlap)

        if self.alpha != 0:
            if self.class_bal == 0:
                mask0 = targets == 0
                if mask0.sum() > 0:
                    sign_loss[mask0] = 0.1 * torch.abs(sign_overlap[mask0])

                mask1 = targets == 1
                if mask1.sum() > 0:
                    sign_loss[mask1] = F.relu(self.thresh - sign_overlap[mask1])
            else:
                sign_loss = F.relu(self.thresh - sign_overlap)

        return (1 - self.alpha) * bce_loss + self.alpha * safe_mean(sign_loss)


class DualLoss(nn.Module):
    def __init__(
        self,
        device,
        alpha=1.0,
        thresh=5.0,
        stim_idx=None,
        gng_idx=None,
        cue_idx=None,
        test_idx=None,
        rwd_idx=None,
        zero_idx=None,
        read_idx=None,
        class_bal=None,
    ):
        super().__init__()
        self.alpha = alpha
        self.thresh = thresh
        self.class_bal = class_bal or [0]

        self.zero_idx = zero_idx if zero_idx is not None else []
        self.stim_idx = torch.tensor(stim_idx or [], dtype=torch.long, device=device)
        self.gng_idx = torch.tensor(gng_idx or [], dtype=torch.long, device=device)
        self.cue_idx = torch.tensor(cue_idx or [], dtype=torch.long, device=device)
        self.test_idx = torch.tensor(test_idx or [], dtype=torch.long, device=device)
        self.rwd_idx = torch.tensor(rwd_idx or [], dtype=torch.long, device=device)

        self.read_idx = read_idx or [-1]
        self.loss = SignBCELoss(self.alpha, self.thresh)
        self.l1loss = nn.SmoothL1Loss()

    def forward(self, readout, targets):
        loss = 0.0

        if len(self.zero_idx) > 0:
            zeros = torch.zeros_like(readout[:, self.zero_idx, 0])
            loss = self.l1loss(readout[:, self.zero_idx, self.read_idx[0]], zeros)

            if len(self.read_idx) > 1 and self.stim_idx.numel() > 0:
                stim0 = int(self.stim_idx[0].item())
                if stim0 > 1:
                    loss = loss + self.l1loss(
                        readout[:, :stim0 - 1, self.read_idx[1]],
                        zeros[:, :stim0 - 1],
                    )

        if self.test_idx.numel() != 0:
            self.loss.class_bal = 1
            loss = loss + self.loss(
                readout[:, self.test_idx, self.read_idx[-1]],
                targets[:, -1, : self.test_idx.shape[0]],
            )

        if self.cue_idx.numel() != 0:
            self.loss.class_bal = self.class_bal[3]
            loss = loss + self.loss(
                readout[:, self.cue_idx, self.read_idx[3]],
                targets[:, 2, : self.cue_idx.shape[0]],
            )

        if self.gng_idx.numel() != 0:
            self.loss.class_bal = 1
            loss = loss + self.loss(
                readout[:, self.gng_idx, self.read_idx[2]],
                targets[:, 2, : self.gng_idx.shape[0]],
            )

        if self.stim_idx.numel() != 0:
            self.loss.class_bal = 1
            loss = loss + self.loss(
                readout[:, self.stim_idx, self.read_idx[1]],
                targets[:, 1, : self.stim_idx.shape[0]],
            )

        if self.rwd_idx.numel() != 0:
            self.loss.class_bal = self.class_bal[0]
            loss = loss + self.loss(
                readout[:, self.rwd_idx, self.read_idx[0]],
                targets[:, 0, : self.rwd_idx.shape[0]],
            )

        return loss

    else:
        stratify = Y[:, 0].detach().cpu().numpy()

    X_train, X_val, Y_train, Y_val = train_test_split(
        X,
        Y,
        train_size=train_perc,
        stratify=stratify,
        shuffle=True,
    )

    print("train X:", tuple(X_train.shape), "val X:", tuple(X_val.shape))
    print("train Y:", tuple(Y_train.shape), "val Y:", tuple(Y_val.shape))

    train_loader = DataLoader(
        TensorDataset(X_train, Y_train),
        batch_size=batch_size,
        shuffle=True,

    val_loader = DataLoader(
        TensorDataset(X_val, Y_val),

def get_steps(model):
    return np.arange(0, model.N_STEPS - model.N_STEADY, model.N_WINDOW)


def make_dpa_dataset(model, device, n_batch, rank):
    steps = get_steps(model)

    mask_rwd = steps >= (model.N_STIM_ON[-1].detach().cpu().numpy() - model.N_STEADY)
    rwd_idx = np.where(mask_rwd)[0]

    test_idx = rwd_idx if rank == 3 else []

    mask_stim = (
        (steps >= (model.N_STIM_ON[0].detach().cpu().numpy() - model.N_STEADY)) &
        (steps < (model.N_STIM_ON[-1].detach().cpu().numpy() - model.N_STEADY))
    )
    stim_idx = np.where(mask_stim)[0]

    zero_idx = np.where(~mask_rwd)[0]

    model.N_BATCH = n_batch
    model.lr_eval_win = max(rwd_idx.shape[0], stim_idx.shape[0])

    ff_input = []
    labels = np.zeros((3, 4, model.N_BATCH, model.lr_eval_win), dtype=np.float32)

    l = 0
    for i in [-1, 1]:
        for k in [-1, 1]:
            model.I0[0] = i
            model.I0[4] = k

            if k == 1:
                labels[2, l] = 1.0

            if i == 1:
                labels[1, l] = 1.0

            if i == k:
                labels[0, l] = 1.0

            ff_input.append(model.init_ff_input())
            l += 1

    labels = (
        torch.tensor(labels, dtype=torch.float32, device=device)
        .reshape(3, -1, model.lr_eval_win)
        .transpose(0, 1)
    )
    ff_input = torch.vstack(ff_input)

    idx = {
        "rwd_idx": rwd_idx.tolist(),
        "stim_idx": stim_idx.tolist(),
        "test_idx": test_idx.tolist() if isinstance(test_idx, np.ndarray) else test_idx,
        "zero_idx": zero_idx.tolist(),
    }
    return ff_input, labels, idx


def make_gonogo_dataset(model, device, n_batch):
    steps = get_steps(model)

    rwd_mask = (
        (steps >= (model.N_STIM_ON[2].detach().cpu().numpy() - model.N_STEADY)) &
        (steps < (model.N_STIM_ON[4].detach().cpu().numpy() - model.N_STEADY))
    )
    rwd_idx = np.where(rwd_mask)[0]

    stim_mask = (
        (steps >= (model.N_STIM_ON[1].detach().cpu().numpy() - model.N_STEADY)) &
        (steps < (model.N_STIM_ON[2].detach().cpu().numpy() - model.N_STEADY))
    )
    stim_idx = np.where(stim_mask)[0]

    zero_idx = np.where(steps < (model.N_STIM_ON[1].detach().cpu().numpy() - model.N_STEADY))[0]

    model.N_BATCH = n_batch
    model.lr_eval_win = max(rwd_idx.shape[0], stim_idx.shape[0])

    model.I0[0] = 0
    model.I0[1] = 1
    model.I0[2] = 1
    model.I0[3] = 0
    model.I0[4] = 0
    go = model.init_ff_input()

    model.I0[0] = 0
    model.I0[1] = -1
    model.I0[2] = 1
    model.I0[3] = 0
    model.I0[4] = 0
    nogo = model.init_ff_input()

    ff_input = torch.cat((go, nogo), dim=0)

    labels_go = torch.ones((model.N_BATCH, model.lr_eval_win), device=device)
    labels_nogo = torch.zeros((model.N_BATCH, model.lr_eval_win), device=device)
    labels = torch.cat((labels_go, labels_nogo), dim=0)
    labels = labels.repeat((2, 1, 1)).transpose(0, 1)

    idx = {
        "rwd_idx": rwd_idx.tolist(),
        "stim_idx": stim_idx.tolist(),
        "zero_idx": zero_idx.tolist(),
    }
    return ff_input, labels, idx

import torch
import torch.optim as optim


def training_step(dataloader, model, loss_fn, optimizer, zero_grad=None):
    model.train()
    total_loss = 0.0
    total_batches = len(dataloader)

    for X, y in dataloader:
        X = X.to(model.device)
        y = y.to(model.device)

        optimizer.zero_grad()

        model(X)
        y_pred = model.readout
        loss = loss_fn(y_pred, y)

        loss.backward()

        if zero_grad is not None:
            try:
                if zero_grad == "all":
                    model.low_rank.U.grad[:, :] = 0
                    model.low_rank.V.grad[:, :] = 0
                else:
                    model.low_rank.U.grad[:, zero_grad] = 0
                    model.low_rank.V.grad[:, zero_grad] = 0
            except Exception:
                pass

        optimizer.step()
        total_loss += loss.item()

    return total_loss / total_batches


def validation_step(dataloader, model, loss_fn):
    size = len(dataloader.dataset)
    model.eval()
    val_loss = 0.0

    with torch.no_grad():
        for X, y in dataloader:
            X = X.to(model.device)
            y = y.to(model.device)

            model(X)
            y_pred = model.readout
            batch_loss = loss_fn(y_pred, y)
            val_loss += batch_loss.item() * X.size(0)

    val_loss /= size
    return val_loss


def fit(
    model,
    train_loader,
    val_loader,
    criterion,
    optimizer,
    num_epochs=100,
    thresh=0.15,
    zero_grad=None,
):
    scheduler = optim.lr_scheduler.ExponentialLR(optimizer, gamma=0.9)

    history = {
        "train_loss": [],
        "val_loss": [],
    }

    for epoch in range(num_epochs):
        train_loss = training_step(
            train_loader,
            model,
            criterion,
            optimizer,
            zero_grad=zero_grad,
        )
        val_loss = validation_step(val_loader, model, criterion)

        scheduler.step()

        history["train_loss"].append(float(train_loss))
        history["val_loss"].append(float(val_loss))

        print(
            f"Epoch {epoch + 1}/{num_epochs}, "
            f"Training Loss: {train_loss:.4f}, "
            f"Validation Loss: {val_loss:.4f}"
        )

        if val_loss < thresh and train_loss < thresh:
            print(f"Stopping: both losses below threshold ({train_loss:.4f}, {val_loss:.4f})")
            break

        if val_loss > 300:
            print(f"Stopping: validation loss too high ({val_loss:.4f})")
            break

        if torch.isnan(torch.tensor(train_loss)):
            print("Stopping: NaN loss")
            break

    return history


def freeze_for_dpa(model):
    model.J_STP.requires_grad = True

    if getattr(model, "LR_READOUT", False):
        for param in model.low_rank.linear.parameters():
            param.requires_grad = False
        model.low_rank.linear.bias.requires_grad = False


def freeze_for_gonogo(model):
    model.J_STP.requires_grad = False


def freeze_for_dual(model):
    model.J_STP.requires_grad = False


def print_trainable_parameters(model):
    for name, param in model.named_parameters():
        if param.requires_grad:
            print(name, tuple(param.shape))

        "task": task,
        "args": vars(args),
        "history": history,
    }

                "args": vars(args),
                "history": history,

def build_single_cmd(args, seed, gpu):
    outdir = Path(args.out_root) / f"seed{seed}"
    logfile = Path(args.log_root) / f"seed{seed}.log"

    cmd = [
        args.python, "-m", args.module,
        "--task", args.task,
        "--seed", str(seed),
        "--device", f"cuda:{gpu}",
        "--conf-name", args.conf_name,
        "--rank", str(args.rank),
        "--gain", str(args.gain),
        "--lr-ini", str(args.lr_ini),
        "--var-ff", str(args.var_ff),
        "--train-scale", str(args.train_scale),
        "--lr-type", str(args.lr_type),
        "--K", str(args.K),
        "--n-batch", str(args.n_batch),
        "--batch-size", str(args.batch_size),
        "--learning-rate", str(args.learning_rate),
        "--epochs", str(args.epochs),
        "--bce-alpha", str(args.bce_alpha),
        "--loss-thresh", str(args.loss_thresh),
        "--stop-loss", str(args.stop_loss),
        "--train-perc", str(args.train_perc),
    ]

    if args.task == "sequence":
        cmd += ["--outdir", str(outdir)]
        done_file = outdir / f"dual_seed{seed}.pth"
    else:
        outdir.mkdir(parents=True, exist_ok=True)
        save_path = outdir / f"{args.task}_seed{seed}.pth"
        cmd += ["--save-path", str(save_path)]
        done_file = save_path

    return cmd, logfile, done_file


def launch_process(cmd, logfile, gpu, dry_run=False):
    env = os.environ.copy()
    env["CUDA_VISIBLE_DEVICES"] = str(gpu)

    logfile.parent.mkdir(parents=True, exist_ok=True)

    pretty = " ".join(cmd)
    print(f"[launch gpu={gpu}] {pretty}")
    print(f"[log] {logfile}")

    if dry_run:
        return None

    f = open(logfile, "w")
    proc = subprocess.Popen(
        cmd,
        stdout=f,
        stderr=subprocess.STDOUT,
        env=env,
    )
    return {"proc": proc, "logfile": f, "gpu": gpu, "cmd": cmd}


def cleanup_finished(active):
    still_active = []
    for item in active:
        ret = item["proc"].poll()
        if ret is None:
            still_active.append(item)
        else:
            item["logfile"].close()
            status = "OK" if ret == 0 else f"FAIL({ret})"
            print(f"[done gpu={item['gpu']}] {status} :: {' '.join(item['cmd'])}")
    return still_active


def count_gpu_load(active, gpu):
    return sum(1 for item in active if str(item["gpu"]) == str(gpu))


def main():
    args = parse_args()

    seeds = list(range(args.start_seed, args.start_seed + args.n_runs))
    Path(args.out_root).mkdir(parents=True, exist_ok=True)
    Path(args.log_root).mkdir(parents=True, exist_ok=True)

    jobs = []
    for seed in seeds:
        for gpu in [None]:
            cmd, logfile, done_file = build_single_cmd(args, seed, gpu="0")  # placeholder
            jobs.append((seed, done_file))

    pending = []
    for seed in seeds:
        if args.task == "sequence":
            done_file = Path(args.out_root) / f"seed{seed}" / f"dual_seed{seed}.pth"
        else:
            done_file = Path(args.out_root) / f"seed{seed}" / f"{args.task}_seed{seed}.pth"

        if done_file.exists() and not args.overwrite:
            print(f"[skip] seed={seed} already done: {done_file}")
            continue
        pending.append(seed)

    print(f"Requested runs: {len(seeds)}")
    print(f"Pending runs:   {len(pending)}")
    print(f"GPUs:           {args.gpus}")
    print(f"Max/GPU:        {args.max_procs_per_gpu}")

    active = []
    pending_iter = iter(pending)

    while True:
        active = cleanup_finished(active)

        launched_any = False
        for gpu in args.gpus:
            while count_gpu_load(active, gpu) < args.max_procs_per_gpu:
                try:
                    seed = next(pending_iter)
                except StopIteration:
                    break

                cmd, logfile, done_file = build_single_cmd(args, seed, gpu)

                if done_file.exists() and not args.overwrite:
                    print(f"[skip] seed={seed} already done: {done_file}")
                    continue

                proc_info = launch_process(cmd, logfile, gpu, dry_run=args.dry_run)
                if proc_info is not None:
                    active.append(proc_info)
                launched_any = True

        if args.dry_run:
            break

        if not active and not launched_any:
            break

        time.sleep(args.poll_seconds)

    if not args.dry_run:
        while active:
            active = cleanup_finished(active)
            time.sleep(args.poll_seconds)

    print("[all done]")


if __name__ == "__main__":
    main()

**** Important note about GPUs

In your =run_dual.py=, you currently pass =--device cuda:X=.
In the launcher I also set:

#+begin_src python
env["CUDA_VISIBLE_DEVICES"] = str(gpu)
#+end_src

That can conflict conceptually if you pass global GPU ids and also mask devices.

***** Simplest fix
Use only one of these strategies.

I recommend:

- keep =--device cuda:0= always inside launched process,
- select physical GPU via =CUDA_VISIBLE_DEVICES=.

So in =launch_dual.py=, change:

#+begin_src python
"--device", f"cuda:{gpu}",
#+end_src

to:

#+begin_src python
"--device", "cuda:0",
#+end_src

because once =CUDA_VISIBLE_DEVICES=gpu=, that process sees its assigned GPU as local =cuda:0=.

So the better =build_single_cmd()= line is:

#+begin_src python
"--device", "cuda:0",
#+end_src

Please make that change.
