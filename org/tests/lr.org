    h = seconds // 3600
    m = (seconds % 3600) // 60
    s = seconds % 60
    return h, m, s

def plot_eigen(W):
    eigenvalues = torch.linalg.eigvals(W).cpu().numpy()
    real_parts = eigenvalues.real
    imag_parts = eigenvalues.imag

    plt.figure(figsize=(5, 5))
    plt.scatter(real_parts, imag_parts, s=2, alpha=0.5)
    plt.xlabel('Real Part')
    plt.ylabel('Imaginary Part')
    plt.axhline(y=0, color='k', linestyle='--', linewidth=0.5)
    plt.axvline(x=0, color='k', linestyle='--', linewidth=0.5)

** Stimuli Aligned with Low-Rank Vectors
*** Build Custom FF Input

We inject stimuli proportional to the right vectors $V_{:,0}$ and $V_{:,1}$

M0 = model.config.M0

** Overlap Analysis
*** Compute Overlaps

The overlap (order parameter) $m_k(t) = \frac{1}{N_E} \sum_i r_i(t) V_{i,k}$
measures how much the population activity projects onto each low-rank direction.

    stim_scaled = stim_strength * stim_dir * scale
    ee = model.config.slices[0]
    t_on = int(2.0 / DT) + N_STEADY + N_HEBB
    t_off = int(12.0 / DT) + N_STEADY + N_HEBB
    t_off = min(t_off, ff_i.shape[1])  # guard against overshoot
    ff_i[:, t_on:t_off, ee] += stim_scaled.unsqueeze(0)

    out_i = model(ff_i, return_outputs=True)
    r_i = out_i.rates[0].cpu()
    V_cpu_f = torch.tensor(V, dtype=r_i.dtype)
    m_i = (r_i @ V_cpu_f) / Ne
    # Take last 20% as steady state
    ss_slice = slice(int(0.8 * r_i.shape[0]), r_i.shape[0])
    overlap_ss[i] = m_i[ss_slice].mean(dim=0).numpy()
    mean_rate_ss[i] = r_i[ss_slice].mean().numpy()

print('Sweep done.')
