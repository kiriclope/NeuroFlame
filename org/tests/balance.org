def convert_seconds(seconds):
    h = seconds // 3600
    m = (seconds % 3600) // 60
    s = seconds % 60
    return h, m, s

def plot_eigen(W):
    eigenvalues = torch.linalg.eigvals(W).cpu().numpy()
    real_parts = eigenvalues.real
    imag_parts = eigenvalues.imag

    plt.figure(figsize=(5, 5))
    plt.scatter(real_parts, imag_parts, s=2, alpha=0.5)
    plt.xlabel('Real Part')
    plt.ylabel('Imaginary Part')
    plt.axhline(y=0, color='k', linestyle='--', linewidth=0.5)
    plt.axvline(x=0, color='k', linestyle='--', linewidth=0.5)
    plt.title('Eigenvalue Spectrum')
    plt.show()

def safe_corrcoef(a, b):
    """Pearson correlation that returns NaN for constant inputs."""
    if np.std(a) < 1e-12 or np.std(b) < 1e-12:
        return np.nan
    return np.corrcoef(a, b)[0, 1]

#+begin_src ipython
n_pairs = 500

def sample_corr(rates, idx_a, idx_b, n_pairs):
    corrs = []
    idx_a = np.array(list(idx_a))
    idx_b = np.array(list(idx_b))
    if len(idx_a) < 2 or len(idx_b) < 2:
        return corrs
    for _ in range(n_pairs):
        i = np.random.choice(idx_a)
        j = np.random.choice(idx_b)
        if i == j:
            continue
        c = safe_corrcoef(rates[:, i], rates[:, j])
        if np.isfinite(c):
            corrs.append(c)
    return corrs

ee_corr = sample_corr(rates_ts, range(Ne_ts), range(Ne_ts), n_pairs)

fig, ax = plt.subplots(figsize=(width, height))
if ee_corr:
    ax.hist(ee_corr, bins=40, alpha=0.6, label='E-E', density=True)
