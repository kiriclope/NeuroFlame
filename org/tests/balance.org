def convert_seconds(seconds):
    h = seconds // 3600
    m = (seconds % 3600) // 60
    s = seconds % 60
    return h, m, s

def plot_eigen(W):
    eigenvalues = torch.linalg.eigvals(W).cpu().numpy()
    real_parts = eigenvalues.real
    imag_parts = eigenvalues.imag

    plt.figure(figsize=(5, 5))
    plt.scatter(real_parts, imag_parts, s=2, alpha=0.5)
    plt.xlabel('Real Part')
    plt.ylabel('Imaginary Part')
    plt.axhline(y=0, color='k', linestyle='--', linewidth=0.5)
    plt.axvline(x=0, color='k', linestyle='--', linewidth=0.5)
    plt.title('Eigenvalue Spectrum')
    plt.show()

def safe_corrcoef(a, b):
    """Pearson correlation that returns NaN for constant inputs."""
    if np.std(a) < 1e-12 or np.std(b) < 1e-12:
        return np.nan
    return np.corrcoef(a, b)[0, 1]

def get_steady_state_step(model):
    """Return the ff_input time index corresponding to the last collected output step."""
    N_STEADY = model.config.N_STEADY
    N_HEBB = model.config.N_HEBB
    N_WINDOW = model.config.N_WINDOW
    return N_STEADY + N_HEBB  # start of collection window

def get_ff_for_output_step(model, rates, output_step=-1):
    """Map a collected output step index to the corresponding ff_input window average."""
    N_STEADY = model.config.N_STEADY
    N_HEBB = model.config.N_HEBB
    N_WINDOW = model.config.N_WINDOW
    n_output_steps = rates.shape[0]
    if output_step < 0:
        output_step = n_output_steps + output_step
    ff_start = N_STEADY + N_HEBB + output_step * N_WINDOW
    ff_end = ff_start + N_WINDOW
    return ff_start, ff_end

Here we decompose the total input to each E neuron into its constituent parts:
- FF input (external drive)
- E→E recurrent excitation
- I→E recurrent inhibition

#+begin_src ipython
if rec_input is not None:

    # Get steady-state rates and FF input
    rates_final = rates[-1]  # (N_NEURON,)

    # E→E: excitatory presynaptic (rows 0:Ne) → postsynaptic E (cols 0:Ne)

    # Recurrent input to E neuron i = sum_j rates[j] * Wab_T[j, i]
    rec_EtoE = rates_final[:Ne] @ Wab_T[:Ne, :Ne]   # (Ne,) — E→E input per E neuron
