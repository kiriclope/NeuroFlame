#+begin_src ipython
  def convert_seconds(seconds):
      h = seconds // 3600
      m = (seconds % 3600) // 60
      s = seconds % 60
      return h, m, s

  def decode_bump(rates, axis=-1):
      """Decode bump position from population activity via circular mean."""
      N = rates.shape[axis]
      theta = np.linspace(0, 2 * np.pi, N, endpoint=False)
      r = np.moveaxis(rates, axis, -1)
      m0 = r.mean(axis=-1)
      c = (r * np.cos(theta)).mean(axis=-1)
      s = (r * np.sin(theta)).mean(axis=-1)
      m1 = np.sqrt(c**2 + s**2)
      phi = np.arctan2(s, c) % (2 * np.pi)
      return m0, m1, phi

  # ── Analytical steady-state for Markram STP ──
  def stp_steady_state(r, USE, TAU_FAC, TAU_REC):
      """Return (u_ss, x_ss, A_ss) for constant rate r.

      From the continuous-time Markram equations at steady state:
        du/dt = (USE - u)/TAU_FAC + USE*(1-u)*r = 0
        dx/dt = (1 - x)/TAU_REC   - u*x*r       = 0
      """
      r = np.asarray(r, dtype=float)
      USE = float(USE)
      u_ss = USE * (1.0 + TAU_FAC * r) / (1.0 + USE * TAU_FAC * r)
      x_ss = 1.0 / (1.0 + u_ss * TAU_REC * r)
      return u_ss, x_ss, u_ss * x_ss * r

  def rebuild_model_batch(model, n_batch):
      """Rebuild model internals so end_indices/start_indices match a new batch size.

      This is necessary when concatenating ff_inputs from multiple conditions
      into a single batch that is larger than the original N_BATCH.
      """
      cfg = model.config
      old_nb = cfg.N_BATCH
      cfg.N_BATCH = n_batch

      # Recompute random_shifts and indices for the new batch size
      cfg.random_shifts = torch.zeros((n_batch,), device=cfg.device)
      cfg.start_indices = cfg.N_STIM_ON.unsqueeze(-1) + cfg.random_shifts
      cfg.end_indices = cfg.N_STIM_OFF.unsqueeze(-1) + cfg.random_shifts

      # Rebuild thresh for the new batch size
      cfg.thresh = torch.ones(n_batch, cfg.N_NEURON, device=cfg.device)
      for i_pop in range(cfg.N_POP):
          cfg.thresh[:, cfg.slices[i_pop]] = cfg.THRESH[i_pop]

      # Rebuild sub-configs that depend on batch size
      from src.config.types import (
          SimulationState, StateConfig, SimulationConfig, OutputConfig,
      )
      cfg.sim_state = SimulationState.from_raw(cfg)
      cfg.state_cfg = StateConfig.from_raw(
          cfg, cfg.geo, cfg.time, cfg.flags, cfg.sim_state,
      )
      cfg.output_cfg = OutputConfig.from_raw(
          cfg, cfg.geo, cfg.time, cfg.sim_state,
      )
      cfg.sim_cfg = SimulationConfig.from_raw(
          cfg, cfg.geo, cfg.time, cfg.flags,
          cfg.weight_cfg.trainable, cfg.sim_state,
      )

      # Update the network's references
      model.state_manager.cfg = cfg.state_cfg
      model.sim = cfg.sim_cfg
