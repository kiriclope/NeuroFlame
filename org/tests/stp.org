#+begin_src ipython
  def convert_seconds(seconds):
      h = seconds // 3600
      m = (seconds % 3600) // 60
      s = seconds % 60
      return h, m, s

  def decode_bump(rates, axis=-1):
      """Decode bump position from population activity via circular mean."""
      N = rates.shape[axis]
      theta = np.linspace(0, 2 * np.pi, N, endpoint=False)
      r = np.moveaxis(rates, axis, -1)
      m0 = r.mean(axis=-1)
      c = (r * np.cos(theta)).mean(axis=-1)
      s = (r * np.sin(theta)).mean(axis=-1)
      m1 = np.sqrt(c**2 + s**2)
      phi = np.arctan2(s, c) % (2 * np.pi)
      return m0, m1, phi

  # ── Analytical steady-state for Markram STP ──
  def stp_steady_state(r, USE, TAU_FAC, TAU_REC):
      """Return (u_ss, x_ss, A_ss) for constant rate r."""
      r = np.asarray(r, dtype=float)
      USE = float(USE)
      denom_u = 1.0 + USE * TAU_FAC * r
      u_ss = USE / denom_u + USE * TAU_FAC * r / denom_u
      # More precisely from du/dt=0, dx/dt=0:
      # u_ss = USE * (1 + TAU_FAC * r) / (1 + USE * TAU_FAC * r)  [wrong]
      # Correct: from the Euler eqs at steady state:
      #   u = USE + USE*(1-u)*r*DT  =>  in continuous limit:
      #   0 = (USE - u)/TAU_FAC + USE*(1-u)*r
      #   u = USE*(1 + TAU_FAC*r) / (1 + USE*TAU_FAC*r)
      u_ss = USE * (1.0 + TAU_FAC * r) / (1.0 + USE * TAU_FAC * r)
      # 0 = (1 - x)/TAU_REC - u*x*r
      # x = 1 / (1 + u*TAU_REC*r)
      x_ss = 1.0 / (1.0 + u_ss * TAU_REC * r)
      return u_ss, x_ss, u_ss * x_ss * r

** 2. Steady-State $A_{ux}$ vs Firing Rate (with Analytical Comparison)

Sweep input rate, let STP converge, compare to closed-form steady state.

#+begin_src ipython
  tau_list = torch.tensor([1.0, 0.75, 0.5, 0.25])
  N_BATCH = tau_list.shape[0]
  N_NEURON = 10
  N_RATES = 100
  N_STEPS = 500   # increased for convergence
  DT = 0.01
  USE_val = 0.03
  TAU_REC_val = 0.2

  stp = Plasticity(
      USE=USE_val, TAU_FAC=tau_list, TAU_REC=TAU_REC_val,
      DT=DT, size=(N_BATCH, N_NEURON),
      device=DEVICE, STP_TYPE='markram',
  )

  A_ss, u_ss, x_ss = [], [], []

  for i in range(N_RATES):
      # Reset STP state for each rate
      stp.u_stp = stp.USE.expand(N_BATCH, N_NEURON).clone()
      stp.x_stp = torch.ones((N_BATCH, N_NEURON), device=DEVICE)
      # Constant rate + small noise for stability
      rate_val = float(i)
      rates = rate_val + 0.1 * torch.randn((N_BATCH, N_NEURON), device=DEVICE)
      rates = torch.relu(rates)
      for j in range(N_STEPS):
          _ = stp(rates)
      A_ss.append((stp.u_stp * stp.x_stp * rates).mean(-1).detach())
      u_ss.append(stp.u_stp.mean(-1).detach())
      x_ss.append(stp.x_stp.mean(-1).detach())

  A_ss = torch.stack(A_ss, dim=1).cpu().numpy()   # (N_BATCH, N_RATES)
  u_ss = torch.stack(u_ss, dim=1).cpu().numpy()
  x_ss = torch.stack(x_ss, dim=1).cpu().numpy()
  print(f'A_ss: {A_ss.shape} | u_ss: {u_ss.shape} | x_ss: {x_ss.shape}')
