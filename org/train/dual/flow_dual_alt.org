def del_tensor(obj):
    try:
        if hasattr(obj, "device"):
            device = obj.device
        elif hasattr(obj, "parameters"):
            params = list(obj.parameters())
            device = params[0].device if len(params) > 0 else None
        else:
            device = None
    except Exception:
        device = None

    del obj
    gc.collect()

    if torch.cuda.is_available():
        torch.cuda.empty_cache()
        if device is not None and "cuda" in str(device):
            try:
                torch.cuda.synchronize(device)
            except Exception:
                pass

def run_grid(GRID_RANGE, seed, task, **kwargs):
    """
    Evaluate model on 4 quadrants of the low-rank grid.
    Returns shape: (4 * GRID_SIZE^2, time, neurons)
    """
    GRID_LIST = [[-GRID_RANGE, 0], [0, GRID_RANGE]]

    rates_grid = []
    with torch.no_grad():
        for GRID_X_RANGE in GRID_LIST:
            for GRID_Y_RANGE in GRID_LIST:
                model = init_model(task, seed, **kwargs)

                model.GRID_X_RANGE = GRID_X_RANGE
                model.GRID_Y_RANGE = GRID_Y_RANGE
                model.N_BATCH = int(model.GRID_SIZE * model.GRID_SIZE)

                ff_input = model.init_ff_input()
                rates = model(ff_input, RET_REC=0).detach().cpu().numpy()
                rates_grid.append(rates)

                del_tensor(ff_input)
                del_tensor(model)

    return np.vstack(rates_grid)

def get_low_rank(rates, model, IF_REC=0):
    """
    Project rates onto low-rank vectors using an orthonormalized basis.

    Parameters
    ----------
    rates : torch.Tensor
        Shape (..., n_neurons)
    IF_REC : int
        If 0, use V vectors.
        If 1, use U vectors on recurrent input.

    Returns
    -------
    overlaps : np.ndarray
        Shape (..., n_vec)
    vec : np.ndarray
        Shape (n_vec, n_neurons)
    """
    def orthonormalize(vectors):
        basis = []
        for v in vectors:
            v = v.clone()
            for b in basis:
                v = v - torch.dot(v, b) * b
            nrm = torch.linalg.norm(v)
            if nrm > 1e-12:
                basis.append(v / nrm)
        if len(basis) == 0:
            raise ValueError("Could not build orthonormal basis.")
        return torch.stack(basis)

    if IF_REC == 0:
        vecs = [model.low_rank.V.T[0], model.low_rank.V.T[1]]
        try:
            vecs.append(model.low_rank.V.T[2])
        except Exception:
            pass

        vec = orthonormalize(vecs)
        vec = vec.to(rates.device, rates.dtype)

        overlaps = rates @ vec.T / model.Na[0]

    else:
        vecs = [model.low_rank.U.T[0], model.low_rank.U.T[1]]
        vec = orthonormalize(vecs)
        vec = vec.to(model.rec_input.device, model.rec_input.dtype)

        overlaps = model.rec_input[0, :, :] @ vec.T

    return overlaps.detach().cpu().numpy(), vec.detach().cpu().numpy()

def compute_flow_field(overlaps, window, size=220, sigma=1.0, method='linear', q_bounds=(1, 99)):
    """
    Compute 2D empirical flow field from low-rank trajectories.

    Parameters
    ----------
    overlaps : ndarray, shape (n_traj, n_time, >=2)
    window : int
        Discard times before this index.
    size : int
        Grid resolution.
    sigma : float
        Gaussian smoothing on interpolated vector field.
    method : str
        scipy.interpolate.griddata method.
    q_bounds : tuple
        Percentile bounds for robust grid limits.

    Returns
    -------
    dict
        Contains grid, vector field, speed, and raw trajectories.
    """
    traj = np.asarray(overlaps)
    assert traj.ndim == 3, f"Expected (n_traj, n_time, n_dim), got {traj.shape}"
    assert traj.shape[-1] >= 2, f"Need at least 2 dims, got {traj.shape}"

    x = traj[:, window:, 0]
    y = traj[:, window:, 1]

    dx = np.diff(x, axis=1)
    dy = np.diff(y, axis=1)

    x0 = x[:, :-1]
    y0 = y[:, :-1]

    x_flat = x0.ravel()
    y_flat = y0.ravel()
    dx_flat = dx.ravel()
    dy_flat = dy.ravel()

    valid = (
        np.isfinite(x_flat) &
        np.isfinite(y_flat) &
        np.isfinite(dx_flat) &
        np.isfinite(dy_flat)
    )

    x_flat = x_flat[valid]
    y_flat = y_flat[valid]
    dx_flat = dx_flat[valid]
    dy_flat = dy_flat[valid]

    if len(x_flat) == 0:
        raise ValueError("No valid trajectory points found.")

    q0, q1 = q_bounds
    x_lo, x_hi = np.percentile(x_flat, [q0, q1])
    y_lo, y_hi = np.percentile(y_flat, [q0, q1])

    x_pad = 0.10 * (x_hi - x_lo + 1e-8)
    y_pad = 0.10 * (y_hi - y_lo + 1e-8)

    xg = np.linspace(x_lo - x_pad, x_hi + x_pad, size)
    yg = np.linspace(y_lo - y_pad, y_hi + y_pad, size)
    xi, yi = np.meshgrid(xg, yg)

    points = np.column_stack([x_flat, y_flat])

    ui = griddata(points, dx_flat, (xi, yi), method=method, fill_value=np.nan)
    vi = griddata(points, dy_flat, (xi, yi), method=method, fill_value=np.nan)

    nanmask = ~np.isfinite(ui) | ~np.isfinite(vi)
    if np.any(nanmask):
        ui_near = griddata(points, dx_flat, (xi, yi), method='nearest')
        vi_near = griddata(points, dy_flat, (xi, yi), method='nearest')
        ui[nanmask] = ui_near[nanmask]
        vi[nanmask] = vi_near[nanmask]

    if sigma is not None and sigma > 0:
        ui = gaussian_filter(ui, sigma=sigma)
        vi = gaussian_filter(vi, sigma=sigma)

    speed = np.sqrt(ui**2 + vi**2)

    return {
        "xg": xg,
        "yg": yg,
        "xi": xi,
        "yi": yi,
        "ui": ui,
        "vi": vi,
        "speed": speed,
        "x": x,
        "y": y,
        "dx": dx,
        "dy": dy,
    }

def get_fp(overlaps, window, n_clusters=5, random_state=0):
    """
    Estimate fixed points from trajectory endpoints via k-means.

    Parameters
    ----------
    overlaps : ndarray, shape (n_traj, n_time, >=2)
    window : int
    n_clusters : int

    Returns
    -------
    centers : ndarray, shape (k, 2)
    endpoints : ndarray, shape (n_traj_valid, 2)
    """
    traj = np.asarray(overlaps)
    assert traj.ndim == 3 and traj.shape[-1] >= 2, traj.shape

    x = traj[:, window:, 0]
    y = traj[:, window:, 1]

    endpoints = np.column_stack([x[:, -1], y[:, -1]])
    valid = np.isfinite(endpoints).all(axis=1)
    endpoints = endpoints[valid]

    n_clusters = min(n_clusters, len(endpoints))
    if n_clusters < 1:
        raise ValueError("No endpoints available for clustering.")

    kmeans = KMeans(n_clusters=n_clusters, random_state=random_state, n_init=20)
    kmeans.fit(endpoints)
    centers = np.asarray(kmeans.cluster_centers_)

    return centers, endpoints

def plot_field(
    overlaps,
    ax,
    window,
    size=220,
    sigma=1.0,
    method='linear',
    q_bounds=(1, 99),
    show_stream=True,
    show_quiver=False,
    show_traj=False,
    show_fp=True,
    n_clusters=5,
    cbar=False,
    cmap='coolwarm',
    title=None,
    symmetric=True,
    equal_axes=True,
):
    """
    Plot low-rank empirical flow field.
    """
    out = compute_flow_field(
        overlaps,
        window=window,
        size=size,
        sigma=sigma,
        method=method,
        q_bounds=q_bounds,
        symmetric=symmetric,
        equal_axes=equal_axes,
    )

    xg = out["xg"]
    yg = out["yg"]
    xi = out["xi"]
    yi = out["yi"]
    ui = out["ui"]
    vi = out["vi"]
    speed = out["speed"]
    x = out["x"]
    y = out["y"]

    speed_z = (speed - np.nanmean(speed)) / (np.nanstd(speed) + 1e-6)
    finite = np.isfinite(speed_z)
    if np.any(finite):
        vmin, vmax = np.nanpercentile(speed_z[finite], [5, 95])
    else:
        vmin, vmax = -1.0, 1.0
    norm = mpl.colors.Normalize(vmin=vmin, vmax=vmax)

    hm = ax.pcolormesh(
        xi,
        yi,
        speed_z,
        cmap=cmap,
        shading='auto',
        norm=norm
    )

    if show_stream:
        ax.streamplot(
            xg,
            yg,
            np.nan_to_num(ui, nan=0.0),
            np.nan_to_num(vi, nan=0.0),
            color=(1, 1, 1, 0.6),
            density=1.0,
            linewidth=0.75,
            arrowsize=0.8
        )

    if show_quiver:
        step = max(1, size // 25)
        Xq = xi[::step, ::step]
        Yq = yi[::step, ::step]
        Uq = ui[::step, ::step]
        Vq = vi[::step, ::step]
        mask = np.isfinite(Uq) & np.isfinite(Vq)

        ax.quiver(
            Xq[mask],
            Yq[mask],
            Uq[mask],
            Vq[mask],
            color=(1, 1, 1, 0.7),
            angles='xy',
            scale_units='xy',
            scale=None,
            width=0.003
        )

    if show_traj:
        for i in range(x.shape[0]):
            ax.plot(x[i], y[i], color='k', alpha=0.08, lw=0.5)

    if show_fp:
        centers, endpoints = get_fp(overlaps, window, n_clusters=n_clusters)
        ax.scatter(endpoints[:, 0], endpoints[:, 1], s=8, color='k', alpha=0.20, zorder=4)
        ax.scatter(centers[:, 0], centers[:, 1], s=90, color='w', edgecolor='k', zorder=5)

    ax.set_xlim(*out["xlim"])
    ax.set_ylim(*out["ylim"])
    ax.set_xlabel('Sample Overlap')
    ax.set_ylabel('Choice Overlap')
    ax.set_aspect('equal')

    if title is not None:
        ax.set_title(title)

    if cbar:
        add_cbar(ax, hm, label='Normalized speed', size='4%', pad=0.05)

    return out

def compute_binned_flow_field(
    overlaps,
    window,
    size=60,
    sigma=1.0,
    q_bounds=(1, 99),
    min_count=1,
    symmetric=True,
    equal_axes=True,
):
    """
    Compute a 2D binned drift field from low-rank trajectories.
    """
    traj = np.asarray(overlaps)
    assert traj.ndim == 3, f"Expected (n_traj, n_time, n_dim), got {traj.shape}"
    assert traj.shape[-1] >= 2, f"Need at least 2 dims, got {traj.shape}"

    x = traj[:, window:, 0]
    y = traj[:, window:, 1]

    dx = np.diff(x, axis=1)
    dy = np.diff(y, axis=1)

    x0 = x[:, :-1]
    y0 = y[:, :-1]

    x_flat = x0.ravel()
    y_flat = y0.ravel()
    dx_flat = dx.ravel()
    dy_flat = dy.ravel()

    valid = (
        np.isfinite(x_flat) &
        np.isfinite(y_flat) &
        np.isfinite(dx_flat) &
        np.isfinite(dy_flat)
    )

    x_flat = x_flat[valid]
    y_flat = y_flat[valid]
    dx_flat = dx_flat[valid]
    dy_flat = dy_flat[valid]

    if len(x_flat) == 0:
        raise ValueError("No valid trajectory points found.")

    q0, q1 = q_bounds
    x_lo, x_hi = np.percentile(x_flat, [q0, q1])
    y_lo, y_hi = np.percentile(y_flat, [q0, q1])

    if symmetric:
        x_rad = max(abs(x_lo), abs(x_hi))
        y_rad = max(abs(y_lo), abs(y_hi))

        if equal_axes:
            rad = max(x_rad, y_rad)
            x_lo, x_hi = -rad, rad
            y_lo, y_hi = -rad, rad
        else:
            x_lo, x_hi = -x_rad, x_rad
            y_lo, y_hi = -y_rad, y_rad

        pad = 0.10 * max(x_hi - x_lo, y_hi - y_lo, 1e-8)
        x_lo, x_hi = x_lo - pad / 2, x_hi + pad / 2
        y_lo, y_hi = y_lo - pad / 2, y_hi + pad / 2
    else:
        x_pad = 0.10 * (x_hi - x_lo + 1e-8)
        y_pad = 0.10 * (y_hi - y_lo + 1e-8)
        x_lo, x_hi = x_lo - x_pad, x_hi + x_pad
        y_lo, y_hi = y_lo - y_pad, y_hi + y_pad

    x_edges = np.linspace(x_lo, x_hi, size)
    y_edges = np.linspace(y_lo, y_hi, size)

    sum_u, _, _ = np.histogram2d(y_flat, x_flat, bins=[y_edges, x_edges], weights=dx_flat)
    sum_v, _, _ = np.histogram2d(y_flat, x_flat, bins=[y_edges, x_edges], weights=dy_flat)
    count, _, _ = np.histogram2d(y_flat, x_flat, bins=[y_edges, x_edges])

    with np.errstate(invalid='ignore', divide='ignore'):
        ui = sum_u / count
        vi = sum_v / count

    bad = count < min_count
    ui[bad] = np.nan
    vi[bad] = np.nan

    ui0 = np.nan_to_num(ui, nan=0.0)
    vi0 = np.nan_to_num(vi, nan=0.0)

    if sigma is not None and sigma > 0:
        ui0 = gaussian_filter(ui0, sigma=sigma)
        vi0 = gaussian_filter(vi0, sigma=sigma)
        count_s = gaussian_filter(count.astype(float), sigma=sigma)
    else:
        count_s = count.astype(float)

    support = count_s > 0
    ui = np.where(support, ui0, np.nan)
    vi = np.where(support, vi0, np.nan)

    speed = np.sqrt(ui**2 + vi**2)

    xc = 0.5 * (x_edges[:-1] + x_edges[1:])
    yc = 0.5 * (y_edges[:-1] + y_edges[1:])
    xi, yi = np.meshgrid(xc, yc)

    return {
        "xc": xc,
        "yc": yc,
        "xi": xi,
        "yi": yi,
        "ui": ui,
        "vi": vi,
        "speed": speed,
        "count": count,
        "x": x,
        "y": y,
        "dx": dx,
        "dy": dy,
        "xlim": (x_lo, x_hi),
        "ylim": (y_lo, y_hi),
    }

def plot_binned_field(
    overlaps,
    ax,
    window,
    size=60,
    sigma=1.0,
    q_bounds=(1, 99),
    min_count=1,
    show_stream=True,
    show_quiver=False,
    show_traj=False,
    show_fp=True,
    n_clusters=5,
    cbar=False,
    cmap='coolwarm',
    title=None,
    show_count_contours=False,
    symmetric=True,
    equal_axes=True,
):
    """
    Plot 2D binned drift field.
    """
    out = compute_binned_flow_field(
        overlaps,
        window=window,
        size=size,
        sigma=sigma,
        q_bounds=q_bounds,
        min_count=min_count,
        symmetric=symmetric,
        equal_axes=equal_axes,
    )

    xc = out["xc"]
    yc = out["yc"]
    xi = out["xi"]
    yi = out["yi"]
    ui = out["ui"]
    vi = out["vi"]
    speed = out["speed"]
    count = out["count"]
    x = out["x"]
    y = out["y"]

    speed_z = (speed - np.nanmean(speed)) / (np.nanstd(speed) + 1e-6)
    finite = np.isfinite(speed_z)
    if np.any(finite):
        vmin, vmax = np.nanpercentile(speed_z[finite], [5, 95])
    else:
        vmin, vmax = -1.0, 1.0
    norm = mpl.colors.Normalize(vmin=vmin, vmax=vmax)

    hm = ax.pcolormesh(
        xc,
        yc,
        speed_z,
        cmap=cmap,
        shading='auto',
        norm=norm,
    )

    if show_stream:
        u_plot = np.nan_to_num(ui, nan=0.0)
        v_plot = np.nan_to_num(vi, nan=0.0)

        ax.streamplot(
            xc,
            yc,
            u_plot,
            v_plot,
            color=(1, 1, 1, 0.65),
            density=1.0,
            linewidth=0.8,
            arrowsize=0.9,
        )

    if show_quiver:
        step = max(1, len(xc) // 20)
        Xq = xi[::step, ::step]
        Yq = yi[::step, ::step]
        Uq = ui[::step, ::step]
        Vq = vi[::step, ::step]
        mask = np.isfinite(Uq) & np.isfinite(Vq)

        ax.quiver(
            Xq[mask],
            Yq[mask],
            Uq[mask],
            Vq[mask],
            color=(1, 1, 1, 0.75),
            angles='xy',
            scale_units='xy',
            scale=None,
            width=0.003,
        )

    if show_traj:
        for i in range(x.shape[0]):
            ax.plot(x[i], y[i], color='k', alpha=0.06, lw=0.5)

    if show_fp:
        centers, endpoints = get_fp(overlaps, window, n_clusters=n_clusters)
        ax.scatter(endpoints[:, 0], endpoints[:, 1], s=8, color='k', alpha=0.20, zorder=4)
        ax.scatter(centers[:, 0], centers[:, 1], s=90, color='w', edgecolor='k', zorder=5)

    if show_count_contours:
        ax.contour(xi, yi, count, levels=5, colors='k', alpha=0.25, linewidths=0.5)

    ax.set_xlim(*out["xlim"])
    ax.set_ylim(*out["ylim"])
    ax.set_xlabel('Sample Overlap')
    ax.set_ylabel('Choice Overlap')
    ax.set_aspect('equal')

    if title is not None:
        ax.set_title(title)

    if cbar:
        add_cbar(ax, hm, label='Normalized speed', size='4%', pad=0.05)

    return out
