from dataclasses import dataclass
import numpy as np
import torch


@dataclass(frozen=True)
class TaskTiming:
    stim_on: list[float]
    stim_off: list[float]
    t_steps: float
    dt: float

    @property
    def n_stim_on(self) -> torch.Tensor:
        return torch.tensor([int(t / self.dt) for t in self.stim_on], dtype=torch.long)

    @property
    def n_stim_off(self) -> torch.Tensor:
        return torch.tensor([int(t / self.dt) for t in self.stim_off], dtype=torch.long)

    @property
    def n_steps(self) -> int:
        return int(self.t_steps / self.dt)

    @property
    def x_time(self) -> np.ndarray:
        return np.linspace(0, self.n_steps * self.dt, self.n_steps)

def generate_dpa_trials(
    n_trials: int,
    timing: TaskTiming,
    input_size: int = 8,
    rank: int = 2,
    noise: float = 0.1,
    baseline_value: float = 1e-3,
):
    n_steps = timing.n_steps
    n_on = timing.n_stim_on
    n_off = timing.n_stim_off

    inputs = noise * torch.randn(n_trials, n_steps, input_size)
    targets = torch.zeros(n_trials, n_steps, rank)

    # First cue: A vs B.
    idx_A = torch.rand(n_trials) > 0.5
    idx_B = ~idx_A

    # Second cue: C vs D.
    idx_C = torch.rand(n_trials) > 0.5
    idx_D = ~idx_C

    # sample input
    inputs[idx_A, n_on[0]:n_off[0], 0] += 1.0
    inputs[idx_B, n_on[0]:n_off[0], 1] += 1.0

    # test input
    inputs[idx_C, n_on[1]:n_off[1], 2] += 1.0
    inputs[idx_D, n_on[1]:n_off[1], 3] += 1.0

    # Paired: AC or BD.
    idx_pair = (idx_A & idx_C) | (idx_B & idx_D)

    targets[..., 0] = torch.nan
    if baseline_value is not None:
        targets[:, :n_on[0]] = baseline_value

    targets[idx_pair, n_on[1]:, -1] = 1.0
    targets[~idx_pair, n_on[1]:, -1] = -1.0
    # targets[:, n_on[1]:n_off[1], -1] = torch.nan

    return inputs, targets

def generate_gng_trials(
    n_trials: int,
    timing: TaskTiming,
    input_size: int = 8,
    rank: int =2,
    noise: float = 0.1,
    baseline_value: float = 1e-3,
):

    n_on = timing.n_stim_on
    n_off = timing.n_stim_off

    inputs = noise * torch.randn(n_trials, n_steps, input_size)
    targets = torch.zeros(n_trials, n_steps, rank)

    idx_go = torch.rand(n_trials) > 0.5
    idx_nogo = ~idx_go

    # Channels:
    # 4 = Go  # 5 = NoGo  # 6 = Cue
    inputs[idx_go, n_on[0]:n_off[0], 4] += 1.0
    inputs[idx_nogo, n_on[0]:n_off[0], 5] += 1.0
    inputs[:, n_on[1]:n_off[1], 6] += 1.0

    targets[..., 0] = torch.nan
    if baseline_value is not None:
        targets[:, :n_on[0]] = baseline_value

    targets[idx_go, n_off[1]:, -1] = 1.0
    targets[idx_nogo, n_off[1]:, -1] = -1.0
    targets[:, n_on[0]:n_off[1], -1] = torch.nan

    # impose zero readout on AB before test
    targets[:, :n_on[1], 0] = 0.0

    return inputs, targets

def generate_dual_trials(
    n_trials: int,
    timing: TaskTiming,
    input_size: int = 8,
    rank : int = 2,
    noise: float = 0.1,
    baseline_value: float = 1e-3,
):
    n_steps = timing.n_steps
    n_on = timing.n_stim_on
    n_off = timing.n_stim_off

    s_sample = slice(int(n_on[0]), int(n_off[0]))
    s_gng    = slice(int(n_on[1]), int(n_off[1]))

    # gng period
    s_is_gng    = slice(int(n_on[1]), int(n_on[3]))

    s_cue    = slice(int(n_on[2]), int(n_off[2]))
    # s_rwd    = slice(int(n_off[2]), int(n_off[2] + (n_on[3] - n_off[2]) /2))
    s_rwd    = slice(int(n_off[2]), int(n_on[3]))
    s_test   = slice(int(n_on[3]), int(n_off[3]))

    specs = []
    for sample in ["A", "B"]:
        for gng in ["none", "go", "nogo"]:
            for test in ["C", "D"]:
                specs.append((sample, gng, test))

    n_types = len(specs)

    reps = int(np.ceil(n_trials / n_types))
    trial_type = torch.arange(n_types).repeat(reps)[:n_trials]
    trial_type = trial_type[torch.randperm(n_trials)]

    sample_code = torch.tensor(
        [0 if s == "A" else 1 for s, _, _ in specs],
        dtype=torch.long,
    )
    gng_code = torch.tensor(
        [0 if g == "none" else 1 if g == "go" else 2 for _, g, _ in specs],
        dtype=torch.long,
    )
    test_code = torch.tensor(
        [0 if t == "C" else 1 for _, _, t in specs],
        dtype=torch.long,
    )

    sample = sample_code[trial_type]
    gng = gng_code[trial_type]
    test = test_code[trial_type]

    idx_A = sample == 0
    idx_B = sample == 1
    idx_C = test == 0
    idx_D = test == 1

    idx_none = gng == 0
    idx_go = gng == 1
    idx_nogo = gng == 2
    idx_gng = idx_go | idx_nogo

    inputs = noise * torch.randn(n_trials, n_steps, input_size)
    targets = torch.zeros(n_trials, n_steps, rank)

    # DPA sample.
    inputs[idx_A, s_sample, 0] += 1.0
    inputs[idx_B, s_sample, 1] += 1.0

    # GNG stimulus.
    inputs[idx_go, s_gng, 4] += 1.0
    inputs[idx_nogo, s_gng, 5] += 1.0

    # Cue only on GNG trials.
    inputs[idx_gng, s_cue, 6] += 1.0

    # DPA test.
    inputs[idx_C, s_test, 2] += 1.0
    inputs[idx_D, s_test, 3] += 1.0

    targets[..., 0] = torch.nan
    if baseline_value is not None:
        targets[:, :s_sample.start] = baseline_value

    # DPA final target.
    idx_pair = (idx_A & idx_C) | (idx_B & idx_D)
    targets[idx_pair, s_test.stop:, -1] = 1.0
    targets[~idx_pair, s_test.stop:, -1] = -1.0

    # GNG cue-period target.
    targets[idx_go, s_rwd, -1] = 1.0
    targets[idx_nogo, s_rwd, -1] = -1.0

    condition_names = np.array([
        f"{s}_{g}_{t}" if g != "none" else f"{s}_{t}"
        for s, g, t in specs
    ])[trial_type.numpy()]

    return inputs, targets, trial_type, condition_names

import torch
from torch.utils.data import TensorDataset, DataLoader, random_split

def train_val_split(X, y, batch_size=32, frac=0.8):

    # 1. Wrap tensors into a dataset
    dataset = TensorDataset(X.float(), y.float())

    # 2. Train/validation split
    n_total = len(dataset)
    n_train = int(frac * n_total)
    n_val = n_total - n_train

    generator = torch.Generator().manual_seed(0)
    train_dataset, val_dataset = random_split(dataset, lengths=[n_train, n_val], generator=generator)

    # 3. Create DataLoaders
    train_loader = DataLoader(train_dataset, batch_size, shuffle=True)
    val_loader = DataLoader(val_dataset, batch_size, shuffle=False)

    return train_loader, val_loader

import math
import copy
from typing import Any, Optional

import torch
from torch.utils.data import DataLoader


class Optimization:
    """Small PyTorch train/validation loop."""

    def __init__(
        self,
        model: torch.nn.Module,
        train_loader: DataLoader,
        val_loader: DataLoader,
        criterion: torch.nn.Module,
        optimizer: torch.optim.Optimizer,
        scheduler: Any | None = None,
        grad_clip_norm: float | None = None,
        num_epochs: int = 100,
        warmup_epochs: int = 25,
        stop_loss: float = 0.005,
        max_val_loss: float = 100.0,
        zero_low_rank_grad: Any | None = None,
        freeze_input_dims: Any | None = None,
        device: str | torch.device | None = None,
        keep_best: bool = False,        # FIX: optional best-val checkpointing
        verbose: bool = True,
    ):
        self.model = model
        self.train_loader = train_loader
        self.val_loader = val_loader
        self.criterion = criterion
        self.optimizer = optimizer
        self.num_epochs = num_epochs
        self.warmup_epochs = warmup_epochs
        self.stop_loss = stop_loss
        self.max_val_loss = max_val_loss
        self.scheduler = scheduler
        self.zero_low_rank_grad = zero_low_rank_grad
        self.freeze_input_dims = freeze_input_dims
        self.grad_clip_norm = grad_clip_norm
        self.keep_best = keep_best
        self.verbose = verbose

        self.device = torch.device(device) if device is not None else self._infer_device()
        self.model.to(self.device)
        self.model.device = self.device

        self.train_losses: list[float] = []
        self.val_losses: list[float] = []
        self.learning_rates: list[float] = []
        self.learning_rate = self.optimizer.param_groups[0]["lr"]

        self.best_val_loss = float("inf")
        self.best_state_dict: dict | None = None

        # Snapshot frozen input columns to restore after each optimizer.step().
        self._frozen_wi_weight = None
        if self.freeze_input_dims is not None:
            self._frozen_wi_weight = self.model.wi.weight.detach().clone()

    def _infer_device(self) -> torch.device:
        if hasattr(self.model, "device") and isinstance(self.model.device, torch.device):
            return self.model.device
        return next(self.model.parameters()).device

    def _predict(self, X: torch.Tensor, y: Optional[torch.Tensor] = None) -> torch.Tensor:
        return self.model(X, y)

    def _zero_low_rank_grads(self) -> None:
        if self.zero_low_rank_grad is None:
            return
        for param in (self.model.m, self.model.n):
            if param.grad is None:
                continue
            if self.zero_low_rank_grad == "all":
                param.grad.zero_()
            else:
                param.grad[:, self.zero_low_rank_grad] = 0.0

    def _zero_input_grads(self) -> None:
        if self.freeze_input_dims is None:
            return
        if self.model.wi.weight.grad is not None:
            self.model.wi.weight.grad[:, self.freeze_input_dims] = 0.0

    def _restore_frozen_input_weights(self) -> None:
        if self.freeze_input_dims is None:
            return
        with torch.no_grad():
            self.model.wi.weight[:, self.freeze_input_dims] = (
                self._frozen_wi_weight[:, self.freeze_input_dims]
            )

    def _run_epoch(self, loader: DataLoader, train: bool) -> float:
        self.model.train(train)

        total_loss = 0.0
        total_examples = 0
        context = torch.enable_grad() if train else torch.no_grad()

        with context:
            for X, y in loader:
                X = X.to(self.device, non_blocking=True)
                y = y.to(self.device, non_blocking=True)
                batch_size = X.size(0)

                if train:
                    self.optimizer.zero_grad(set_to_none=True)

                y_pred = self._predict(X, y)
                loss = self.criterion(y_pred, y)

                if not torch.isfinite(loss):
                    return float("nan")

                if train:
                    loss.backward()
                    self._zero_low_rank_grads()
                    self._zero_input_grads()
                    if self.grad_clip_norm is not None:
                        torch.nn.utils.clip_grad_norm_(
                            self.model.parameters(), self.grad_clip_norm
                        )
                    self.optimizer.step()
                    self._restore_frozen_input_weights()

                total_loss += loss.item() * batch_size
                total_examples += batch_size

        if total_examples == 0:
            raise ValueError("Dataloader produced zero examples.")

        return total_loss / total_examples

    def _step_scheduler(self, val_loss: float) -> None:
        if self.scheduler is None:
            return
        if isinstance(self.scheduler, torch.optim.lr_scheduler.ReduceLROnPlateau):
            self.scheduler.step(val_loss)
        else:
            self.scheduler.step()

    def _current_lr(self) -> float:
        # during warmup and across scheduler types.
        return self.optimizer.param_groups[0]["lr"]

    def fit(self) -> tuple[list[float], list[float], list[float]]:
        self.train_losses.clear()
        self.val_losses.clear()
        self.learning_rates.clear()
        self.best_val_loss = float("inf")
        self.best_state_dict = None

        for epoch in range(1, self.num_epochs + 1):
            train_loss = self._run_epoch(self.train_loader, train=True)
            val_loss = self._run_epoch(self.val_loader, train=False)

            if epoch >= self.warmup_epochs:
                self._step_scheduler(val_loss)

            self.learning_rate = self._current_lr()

            self.train_losses.append(train_loss)
            self.val_losses.append(val_loss)
            self.learning_rates.append(self.learning_rate)

            # FIX: optional best-val checkpointing.
            if self.keep_best and math.isfinite(val_loss) and val_loss < self.best_val_loss:
                self.best_val_loss = val_loss
                self.best_state_dict = copy.deepcopy(self.model.state_dict())

            if self.verbose and (epoch % 5 == 0):
                print(
                    f"Epoch {epoch:03d}/{self.num_epochs} | "
                    f"lr: {self.learning_rate:.5f} | "
                    f"train: {train_loss:.4f} | val: {val_loss:.4f}"
                )

            if not math.isfinite(train_loss) or not math.isfinite(val_loss):
                print("Stopping: non-finite loss.")
                break

            if train_loss < self.stop_loss and val_loss < self.stop_loss:
                print(f"Stopping: losses below {self.stop_loss}.")
                break

            if val_loss > self.max_val_loss:
                print(f"Stopping: validation loss above {self.max_val_loss}.")
                break

        # FIX: restore best model if requested.
        if self.keep_best and self.best_state_dict is not None:
            self.model.load_state_dict(self.best_state_dict)

        return self.train_losses, self.val_losses, self.learning_rates

import torch
import torch.nn as nn


class MaskedZeroLoss(nn.Module):
    """
    Masked MSE with three regimes per trial:
      - baseline:  y == baseline_value  -> target 0
      - zero:      |y| <= zero_tol      -> target 0
      - nonzero:   |y|  > zero_tol      -> target y
      - Loss is applied to all the channels (recurrent latents and readout).
      - Baseline term is summed across channels, so the
        objective magnitude does depend on output dimensionality.
      - NaNs in targets are treated as "ignore" everywhere.
    """

    def __init__(
        self,
        criterion: nn.Module | None = None,
        nonzero_weight: float = 1.0,
        zero_weight: float = 1.0,
        baseline_weight: float = 1.0,
        zero_tol: float = 0.0,
        baseline_value: float = 1e-3,
        task_channel: int = -1,
    ):
        super().__init__()
        self.criterion = criterion or nn.MSELoss()
        self.nonzero_weight = nonzero_weight
        self.zero_weight = zero_weight
        self.baseline_weight = baseline_weight
        self.zero_tol = zero_tol
        self.baseline_value = baseline_value
        self.task_channel = task_channel

    @staticmethod
    def _masked_mse(pred, target, mask, criterion):
        if mask.sum() == 0:
            return pred.new_tensor(0.0)
        return criterion(pred[mask], target[mask])

    def forward(self, y_pred: torch.Tensor, y: torch.Tensor) -> torch.Tensor:
        # def channel_loss(self, y_pred: torch.Tensor, y: torch.Tensor) -> torch.Tensor:
        """
        y_pred: (B, T, C)  -- C output channels
        y:      (B, T, C)     -- scalar target per time step and channel (NaN = ignore)
        """

        finite = torch.isfinite(y_pred) & torch.isfinite(y)
        bl_mask = finite & (y == self.baseline_value)
        nonzero_mask = finite & (~bl_mask) & (y.abs() >  self.zero_tol)
        zero_mask = finite & (~bl_mask) & (y.abs() <= self.zero_tol)

        zeros = torch.zeros_like(y)

        loss_bl = self._masked_mse(y_pred, zeros, bl_mask, self.criterion)
        loss_zero = self._masked_mse(y_pred, zeros, zero_mask, self.criterion)
        loss_nonzero = self._masked_mse(y_pred, y, nonzero_mask, self.criterion)

        return self.baseline_weight * loss_bl + self.zero_weight   * loss_zero + self.nonzero_weight * loss_nonzero

** Description
The Vanilla RNN updates its hidden state based on the previous hidden state and the current input:

Let:
- \( x_t \): feedforward input at time step \( t \)
- \( h_t \): recurrent input at time \( t \)

class LowRankModel(nn.Module):
    """
    Low-rank RNN with optional reward-feedback input channel.

    reward feedback is now teacher-forced during training

        training (targets is not None):
            rwd_next[b] = 1  if  target[b, t] == +1   else 0
        eval (targets is None):
            rwd_next[b] = 0

    """

    def __init__(
        self,
        input_size,
        hidden_size,
        output_size,
        rank=1,
        gain=1.0,
        alpha=0.01,
        train_inputs=True,
        train_outputs=True,
        warmup_steps=500,
        reward_channel=-1,
        reward_threshold=0.5,
        reward_target_value=1.0,
        task_channel=-1,
        device="cpu",
    ):
        super().__init__()
        self.device = torch.device(device)
        self.warmup_steps = warmup_steps
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.rank = rank
        self.gain = gain

        self.reward_channel = reward_channel
        self.reward_threshold = reward_threshold
        self.reward_target_value = reward_target_value
        self.task_channel = task_channel

        # input layer
        self.wi = None
        if input_size > 0:
            self.Ai = 1.0
            self.wi = nn.Linear(input_size, hidden_size, device=self.device)
            with torch.no_grad():
                nn.init.normal_(self.wi.weight, mean=0.0, std=1.0 / hidden_size**0.5)
                nn.init.zeros_(self.wi.bias)

            if not train_inputs:
                self.Ai = nn.Parameter(torch.randn(1, device=self.device))
                self.wi.weight.requires_grad_(False)
                self.wi.bias.requires_grad_(False)

        # low-rank recurrent layer
        self.m = nn.Parameter(torch.randn(hidden_size, rank, device=self.device))
        self.n = nn.Parameter(torch.randn(hidden_size, rank, device=self.device))

        # output layer
        self.wo = None
        if output_size > 0:
            self.Ao = 1.0
            self.wo = nn.Linear(hidden_size, output_size, device=self.device)
            with torch.no_grad():
                nn.init.normal_(self.wo.weight, mean=0.0, std=1.0 / hidden_size**0.5)
                nn.init.zeros_(self.wo.bias)

            if not train_outputs:
                self.Ao = nn.Parameter(torch.randn(1, device=self.device))
                self.wo.weight.requires_grad_(False)
                self.wo.bias.requires_grad_(False)

        self.nonlinearity = torch.tanh
        self.register_buffer("alpha", torch.tensor(float(alpha)))
        self.register_buffer("exp_alpha", torch.exp(-torch.tensor(float(alpha))))

    def get_readout(self, rates, rec_inputs):
        rec = rates @ self.n / self.hidden_size
        if self.wo is not None:
            ext = self.Ao * self.wo(rates)
            return torch.cat((rec, ext), dim=-1)
        return rec

    def update_dynamics(self, ff_inputs, rec_inputs, rates):
        input_drive = self.Ai * self.wi(ff_inputs) if self.wi is not None else 0.0
        hidden = (rates @ self.n) @ self.m.T / self.hidden_size
        rec_inputs = self.exp_alpha * rec_inputs + (1.0 - self.exp_alpha) * hidden
        rates = self.nonlinearity(self.gain * ( input_drive + rec_inputs))
        return rates, rec_inputs

    def _reward_signal(self, target_t, readout_t):
        """
        FIX: teacher-forced during training, model-driven at eval.
        target_t: (B,) or None
        readout_t: (B, C)
        returns: (B,) float tensor in {0, 1}
        """
        if target_t is None:  # eval
            sig = 0.0 * (readout_t[..., self.task_channel] > self.reward_threshold)
        else:                 # train: teacher forcing
            finite = torch.isfinite(target_t)
            sig = finite & (target_t == self.reward_target_value)
        return sig.to(dtype=readout_t.dtype)

    def forward(self, ff_inputs, targets=None, ret_rates=False):
        B, T = ff_inputs.shape[0], ff_inputs.shape[1]
        N = self.hidden_size

        rec_inputs = 0.1 * torch.randn(B, N, device=self.device)
        rates = self.nonlinearity(self.gain * rec_inputs)

        # warmup with zero input
        zero_inputs = torch.zeros(B, max(self.input_size, 0), device=self.device)
        with torch.no_grad():
            for _ in range(self.warmup_steps):
                rates, rec_inputs = self.update_dynamics(zero_inputs, rec_inputs, rates)
        rates = rates.detach()
        rec_inputs = rec_inputs.detach()

        readout_list = []
        rates_list = [] if ret_rates else None
        inputs_list = [] if ret_rates else None

        rwd_next = torch.zeros(B, device=self.device, dtype=ff_inputs.dtype)

        for step in range(T):
            x_t = ff_inputs[:, step]

            if self.reward_channel is not None:
                x_t = x_t.clone()
                x_t[:, self.reward_channel] = x_t[:, self.reward_channel] + rwd_next

            rates, rec_inputs = self.update_dynamics(x_t, rec_inputs, rates)
            readout = self.get_readout(rates, rec_inputs)
            readout_list.append(readout)

            if ret_rates:
                rates_list.append(rates)
                inputs_list.append(rec_inputs)

            target_t = targets[:, step, self.reward_channel] if targets is not None else None
            rwd_next = self._reward_signal(target_t, readout)

        readout_list = torch.stack(readout_list, dim=1)
        if ret_rates:
            return (
                readout_list,
                torch.stack(rates_list, dim=1),
                torch.stack(inputs_list, dim=1),
            )
        return readout_list

import warnings
import torch
import torch.nn as nn
from typing import Optional


class LowRankInitializer:
    """
    Initialize a LowRankModel's loadings (m, n, wi, wo) as i.i.d. samples
    from N(0, Sigma) across neurons.

    Loading vector per neuron:
        [ m_1..m_R, n_1..n_R, I_1..I_{D_in}, w_1..w_{D_out} ]
    """

    BLOCKS = ("m", "n", "in", "out")

    def __init__(
        self,
        sigma: Optional[torch.Tensor] = None,
        var_m: float | torch.Tensor = 1.0,
        var_n: float | torch.Tensor = 1.0,
        var_in: float | torch.Tensor = 1.0,
        var_out: float | torch.Tensor = 1.0,
        overlaps: Optional[dict] = None,
        jitter: float = 1e-8,
        zero_biases: bool = True,
        auto_psd: bool = True,
        psd_tol: float = 1e-8,
        psd_warn_threshold: float = 1e-3,
        preserve_diagonal: bool = True,
    ):
        self.sigma = sigma
        self.var_m = var_m
        self.var_n = var_n
        self.var_in = var_in
        self.var_out = var_out
        self.overlaps = overlaps or {}
        self.jitter = jitter
        self.zero_biases = zero_biases
        self.auto_psd = auto_psd
        self.psd_tol = psd_tol
        self.psd_warn_threshold = psd_warn_threshold
        self.preserve_diagonal = preserve_diagonal

    # ------------------------------------------------------------------
    def _block_sizes(self, model) -> dict:
        R = model.rank
        D_in = model.input_size
        D_out = model.wo.weight.shape[0] if model.wo is not None else 0
        return {"m": R, "n": R, "in": D_in, "out": D_out}

    def _block_offsets(self, sizes: dict) -> dict:
        offsets, cur = {}, 0
        for b in self.BLOCKS:
            offsets[b] = cur
            cur += sizes[b]
        return offsets

    def _expand_var(self, v, size: int) -> torch.Tensor:
        if size == 0:
            return torch.zeros(0)
        if torch.is_tensor(v):
            v = v.flatten().float()
            assert v.numel() == size, f"Expected {size} variances, got {v.numel()}"
            return v
        return torch.full((size,), float(v))

    # ------------------------------------------------------------------
    def _project_psd(self, Sigma: torch.Tensor) -> torch.Tensor:
        """
        Project a symmetric matrix onto the PSD cone via eigenvalue clipping.

        If preserve_diagonal=True, the diagonal of the projected matrix is
        rescaled to match the input's diagonal (so user-specified variances
        are preserved up to a final symmetric rescaling).
        """
        # Symmetrize defensively
        S = 0.5 * (Sigma + Sigma.T)
        d_orig = torch.diag(S).clone()

        evals, evecs = torch.linalg.eigh(S)
        evals_clipped = evals.clamp_min(self.psd_tol)
        S_psd = (evecs * evals_clipped) @ evecs.T

        if self.preserve_diagonal:
            d_new = torch.diag(S_psd).clamp_min(self.psd_tol)
            # Rescale so diag(S_psd) == d_orig, preserving correlation structure
            scale = torch.sqrt(d_orig.clamp_min(0.0) / d_new)
            S_psd = S_psd * scale.unsqueeze(0) * scale.unsqueeze(1)
            # Diagonal is now exactly d_orig (up to fp), confirm PSD after rescale
            # (rescaling by diag(s)·M·diag(s) preserves PSDness)

        # Re-symmetrize against fp drift
        S_psd = 0.5 * (S_psd + S_psd.T)
        return S_psd, evals.min().item()

    def _maybe_project(self, Sigma: torch.Tensor) -> torch.Tensor:
        evals = torch.linalg.eigvalsh(Sigma)
        min_ev = float(evals.min())

        if min_ev >= -self.psd_tol:
            return Sigma  # already PSD up to tol

        if not self.auto_psd:
            raise ValueError(
                f"Sigma is not PSD (min eigenvalue = {min_ev:.3e}). "
                "Pass auto_psd=True to project onto the PSD cone, or fix the spec."
            )

        Sigma_psd, _ = self._project_psd(Sigma)

        # Quantify correction
        delta = (Sigma_psd - Sigma)
        rel_change = (
            torch.linalg.matrix_norm(delta, ord="fro")
            / torch.linalg.matrix_norm(Sigma, ord="fro").clamp_min(1e-12)
        ).item()

        msg = (
            f"Sigma was not PSD (min eigval = {min_ev:.3e}); "
            f"projected to nearest PSD (Frobenius rel. change = {rel_change:.3e})."
        )
        if rel_change > self.psd_warn_threshold:
            msg += (
                " Correction is large — consider reducing overlap magnitudes "
                "or increasing the relevant variances."
            )
        warnings.warn(msg, stacklevel=3)

        return Sigma_psd

    # ------------------------------------------------------------------
    def build_sigma(self, model) -> torch.Tensor:
        sizes = self._block_sizes(model)
        offsets = self._block_offsets(sizes)
        P = sum(sizes.values())

        if self.sigma is not None:
            assert self.sigma.shape == (P, P), (
                f"sigma must be ({P},{P}), got {tuple(self.sigma.shape)}"
            )
            return self._maybe_project(self.sigma.float().clone())

        diag = torch.cat([
            self._expand_var(self.var_m,   sizes["m"]),
            self._expand_var(self.var_n,   sizes["n"]),
            self._expand_var(self.var_in,  sizes["in"]),
            self._expand_var(self.var_out, sizes["out"]),
        ])
        Sigma = torch.diag(diag)

        for ((ba, ia), (bb, ib)), cov in self.overlaps.items():
            assert ba in self.BLOCKS and bb in self.BLOCKS, (
                f"Unknown block: {ba} or {bb}; must be in {self.BLOCKS}"
            )
            assert 0 <= ia < sizes[ba], f"Index {ia} out of range for block {ba}"
            assert 0 <= ib < sizes[bb], f"Index {ib} out of range for block {bb}"
            a = offsets[ba] + ia
            b = offsets[bb] + ib
            Sigma[a, b] = cov
            Sigma[b, a] = cov

        return self._maybe_project(Sigma)

    # ------------------------------------------------------------------
    def _sample_loadings(
        self, Sigma: torch.Tensor, N: int, device, generator=None
    ) -> torch.Tensor:
        P = Sigma.shape[0]
        Sigma_j = Sigma + self.jitter * torch.eye(P)

        try:
            L = torch.linalg.cholesky(Sigma_j)
        except RuntimeError as e:
            warnings.warn(
                f"Cholesky failed even after PSD projection ({e}); "
                "falling back to eigendecomposition."
            )
            evals, evecs = torch.linalg.eigh(Sigma_j)
            evals = evals.clamp_min(0.0)
            L = evecs * evals.sqrt().unsqueeze(0)

        z = torch.randn(N, P, generator=generator)
        return (z @ L.T).to(device)

    @torch.no_grad()
    def apply(self, model: nn.Module, seed: Optional[int] = None) -> torch.Tensor:
        Sigma = self.build_sigma(model)
        sizes = self._block_sizes(model)
        offsets = self._block_offsets(sizes)
        N = model.hidden_size
        device = model.device

        gen = None
        if seed is not None:
            gen = torch.Generator()
            gen.manual_seed(seed)

        X = self._sample_loadings(Sigma, N, device, generator=gen)

        def slc(b):
            return slice(offsets[b], offsets[b] + sizes[b])

        model.m.copy_(X[:, slc("m")])
        model.n.copy_(X[:, slc("n")])

        if sizes["in"] > 0 and model.wi is not None:
            model.wi.weight.copy_(X[:, slc("in")])
            if self.zero_biases:
                model.wi.bias.zero_()

        if sizes["out"] > 0 and model.wo is not None:
            model.wo.weight.copy_(X[:, slc("out")].T)
            if self.zero_biases:
                model.wo.bias.zero_()

        return Sigma

    # ------------------------------------------------------------------
    def validate(self, model, raise_on_fail: bool = True) -> tuple[bool, float]:
        """Build Sigma without projection and report PSD status."""
        old = self.auto_psd
        self.auto_psd = False
        try:
            try:
                Sigma = self.build_sigma(model)
                min_ev = float(torch.linalg.eigvalsh(Sigma).min())
                return True, min_ev
            except ValueError as e:
                if raise_on_fail:
                    raise
                # Re-extract min eigenvalue
                self.auto_psd = True  # to actually build it
                Sigma = self.build_sigma(model)
                min_ev = float(torch.linalg.eigvalsh(Sigma).min())
                return False, min_ev
        finally:
            self.auto_psd = old

import torch
import torch.nn as nn
from typing import Optional, List

class MixtureLowRankInitializer:
    """
    Initialize a LowRankModel's m and n loadings from a Gaussian mixture
    with K components, independently for each neuron.

    Neuron i is assigned to population k ~ Categorical(pop_probs).
    Then (m_i, n_i) ~ N(mu_k, Sigma_k) where mu_k, Sigma_k are
    provided per population.
    """

    def __init__(
        self,
        n_populations: int = 1,
        pop_means: Optional[List[torch.Tensor]] = None,
        pop_covs: Optional[List[torch.Tensor]] = None,
        pop_probs: Optional[torch.Tensor] = None,
        jitter: float = 1e-6,
        zero_biases: bool = True,
        # Optional: input/output weight handling if desired
        input_init: Optional[dict] = None,   # e.g. {"var": 1.0}
        output_init: Optional[dict] = None,  # e.g. {"var": 1.0}
    ):
        self.n_populations = n_populations
        self.jitter = jitter
        self.zero_biases = zero_biases
        self.input_init = input_init or {}
        self.output_init = output_init or {}

        # Validate / set defaults for mixture parameters
        if pop_means is None:
            pop_means = [torch.zeros(1) for _ in range(n_populations)]  # dummy, will be replaced
        if pop_covs is None:
            pop_covs = [torch.eye(pop_means[0].numel()) for _ in range(n_populations)]
        if pop_probs is None:
            pop_probs = torch.ones(n_populations) / n_populations

        assert len(pop_means) == n_populations
        assert len(pop_covs) == n_populations
        self.p_means = [m.flatten() for m in pop_means]
        self.p_covs = [c for c in pop_covs]  # assume 2D square
        self.p_probs = pop_probs.float()
        # dimensionality of concatenated (m, n)
        self.loading_dim = self.p_means[0].numel()

    @torch.no_grad()
    def apply(self, model: LowRankModel, seed: Optional[int] = None) -> None:
        N = model.hidden_size
        R = model.rank
        device = model.device

        if seed is not None:
            gen = torch.Generator(device=device)
            gen.manual_seed(seed)
        else:
            gen = None

        # 1. Assign populations
        pop_ids = torch.multinomial(
            self.p_probs, N, replacement=True, generator=gen
        ).to(device)   # (N,)

        # 2. Sample from each component
        samples = torch.zeros(N, self.loading_dim, device=device)
        for k in range(self.n_populations):
            idx = (pop_ids == k)
            n_k = idx.sum().item()
            if n_k == 0:
                continue
            mu_k = self.p_means[k].to(device)
            cov_k = self.p_covs[k].to(device)
            # Cholesky with jitter
            L = torch.linalg.cholesky(cov_k + self.jitter * torch.eye(cov_k.shape[0], device=device))
            z = torch.randn(n_k, self.loading_dim, device=device, generator=gen)
            samples[idx] = mu_k + (z @ L.T)

        # Split into m and n
        assert samples.shape[1] == 2 * R, (
            f"Loading dimension {samples.shape[1]} != 2*rank {2*R}"
        )
        model.m.copy_(samples[:, :R])
        model.n.copy_(samples[:, R:])

        # 3. Optional input/output weights (default: standard normal)
        if model.wi is not None:
            wi_var = self.input_init.get("var", 1.0 / model.hidden_size**0.5)
            nn.init.normal_(model.wi.weight, mean=0.0, std=wi_var)
            if self.zero_biases:
                nn.init.zeros_(model.wi.bias)
        if model.wi is not None and "mean" in self.input_init:
            model.wi.weight.add_(self.input_init["mean"])

        if model.wo is not None:
            wo_var = self.output_init.get("var", 1.0 / model.hidden_size**0.5)
            nn.init.normal_(model.wo.weight, mean=0.0, std=wo_var)
            if self.zero_biases:
                nn.init.zeros_(model.wo.bias)

        pos = F.softplus(self.n_raw)  # always >= 0
        # For I neurons, flip the sign
        n_con = torch.where(self.exc_mask.unsqueeze(1), pos, -pos)
        return n_con

        input_drive = self.Ai * self.wi(ff_inputs) if self.wi is not None else 0.0

* Vector fields

The low-rank recurrent state lives approximately on the manifold

\[
h = M \kappa,\qquad M = [m_1,m_2]\in \mathbb{R}^{N\times 2}.
\]

For a fixed feedforward input \(x\), the induced rank-coordinate dynamics are

\[
\kappa^{+}
= (1-\alpha)\kappa + \alpha \Psi(\kappa; x),
\]

where

\[
\Psi(\kappa; x)
= \frac{1}{N} n^\top \tanh(W_i x + b_i + M\kappa).
\]

Thus the phase portrait in the \((\kappa_1,\kappa_2)\) plane is given by the vector field

\[
\Delta \kappa
= \kappa^{+}-\kappa
= \alpha\left[\Psi(\kappa;x)-\kappa\right].
\]

def low_rank_field_np(params, kappa, ff_input=None, include_beta=False):
    """
    Vector field in coordinates h = Mκ.

        F(κ; x) = Ψ(κ; x) - κ
        Ψ(κ; x) = N^{-1} n^T tanh(W_i x + b_i + Mκ)

    If include_beta=True, returns βF, exactly matching the discrete update
    increment κ⁺ - κ for fixed x.
    """
    M = params["M"]
    Nvec = params["Nvec"]
    Wi = params["Wi"]
    bi = params["bi"]
    Ai = params["Ai"]

    kappa = np.asarray(kappa, dtype=np.float64)
    orig_shape = kappa.shape
    rank = M.shape[1]
    kappa_flat = kappa.reshape(-1, rank)

    if ff_input is None:
        ff_input = np.zeros(Wi.shape[1], dtype=np.float64)
    else:
        ff_input = np.asarray(ff_input, dtype=np.float64)

    drive = Ai * (ff_input @ Wi.T + bi)                  # [N]
    h = kappa_flat @ M.T                          # [B, N]
    r = np.tanh(h + drive[None, :])               # [B, N]
    psi = r @ Nvec / M.shape[0]                   # [B, R]

    field = psi - kappa_flat
    if include_beta:
        field = params["beta"] * field

    return field.reshape(orig_shape)


def low_rank_jacobian_flow_np(params, kappa, ff_input=None):
    """Jacobian of F(κ; x) = Ψ(κ; x) - κ."""
    M = params["M"]
    Nvec = params["Nvec"]
    Wi = params["Wi"]
    bi = params["bi"]
    Ai = params["Ai"]

    kappa = np.asarray(kappa, dtype=np.float64).reshape(-1)

    if ff_input is None:
        ff_input = np.zeros(Wi.shape[1], dtype=np.float64)
    else:
        ff_input = np.asarray(ff_input, dtype=np.float64)

    u = M @ kappa + Ai * (ff_input @ Wi.T + bi)
    phi_prime = 1.0 - np.tanh(u) ** 2

    J = Nvec.T @ (phi_prime[:, None] * M) / M.shape[0]
    J -= np.eye(M.shape[1])
    return J


def low_rank_jacobian_map_np(params, kappa, ff_input=None):
    """Jacobian of κ⁺ = κ + βF(κ)."""
    J_flow = low_rank_jacobian_flow_np(params, kappa, ff_input=ff_input)
    return np.eye(J_flow.shape[0]) + params["beta"] * J_flow


def project_rec_inputs_to_kappa(model, rec_inputs):
    """
    Least-squares coordinates for h ≈ Mκ.

    This is the coordinate system used by low_rank_field_np. Do not replace this
    by h@m/||m||^2 unless M columns are known orthogonal.
    """
    device = _model_device(model)
    dtype = _model_dtype(model)

    h = torch.as_tensor(rec_inputs, device=device, dtype=dtype)
    orig_shape = h.shape
    hidden_size = model.m.shape[0]

    if orig_shape[-1] != hidden_size:
        raise ValueError(f"Expected last dim {hidden_size}, got {orig_shape[-1]}.")

    h_flat = h.reshape(-1, hidden_size)  # [B, N]

    with torch.no_grad():
        sol = torch.linalg.lstsq(model.m, h_flat.T).solution  # [R, B]
        kappa = sol.T.reshape(*orig_shape[:-1], model.m.shape[1])

    return kappa

    dtype = _model_dtype(model)

    ff_inputs = ff_inputs.to(device=device, dtype=dtype)
    if targets is not None:
        targets = targets.to(device=device)

    batch_size, T, input_size = ff_inputs.shape
    hidden_size = model.m.shape[0]

    rec_inputs = torch.randn(batch_size, hidden_size, device=device, dtype=dtype)
    rates = model.nonlinearity(rec_inputs)

    # Match LowRankModel warmup.
    for _ in range(T):
        rand_inputs = torch.randn(batch_size, input_size, device=device, dtype=dtype)
        rates, rec_inputs = model.update_dynamics(rand_inputs, rec_inputs, rates)

    readout_list = []
    rates_list = []
    rec_list = []
    eff_input_list = []

    rwd_next = torch.zeros(batch_size, device=device, dtype=dtype)
    if use_reward_feedback is None:
        use_reward_feedback = targets is not None and getattr(model, "reward_channel", None) is not None

    for step in range(T):
        x_t = ff_inputs[:, step].clone()

        if use_reward_feedback:
            x_t[:, model.reward_channel] = x_t[:, model.reward_channel] + rwd_next

        rates, rec_inputs = model.update_dynamics(x_t, rec_inputs, rates)
        readout = model.get_readout(rates, rec_inputs)

        readout_list.append(readout)
        rates_list.append(rates)
        rec_list.append(rec_inputs)
        eff_input_list.append(x_t)

        if use_reward_feedback:
            rwd_mask = (targets[:, step, model.reward_channel] == 1) & (readout[:, reward_readout_index] > 0.5)
            rwd_next = rwd_mask.to(dtype=dtype)
        else:
            rwd_next.zero_()

    return (
        torch.stack(readout_list, dim=1),
        torch.stack(rates_list, dim=1),
        torch.stack(rec_list, dim=1),
        torch.stack(eff_input_list, dim=1),
    )


def auto_limits_from_trajs(kappa_traj, qlo=1, qhi=99, pad=1.35, symmetric=True):
    z = kappa_traj.reshape(-1, kappa_traj.shape[-1])
    lo = np.percentile(z[:, :2], qlo, axis=0)
    hi = np.percentile(z[:, :2], qhi, axis=0)

    if symmetric:
        lim = float(pad * np.max(np.abs(np.concatenate([lo, hi]))))
        lim = max(lim, 1e-6)
        return (-lim, lim), (-lim, lim)

    center = 0.5 * (lo + hi)
    half = 0.5 * (hi - lo) * pad
    return (
        (float(center[0] - half[0]), float(center[0] + half[0])),
        (float(center[1] - half[1]), float(center[1] + half[1])),
    )


def make_vector_field_grid(
    model,
    ff_input,
    xlim,
    ylim,
    n_grid=180,
    include_beta=False,
):
    params = low_rank_numpy_params(model)
    ff_input_np = torch.as_tensor(ff_input).detach().cpu().numpy().astype(np.float64)

    k1 = np.linspace(xlim[0], xlim[1], n_grid)
    k2 = np.linspace(ylim[0], ylim[1], n_grid)
    K1, K2 = np.meshgrid(k1, k2)
    K = np.stack([K1, K2], axis=-1)

    field = low_rank_field_np(
        params,
        K,
        ff_input=ff_input_np,
        include_beta=include_beta,
    )

    U = field[..., 0]
    V = field[..., 1]
    speed = np.sqrt(U**2 + V**2)

    return K1, K2, U, V, speed, params, ff_input_np


def merge_roots(roots, residuals, merge_tol=5e-2):
    if len(roots) == 0:
        return np.empty((0, 2)), np.empty((0,))

    roots = np.asarray(roots, dtype=np.float64)
    residuals = np.asarray(residuals, dtype=np.float64)

    order = np.argsort(residuals)
    kept = []
    kept_res = []

    for idx in order:
        r = roots[idx]
        if all(np.linalg.norm(r - q) > merge_tol for q in kept):
            kept.append(r)
            kept_res.append(residuals[idx])

    return np.stack(kept, axis=0), np.asarray(kept_res)


def find_all_fixed_points(
    model,
    xlim,
    ylim,
    ff_input,
    n_seeds=45,
    residual_tol=1e-8,
    merge_tol=5e-2,
):
    params = low_rank_numpy_params(model)
    ff_input_np = torch.as_tensor(ff_input).detach().cpu().numpy().astype(np.float64)

    def fun(k):
        return low_rank_field_np(params, k, ff_input=ff_input_np).reshape(-1)

    def jac(k):
        return low_rank_jacobian_flow_np(params, k, ff_input=ff_input_np)

    k1_seeds = np.linspace(xlim[0], xlim[1], n_seeds)
    k2_seeds = np.linspace(ylim[0], ylim[1], n_seeds)

    roots = []
    residuals = []

    for k1_0 in k1_seeds:
        for k2_0 in k2_seeds:
            sol = root(
                fun,
                np.array([k1_0, k2_0], dtype=np.float64),
                jac=jac,
                method="hybr",
                options={"xtol": 1e-12, "maxfev": 1000},
            )
            if not sol.success:
                continue

            fp = sol.x
            residual = np.linalg.norm(fun(fp))
            in_bounds = xlim[0] <= fp[0] <= xlim[1] and ylim[0] <= fp[1] <= ylim[1]

            if residual <= residual_tol and in_bounds:
                roots.append(fp)
                residuals.append(residual)

    return merge_roots(roots, residuals, merge_tol=merge_tol)


def classify_fixed_points(model, fixed_points, ff_input, eig_tol=1e-5):
    params = low_rank_numpy_params(model)
    ff_input_np = torch.as_tensor(ff_input).detach().cpu().numpy().astype(np.float64)

    labels = []
    eigvals = []

    for fp in fixed_points:
        J = low_rank_jacobian_map_np(params, fp, ff_input=ff_input_np)
        ev = np.linalg.eigvals(J)
        rad = np.abs(ev)

        n_stable = np.sum(rad < 1.0 - eig_tol)
        n_unstable = np.sum(rad > 1.0 + eig_tol)

        if n_stable == len(rad):
            label = "attractor"
        elif n_unstable == len(rad):
            label = "repeller"
        elif n_stable > 0 and n_unstable > 0:
            label = "saddle"
        else:
            label = "nonhyperbolic"

        labels.append(label)
        eigvals.append(ev)

    if len(eigvals) == 0:
        return np.asarray([], dtype=object), np.empty((0, 2), dtype=np.complex128)

    return np.asarray(labels, dtype=object), np.stack(eigvals, axis=0)


def condition_indices_for_task(inputs, timing, task, condition_names=None):
    task = task.lower()
    n_on = timing.n_stim_on
    n_off = timing.n_stim_off

    if task == "dpa":
        sample_epoch = slice(int(n_on[0]), int(n_off[0]))
        test_epoch = slice(int(n_on[1]), int(n_off[1]))

        sample_drive = inputs[:, sample_epoch, :2].mean(dim=1)
        test_drive = inputs[:, test_epoch, 2:4].mean(dim=1)

        is_A = sample_drive[:, 0] > sample_drive[:, 1]
        is_B = ~is_A
        is_C = test_drive[:, 0] > test_drive[:, 1]
        is_D = ~is_C

        return {
            "AC": torch.where(is_A & is_C)[0],
            "AD": torch.where(is_A & is_D)[0],
            "BC": torch.where(is_B & is_C)[0],
            "BD": torch.where(is_B & is_D)[0],
        }

    if task == "gng":
        stim_epoch = slice(int(n_on[0]), int(n_off[0]))
        stim_drive = inputs[:, stim_epoch, 4:6].mean(dim=1)

        is_go = stim_drive[:, 0] > stim_drive[:, 1]
        is_nogo = ~is_go

        return {
            "Go": torch.where(is_go)[0],
            "NoGo": torch.where(is_nogo)[0],
        }

    if task == "dual":
        if condition_names is not None:
            names = np.asarray(condition_names).astype(str)
            groups = {k: [] for k in ["AC", "AD", "BC", "BD", "Go", "NoGo", "DPAOnly"]}

            for i, name in enumerate(names):
                parts = name.split("_")
                if len(parts) == 2:
                    sample, test = parts
                    gng = "none"
                elif len(parts) == 3:
                    sample, gng, test = parts
                else:
                    raise ValueError(f"Unexpected condition name: {name}")

                pair = f"{sample}{test}"
                if pair in groups:
                    groups[pair].append(i)

                if gng == "go":
                    groups["Go"].append(i)
                elif gng == "nogo":
                    groups["NoGo"].append(i)
                elif gng == "none":
                    groups["DPAOnly"].append(i)

            return {k: torch.as_tensor(v, dtype=torch.long) for k, v in groups.items()}

        sample_epoch = slice(int(n_on[0]), int(n_off[0]))
        gng_epoch = slice(int(n_on[1]), int(n_off[1]))
        test_epoch = slice(int(n_on[3]), int(n_off[3]))

        sample_drive = inputs[:, sample_epoch, :2].mean(dim=1)
        gng_drive = inputs[:, gng_epoch, 4:6].mean(dim=1)
        test_drive = inputs[:, test_epoch, 2:4].mean(dim=1)

        is_A = sample_drive[:, 0] > sample_drive[:, 1]
        is_B = ~is_A
        is_C = test_drive[:, 0] > test_drive[:, 1]
        is_D = ~is_C
        is_go = gng_drive[:, 0] > 0.25
        is_nogo = gng_drive[:, 1] > 0.25
        is_dpa_only = ~(is_go | is_nogo)

        return {
            "AC": torch.where(is_A & is_C)[0],
            "AD": torch.where(is_A & is_D)[0],
            "BC": torch.where(is_B & is_C)[0],
            "BD": torch.where(is_B & is_D)[0],
            "Go": torch.where(is_go)[0],
            "NoGo": torch.where(is_nogo)[0],
            "DPAOnly": torch.where(is_dpa_only)[0],
        }

    raise ValueError(f"Unknown task: {task}")


def flow_specs_for_task(timing, task):
    task = task.lower()
    n_on = [int(x) for x in timing.n_stim_on]
    n_off = [int(x) for x in timing.n_stim_off]

    if task == "dpa":
        return [
            dict(name="Autonomous", dims=None, conds=["AC", "AD", "BC", "BD"]),

            dict(name="NoGo", dims=[5], conds=["NoGo"]),
            dict(name="Cue", dims=[6], conds=["Go", "NoGo"]),

    if task == "dual":

def _canonical_input_mask(effective_x, dims, threshold=0.35, atol_inactive=0.35):
    """
    Boolean [B, T] mask selecting time points whose effective input matches a
    frozen-input panel.

    For dims=None, selects approximately zero-input points.
    For dims=[k], selects points where channel k is active and all other task
    channels are inactive. Noise is tolerated by thresholds.
    """
    x = effective_x.detach().cpu()

    if dims is None:
        return x.abs().amax(dim=-1) < atol_inactive

    dims = list(dims)
    active = torch.ones(x.shape[:2], dtype=torch.bool)
    for d in dims:
        active &= x[..., d] > threshold

    inactive_dims = [d for d in range(x.shape[-1]) if d not in dims]
    inactive = x[..., inactive_dims].abs().amax(dim=-1) < atol_inactive

    return active & inactive


def _segments_from_mask(mask_1d, min_len=2):
    idx = np.where(mask_1d)[0]
    if idx.size == 0:
        return []

    breaks = np.where(np.diff(idx) > 1)[0] + 1
    chunks = np.split(idx, breaks)

    out = []
    for c in chunks:
        if len(c) >= min_len:
            out.append(slice(int(c[0]), int(c[-1]) + 1))
    return out


def default_arrow_times(T, n_arrows=4, arrow_step=3):
    if T <= arrow_step + 1:
        return np.array([], dtype=int)

    lo = 1
    hi = T - arrow_step - 1
    n_arrows = min(n_arrows, max(1, hi - lo + 1))
    return np.unique(np.linspace(lo, hi, n_arrows).astype(int))


def add_time_arrows(
    ax,
    xy,
    times=None,
    color="k",
    arrow_step=3,
    mutation_scale=13,
    lw=1.5,
    alpha=0.95,
    zorder=8,
    min_length=1e-4,
):
    xy = np.asarray(xy, dtype=np.float64)
    T = xy.shape[0]

    if times is None:
        times = default_arrow_times(T, n_arrows=4, arrow_step=arrow_step)

    for t in times:
        t = int(t)
        if t < 0:
            t += T
        if t < 0 or t >= T - 1:
            continue

        j0 = t
        j1 = min(t + arrow_step, T - 1)

        while j1 < T and np.linalg.norm(xy[j1] - xy[j0]) < min_length:
            j0 += 1
            j1 = min(j0 + arrow_step, T - 1)
            if j0 >= T - 1:
                break

        if j0 >= T - 1:
            continue

        p0 = xy[j0]
        p1 = xy[j1]

        if np.linalg.norm(p1 - p0) < min_length:
            continue

        ax.annotate(
            "",
            xy=p1,
            xytext=p0,
            arrowprops=dict(
                arrowstyle="-|>",
                color=color,
                lw=lw,
                alpha=alpha,
                shrinkA=0,
                shrinkB=0,
                mutation_scale=mutation_scale,
            ),
            zorder=zorder,
        )

def concrete_conditions(pair_status: str, gng_status: str) -> list[str]:
    """
    Return the two concrete condition names for a panel.

    pair_status: "pair" or "unpair"
    gng_status : "none", "go", or "nogo"
    """
    if pair_status == "pair":
        base = [("A", "C"), ("B", "D")]
    elif pair_status == "unpair":
        base = [("A", "D"), ("B", "C")]
    else:
        raise ValueError(pair_status)

    out = []
    for sample, test in base:
        if gng_status == "none":
            out.append(f"{sample}_{test}")
        else:
            out.append(f"{sample}_{gng_status}_{test}")

    return out

* How do I interpret the result of the training in terms of populations as in Dubreuil et al.?
** Interpreting the trained DPA model through the Dubreuil et al. lens

*Mode: Explain.*

*** 1. The Dubreuil framework in one paragraph

Dubreuil, Valente, Beiran, Mastrogiuseppe & Ostojic (2022, /Nat. Neurosci.*) showed that trained low-rank RNNs can be summarized by clustering neurons in the *loading space* — the joint vector

\[
\ell_i = (m_i^{(1)}, \ldots, m_i^{(R)}, n_i^{(1)}, \ldots, n_i^{(R)}, I_i^{(1)}, \ldots, I_i^{(D_\text{in})}, w_i^{(1)}, \ldots, w_i^{(D_\text{out})}) \in \mathbb{R}^{2R + D_\text{in} + D_\text{out}}
\]

into *populations* (typically a Gaussian mixture). Each population is then characterized by its *covariance structure* in loading space — specifically the overlaps

\[
\sigma_{m^a n^b} = \frac{1}{N_p}\sum_{i \in p} m_i^a n_i^b, \qquad \sigma_{I^k n^a} = \frac{1}{N_p}\sum_{i \in p} I_i^k n_i^a, \qquad \text{etc.}
\]

These overlaps determine, in the mean-field limit, the *effective circuit*: which inputs project onto which latent dimensions, which latent dimensions couple to which others, and which read out. The number of populations needed reflects the minimal circuit complexity required by the task. [cite:@dubreuil2022role]

*** 2. What to look at in /your/ trained model

You have rank \(R=2\), \(D_\text{in}=8\), \(D_\text{out}=0\) (linear readout from \(n\)), \(N=512\). The loading per neuron is 12-dimensional:

\[
\ell_i = (m_i^1, m_i^2, n_i^1, n_i^2, I_i^1, \ldots, I_i^8).
\]

**** 2.1 The four-cluster signature in your \((m, n)\) scatter

Your final-figure scatter of \((m_a, n_b)\) shows a clear *bimodal / four-quadrant structure*, most visible in \((m_0, n_0)\), \((m_0, n_1)\), and especially \((m_1, n_1)\) — the latter shows two well-separated blobs along the diagonal. This is the hallmark of a *discrete-population solution* rather than a single-Gaussian (unstructured) solution. In Dubreuil's terms, the network has learned to use ≥2 populations whose loadings differ in sign along specific axes.

A single multivariate Gaussian over \(\ell_i\) cannot produce two disjoint blobs; only a /mixture/ can. The DPA task is one of the cases in [cite:@dubreuil2022role] where a multi-population solution is *necessary* because the task requires a context-dependent (XOR-like) computation: the sign of the readout at test time depends on the /product/ sample × test, which cannot be implemented by a single Gaussian rank-2 network.

**** 2.2 The fixed-point picture confirms it

Your "Autonomous" panel shows *two stable fixed points* roughly at \((\kappa_1,\kappa_2)\approx(\pm 0, \mp 0.8)\) plus additional ones — i.e. a discrete attractor structure carrying the sample memory along \(\kappa_2\). Stimulus panels (A, B, C, D) show that:

- *A vs B* push \(\kappa_1\) in opposite directions (sample axis along \(\kappa_1\)).
- *C vs D* rotate the flow such that the attractor reached during test depends on the /combination/ with the stored sample.

This is exactly the "selection vector × memory vector" geometry that Dubreuil et al. dissect for context-dependent and DMS-like tasks.

*** 3. How to make the population analysis quantitative

To replicate Dubreuil's analysis on your model:

1. *Standardize* each of the 12 loading coordinates (zero mean, unit variance) so the clustering is not dominated by scale.
2. *Fit a Gaussian mixture* on \(\{\ell_i\}_{i=1}^{512}\) with \(K=1,2,3,4,\ldots\) components.
3. *Model selection*: pick \(K\) by BIC, or by the /resampling test/ used in the paper — refit a multivariate Gaussian (\(K=1\)) to the loadings, resimulate the network from this Gaussian, and check whether task performance is preserved. If performance drops, \(K=1\) is insufficient and you need a true mixture. [cite:@dubreuil2022role]
4. *Per-population overlap matrix*: for each cluster \(p\), compute
   \[
   \Sigma^{(p)}_{ab} = \frac{1}{N_p}\sum_{i\in p}\ell_i^a\,\ell_i^b.
   \]
   The non-zero off-diagonal entries between \(I^k\) and \(n^r\) tell you /which inputs that population routes to which latent dimension/; between \(m^r\) and \(n^{r'}\) they give the /effective recurrent connectivity/ in the rank-\(R\) reduced dynamics.

*** 4. Likely interpretation given your initialization and your scatter

Given how you initialized — overlaps roughly:

- \(\langle m_1, n_1\rangle\), \(\langle m_2, n_2\rangle > 0\) (positive self-coupling on each rank dimension → bistability)
- inputs 0/1 (sample A/B) load on \(n_1\) with opposite sign
- inputs 2/3 (test C/D), 4/5 (Go/NoGo), 6 (cue), 7 (reward) load on \(n_2\)

and given that the scatter clearly shows sign-structured clusters, the trained network most plausibly implements a *two-population* (or small-K) solution where:

- *Population P+*: positive \(m_1\), positive \(n_1\) — supports the "A" attractor on \(\kappa_1\).
- *Population P−*: negative \(m_1\), negative \(n_1\) — supports the "B" attractor.
- Within each, the \((m_2, n_2)\) sub-structure plus the input loadings on \(I^{2..7}\) implement the *gating* by the test stimulus (C vs D) and produce the correct sign on the readout via the XOR-like \(\kappa_1 \cdot \kappa_2 \to \) output mapping.

This is the same broad architecture Dubreuil et al. identified for DMS / context-dependent integration: a /memory subspace/ (your \(\kappa_1\)) maintained by within-population self-excitation (\(\langle m\,n\rangle > 0\) per cluster), and a /response subspace/ (your \(\kappa_2\)) into which sample-dependent and test-dependent inputs are routed with population-specific signs. import numpy as np
import torch
import matplotlib.pyplot as plt
from sklearn.mixture import GaussianMixture
from dataclasses import dataclass, field
from typing import Optional


# ----------------------------------------------------------------------
# 1. Loading extraction
# ----------------------------------------------------------------------

def extract_loadings(model) -> tuple[np.ndarray, dict]:
    """
    Returns
    -------
    L : [N, P] loading matrix, columns = [m_1..m_R, n_1..n_R, I_1..I_Din, w_1..w_Dout]
    meta : dict with block sizes and offsets for interpretation.
    """
    m = model.m.detach().cpu().numpy()              # [N, R]
    n = model.n.detach().cpu().numpy()              # [N, R]
    wi = model.wi.weight.detach().cpu().numpy()     # [N, Din]
    D_out = model.wo.weight.shape[0] if model.wo is not None else 0
    if D_out > 0:
        wo = model.wo.weight.detach().cpu().numpy().T  # [N, Dout]
    else:
        wo = np.zeros((m.shape[0], 0))

    L = np.concatenate([m, n, wi, wo], axis=1)
    R = m.shape[1]
    D_in = wi.shape[1]

    sizes = {"m": R, "n": R, "in": D_in, "out": D_out}
    offsets, cur = {}, 0
    for b in ["m", "n", "in", "out"]:
        offsets[b] = cur
        cur += sizes[b]

    return L, {"sizes": sizes, "offsets": offsets, "P": L.shape[1], "N": L.shape[0]}


# ----------------------------------------------------------------------
# 2. Optional task-aligned rotation of (m, n)
# ----------------------------------------------------------------------
# Rank-R loadings have a GL(R) gauge: replacing (M, N_) -> (M Q^{-T}, N_ Q)
# leaves M N_^T (and hence dynamics) invariant. Clustering is basis-dependent,
# so we rotate into a canonical basis where input-to-n overlaps are diagonal.

def rotate_loadings_to_input_basis(
    L: np.ndarray,
    meta: dict,
    input_dim_per_rank: list[int],
) -> tuple[np.ndarray, np.ndarray]:
    """
    Choose Q (R x R) so that <I^{d_r}, n^r> is dominant on the r-th axis,
    by aligning n's columns to the rows of W_i over the selected input dims.

    Parameters
    ----------
    input_dim_per_rank : list of length R
        For rank r, which input channel is supposed to load primarily on n^r.
        e.g. for DPA with rank 2 you might pass [0, 2] (sample on n1, test on n2).

    Returns
    -------
    L_rot : rotated loadings (same shape)
    Q     : the rotation applied to n; m is rotated by Q^{-T}
    """
    off = meta["offsets"]
    R = meta["sizes"]["m"]
    assert len(input_dim_per_rank) == R

    M = L[:, off["m"]:off["m"] + R].copy()
    N_ = L[:, off["n"]:off["n"] + R].copy()
    Wi = L[:, off["in"]:off["in"] + meta["sizes"]["in"]].copy()

    # Target directions: rows of Wi at the chosen input dims, normalized.
    targets = np.stack([Wi[:, d] for d in input_dim_per_rank], axis=1)  # [N, R]
    targets /= (np.linalg.norm(targets, axis=0, keepdims=True) + 1e-12)

    # Find Q such that N_ @ Q is closest to `targets` (Procrustes).
    # Solve: min_Q || N_ Q - targets ||_F
    # => Q = (N_^T N_)^{-1} N_^T targets
    Q = np.linalg.lstsq(N_, targets, rcond=None)[0]  # [R, R]

    # Apply: n -> n Q, m -> m Q^{-T} (preserves M N_^T)
    Qinv_T = np.linalg.inv(Q).T
    M_rot = M @ Qinv_T
    N_rot = N_ @ Q

    L_rot = L.copy()
    L_rot[:, off["m"]:off["m"] + R] = M_rot
    L_rot[:, off["n"]:off["n"] + R] = N_rot
    return L_rot, Q


# ----------------------------------------------------------------------
# 3. GMM fitting + BIC
# ----------------------------------------------------------------------

@dataclass
class GMMFit:
    K: int
    gmm: GaussianMixture
    bic: float
    labels: np.ndarray            # [N]
    means: np.ndarray             # [K, P]
    covariances: np.ndarray       # [K, P, P]
    weights: np.ndarray           # [K]


def fit_gmms(
    L: np.ndarray,
    K_range=range(1, 7),
    standardize: bool = True,
    n_init: int = 10,
    random_state: int = 0,
) -> tuple[list[GMMFit], np.ndarray, np.ndarray]:
    """
    Fit GMMs with K in K_range. Standardization is applied to loadings before
    fitting; means/covariances are returned in the standardized space, with
    the (mu, sigma) used to standardize so you can map back.
    """
    if standardize:
        mu = L.mean(axis=0)
        sd = L.std(axis=0) + 1e-12
        Lz = (L - mu) / sd
    else:
        mu = np.zeros(L.shape[1])
        sd = np.ones(L.shape[1])
        Lz = L

    fits = []
    for K in K_range:
        gmm = GaussianMixture(
            n_components=K,
            covariance_type="full",
            n_init=n_init,
            random_state=random_state,
            reg_covar=1e-4,
        )
        gmm.fit(Lz)
        labels = gmm.predict(Lz)
        fits.append(GMMFit(
            K=K,
            gmm=gmm,
            bic=gmm.bic(Lz),
            labels=labels,
            means=gmm.means_,
            covariances=gmm.covariances_,
            weights=gmm.weights_,
        ))
    return fits, mu, sd


# ----------------------------------------------------------------------
# 4. Resample loadings from a fitted GMM and inject into the model
# ----------------------------------------------------------------------

def sample_loadings_from_gmm(
    fit: GMMFit,
    mu: np.ndarray,
    sd: np.ndarray,
    N: int,
    rng: np.random.Generator,
) -> np.ndarray:
    """Draw N loading vectors from the fitted mixture, in original (unscaled) units."""
    component = rng.choice(fit.K, size=N, p=fit.weights)
    P = mu.size
    out = np.empty((N, P))
    for k in range(fit.K):
        idx = np.where(component == k)[0]
        if idx.size == 0:
            continue
        out[idx] = rng.multivariate_normal(
            mean=fit.means[k],
            cov=fit.covariances[k],
            size=idx.size,
        )
    return out * sd + mu  # un-standardize


def inject_loadings(model, L_new: np.ndarray, meta: dict):
    """Overwrite model.m, model.n, model.wi.weight, model.wo.weight from L_new."""
    off = meta["offsets"]
    R = meta["sizes"]["m"]
    D_in = meta["sizes"]["in"]
    D_out = meta["sizes"]["out"]

    device = model.m.device
    dtype = model.m.dtype

    with torch.no_grad():
        model.m.copy_(torch.tensor(
            L_new[:, off["m"]:off["m"] + R], device=device, dtype=dtype))
        model.n.copy_(torch.tensor(
            L_new[:, off["n"]:off["n"] + R], device=device, dtype=dtype))
        if D_in > 0:
            model.wi.weight.copy_(torch.tensor(
                L_new[:, off["in"]:off["in"] + D_in], device=device, dtype=dtype))
        if D_out > 0 and model.wo is not None:
            model.wo.weight.copy_(torch.tensor(
                L_new[:, off["out"]:off["out"] + D_out].T, device=device, dtype=dtype))


# ----------------------------------------------------------------------
# 5. Performance metric
# ----------------------------------------------------------------------

@torch.no_grad()
def dpa_accuracy(
    model,
    inputs: torch.Tensor,
    targets: torch.Tensor,
    eval_window: int = 10,
) -> float:
    """
    Sign-accuracy on the final `eval_window` time steps, averaged per trial,
    restricted to trials whose target is +/-1 (not baseline).
    """
    device = next(model.parameters()).device
    inputs = inputs.to(device)
    targets = targets.to(device)

    pred = model(inputs, targets)             # [B, T, R]
    readout = pred[..., -1]                   # [B, T]

    y_final = readout[:, -eval_window:].mean(dim=1)         # [B]
    t_final = targets[:, -1]                                # [B]

    mask = (t_final == 1) | (t_final == -1)
    if mask.sum() == 0:
        return float("nan")

    correct = (torch.sign(y_final[mask]) == torch.sign(t_final[mask])).float()
    return correct.mean().item()

@dataclass
class ResamplingResult:
    K: int
    acc_mean: float
    acc_std: float
    acc_samples: list[float] = field(default_factory=list)


def resampling_control(
    model,
    fits: list[GMMFit],
    mu: np.ndarray,
    sd: np.ndarray,
    meta: dict,
    test_inputs: torch.Tensor,
    test_targets: torch.Tensor,
    n_resamples: int = 20,
    seed: int = 0,
) -> dict[int, ResamplingResult]:
    """
    For each fitted K, resample loadings n_resamples times, inject into a copy
    of the model, and report task accuracy.

    Returns dict K -> ResamplingResult.
    """
    # Snapshot original loadings to restore
    original_state = {
        "m": model.m.detach().clone(),
        "n": model.n.detach().clone(),
        "wi_w": model.wi.weight.detach().clone() if model.wi is not None else None,
        "wo_w": model.wo.weight.detach().clone() if model.wo is not None else None,
    }

    rng = np.random.default_rng(seed)
    results = {}

    try:
        for fit in fits:
            accs = []
            for _ in range(n_resamples):
                L_new = sample_loadings_from_gmm(fit, mu, sd, N=meta["N"], rng=rng)
                inject_loadings(model, L_new, meta)
                acc = dpa_accuracy(model, test_inputs, test_targets)
                accs.append(acc)
            accs = np.array(accs)
            results[fit.K] = ResamplingResult(
                K=fit.K,
                acc_mean=float(np.nanmean(accs)),
                acc_std=float(np.nanstd(accs)),
                acc_samples=accs.tolist(),
            )
    finally:
        # Restore original loadings no matter what.
        with torch.no_grad():
            model.m.copy_(original_state["m"])
            model.n.copy_(original_state["n"])
            if original_state["wi_w"] is not None:
                model.wi.weight.copy_(original_state["wi_w"])
            if original_state["wo_w"] is not None:
                model.wo.weight.copy_(original_state["wo_w"])

    return results

def run_dubreuil_analysis(
    model,
    test_inputs: torch.Tensor,
    test_targets: torch.Tensor,
    K_range=range(1, 7),
    n_resamples: int = 20,
    rotate_input_dims: Optional[list[int]] = None,
    seed: int = 0,
):
    """
    Full pipeline:
      1. Extract loadings.
      2. (Optionally) rotate to input-aligned basis.
      3. Fit GMMs across K_range, BIC.
      4. Compute trained-model baseline accuracy.
      5. Resample from each K, measure accuracy.
      6. Plot BIC, resampling curve, and cluster scatter at K*.
    """
    L, meta = extract_loadings(model)

    if rotate_input_dims is not None:
        L, Q = rotate_loadings_to_input_basis(L, meta, rotate_input_dims)
        print(f"Applied loading rotation; Q =\n{Q}")

    fits, mu, sd = fit_gmms(L, K_range=K_range, random_state=seed)
    bics = {f.K: f.bic for f in fits}
    K_star = min(bics, key=bics.get)
    print(f"BIC by K: {bics}")
    print(f"K* (min BIC) = {K_star}")

    baseline_acc = dpa_accuracy(model, test_inputs, test_targets)
    print(f"Trained model accuracy: {baseline_acc:.3f}")

    results = resampling_control(
        model, fits, mu, sd, meta,
        test_inputs, test_targets,
        n_resamples=n_resamples, seed=seed,
    )
    for K in sorted(results):
        r = results[K]
        print(f"  K={K}: acc = {r.acc_mean:.3f} ± {r.acc_std:.3f}")

    fig, axes = plt.subplots(1, 2, figsize=(2 * width, height))
    plot_bic(fits, ax=axes[0])
    plot_resampling(results, baseline_acc, ax=axes[1])
    plt.tight_layout()
    plt.show()

    fit_star = next(f for f in fits if f.K == K_star)
    plot_loading_clusters(L, meta, fit_star.labels)
    plt.show()

    return {
        "L": L, "meta": meta, "fits": fits, "mu": mu, "sd": sd,
        "K_star": K_star, "baseline_acc": baseline_acc,
        "results": results,
    }

**/ How to read the output

- /BIC plot/: location of the minimum suggests \(K^\star\). If BIC drops sharply from \(K=1\) to \(K=2\) and then plateaus, you have evidence for a
two-population solution.
- /Resampling plot/: the critical comparison.
  - If \(K=1\) (single Gaussian) resampling collapses to chance (≈0.5) and \(K=2\) or higher recovers near-trained accuracy, the multi-population structure is /necessary/.
  - If \(K=1\) already recovers near-trained performance, the network's effective solution is /unstructured/ — the apparent clusters in the scatter are not computationally required.
  - Variance across resamples tells you how robust the mean-field solution is at each \(K\).
- /Cluster scatter at K\*: gives you the population assignment to interpret each cluster's role (sign of \(m_r\), \(n_r\), input loadings).

/Caveat:/ resampled networks have new \(W_i\) too, so reward feedback through =model.reward_channel= still operates; this is the same regime as training. If you want to isolate the recurrent contribution, set =targets=None= in =dpa_accuracy= (turns off reward feedback) — but then the trained baseline will also drop, so compare both.

*** I would like to plot the gains of the different populations depending on the task epoch (delay, sample, test)

/Mode: Implement./

Population gain in the Dubreuil framework is the slope of the single-neuron transfer function evaluated at the population's mean input, weighted by the relevant overlap. The standard quantity is

\[
g_p(t) \;=\; \frac{1}{N_p}\sum_{i \in p} \phi'\!\big(u_i(t)\big), \qquad u_i(t)=\sum_r m_i^r \kappa_r(t)+\sum_k I_i^k x_k(t)+b_i,
\]

with \(\phi=\tanh\) so \(\phi'(u)=1-\tanh^2(u)\). This is the multiplicative factor that scales every effective overlap \(\sigma_{mn}, \sigma_{In}, \ldots\) contributed by population \(p\) at time \(t\). Plotting \(g_p\) across task epochs shows /when each population is engaged/. [cite:@dubreuil2022role]

You can also plot, with the same machinery, the /effective recurrent overlap/ per population, \(\tilde\sigma^{(p)}_{m^a n^b}(t) = \frac{1}{N_p}\sum_{i\in p} \phi'(u_i(t))\, m_i^a n_i^b\), which is the quantity that actually enters the rank-reduced dynamics. The function below returns both.

#+begin_src jupyter-python
import numpy as np
import torch
import matplotlib.pyplot as plt
from dataclasses import dataclass


@dataclass
class PopulationGains:
    time: np.ndarray                    # [T]
    gain_mean: np.ndarray               # [K, T]      avg over trials of <phi'>_p
    gain_sem:  np.ndarray               # [K, T]
    gain_cond: dict[str, np.ndarray]    # cond -> [K, T]   per condition
    eff_overlap_mn: np.ndarray          # [K, R, R, T]    <phi'·m^a n^b>_p, trial avg
    labels: np.ndarray                  # [N] population label per neuron
    sizes: np.ndarray                   # [K] neurons per population


@torch.no_grad()
def compute_population_gains(
    model,
    inputs: torch.Tensor,
    labels: np.ndarray,
    targets: torch.Tensor | None = None,
    condition_idx: dict[str, torch.Tensor] | None = None,
    dt: float | None = None,
):
    """
    Compute per-population gain g_p(t) = <phi'(u_i(t))>_{i in p}, averaged over trials.

    Parameters
    ----------
    model : trained LowRankModel
    inputs : [B, T, Din]
    labels : [N] integer cluster assignment from your GMM
    targets : optional, passed to model.forward to keep reward-feedback on
    condition_idx : dict name -> indices into the batch dim, for per-condition curves
    dt : if provided, time axis is in seconds; else in steps

    Returns
    -------
    PopulationGains
    """
    device = next(model.parameters()).device
    inputs = inputs.to(device)
    if targets is not None:
        targets = targets.to(device)

    # Get rates and rec_inputs trajectories.
    _, rates, rec_inputs = model(inputs, targets, ret_rates=True)
    # rates: [B, T, N]  (tanh activations -> we recover phi' = 1 - rates^2)
    phi_prime = 1.0 - rates.pow(2)                  # [B, T, N]

    B, T, N = phi_prime.shape
    labels = np.asarray(labels)
    K = int(labels.max()) + 1
    sizes = np.array([(labels == k).sum() for k in range(K)])

    # ---- per-population gain g_p(t), trial-averaged ----
    gain_per_trial = torch.zeros(B, K, T, device=device, dtype=phi_prime.dtype)
    for k in range(K):
        idx = np.where(labels == k)[0]
        if idx.size == 0:
            continue
        gain_per_trial[:, k, :] = phi_prime[:, :, idx].mean(dim=-1)

    gain_mean = gain_per_trial.mean(dim=0).cpu().numpy()             # [K, T]
    gain_sem = (gain_per_trial.std(dim=0)
                / np.sqrt(B)).cpu().numpy()                          # [K, T]

    # ---- per-condition gain curves ----
    gain_cond = {}
    if condition_idx is not None:
        for name, idx in condition_idx.items():
            idx = idx.to(device) if torch.is_tensor(idx) else torch.as_tensor(idx, device=device)
            if idx.numel() == 0:
                continue
            gain_cond[name] = gain_per_trial[idx].mean(dim=0).cpu().numpy()  # [K, T]

    # ---- effective recurrent overlap per population: <phi'·m^a n^b>_p ----
    # rec dynamics: kappa^+ = ... + (1/N) sum_i phi'·m^a n^b · kappa^b (linearization)
    # Per-population contribution: (N_p/N) <phi'·m^a n^b>_p
    M = model.m.detach()        # [N, R]
    Nv = (model.signed_n() if hasattr(model, "signed_n") else model.n).detach()  # [N, R]
    R = M.shape[1]

    eff_overlap = np.zeros((K, R, R, T))
    # Compute on device, then move to CPU once.
    for k in range(K):
        idx = np.where(labels == k)[0]
        if idx.size == 0:
            continue
        Mk = M[idx]                                 # [Nk, R]
        Nk = Nv[idx]                                # [Nk, R]
        # phi'_{b,t,i in k}: [B, T, Nk]
        pp_k = phi_prime[:, :, idx]
        # For each (a, b): mean over i of phi'_i * m_i^a * n_i^b
        # einsum: 'bti, ia, ib -> bt ab'  then mean over b
        mn = torch.einsum("bti,ip,iq->btpq", pp_k, Mk, Nk) / Mk.shape[0]  # [B, T, R, R]
        eff_overlap[k] = mn.mean(dim=0).permute(1, 2, 0).cpu().numpy()    # [R, R, T]

    time = np.arange(T) * (dt if dt is not None else 1.0)
    return PopulationGains(
        time=time,
        gain_mean=gain_mean,
        gain_sem=gain_sem,
        gain_cond=gain_cond,
        eff_overlap_mn=eff_overlap,
        labels=labels,
        sizes=sizes,
    )

/What to look for/

- Within each /task epoch/, the population whose curve rises is the one being /engaged/ — i.e. whose neurons are off-saturation and so contribute to the rank-reduced dynamics in that epoch.
- Compare /Sample/ vs /Test/ epochs: populations that gate the sample memory should peak during Sample and Delay; populations that implement the sample×test comparison should peak during Test/Response.
- Compare /per-condition/ curves: a population that flips its gain sign or amplitude across, e.g., AC vs AD trials, is the one carrying the conditional (XOR-like) computation.
- The /effective overlap/ \(\langle\phi'\,m^r n^r\rangle_p(t)\) tells you not just whether a population is active, but how strongly it contributes to the self-coupling on rank dimension \(r\) — a value near zero means that population is currently saturated out of the recurrent loop, even if its neurons are firing.
