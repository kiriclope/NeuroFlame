from torch.utils.data import TensorDataset, DataLoader, random_split

def train_val_split(X, y, batch_size=32, frac=0.8):
    # 1. Wrap tensors into a dataset
    dataset = TensorDataset(X.float(), y.float())

    # 2. Train/validation split
    n_total = len(dataset)
    n_train = int(frac * n_total)
    n_val = n_total - n_train

    generator = torch.Generator().manual_seed(0)
    train_dataset, val_dataset = random_split(dataset, lengths=[n_train, n_val], generator=generator)

    # 3. Create DataLoaders
    train_loader = DataLoader(train_dataset, batch_size, shuffle=True)
    val_loader = DataLoader(val_dataset, batch_size, shuffle=False)

    return train_loader, val_loader

import math
from typing import Any, Optional

class Optimization:
    """Small PyTorch train/validation loop."""

    def __init__(
        self,
        model: torch.nn.Module,
        train_loader: DataLoader,
        val_loader: DataLoader,
        criterion: torch.nn.Module,
        optimizer: torch.optim.Optimizer,
        scheduler: Any | None = None,
        grad_clip_norm: float | None = None,
        num_epochs: int = 100,
        stop_loss: float = 0.005,
        max_val_loss: float = 100.0,
        zero_low_rank_grad: Any | None = None,
        freeze_input_dims: Any | None = None,
        device: str | torch.device | None = None,
        verbose: bool = True,
    ):
        self.model = model
        self.train_loader = train_loader
        self.val_loader = val_loader
        self.criterion = criterion
        self.optimizer = optimizer
        self.num_epochs = num_epochs
        self.stop_loss = stop_loss
        self.max_val_loss = max_val_loss
        self.scheduler = scheduler
        self.zero_low_rank_grad = zero_low_rank_grad
        self.freeze_input_dims = freeze_input_dims
        self.grad_clip_norm = grad_clip_norm
        self.verbose = verbose

        self.device = torch.device(device) if device is not None else self._infer_device()
        self.model.to(self.device)
        self.model.device = self.device

        self.train_losses: list[float] = []
        self.val_losses: list[float] = []
        self.learning_rates: list[float] = []
        self.learning_rate = self.optimizer.param_groups[0]["lr"]

        # Store frozen input weights so they can be restored after optimizer.step().
        # This is safer than only zeroing gradients when using Adam/momentum/weight decay.
        self._frozen_wi_weight = None
        if self.freeze_input_dims is not None:
            self._frozen_wi_weight = self.model.wi.weight.detach().clone()

    def _infer_device(self) -> torch.device:
        if hasattr(self.model, "device"):
            return torch.device(self.model.device)
        return next(self.model.parameters()).device

    def _predict(self, X: torch.Tensor, y: Optional[torch.Tensor] = None) -> torch.Tensor:

        output = self.model(X, y)
        return self.model.readout if hasattr(self.model, "readout") else output

    def _zero_low_rank_grads(self) -> None:
        if self.zero_low_rank_grad is None:
            return

        for param in (self.model.m, self.model.n):
            if param.grad is None:
                continue

            if self.zero_low_rank_grad == "all":
                param.grad.zero_()
            else:
                param.grad[:, self.zero_low_rank_grad] = 0.0

    def _zero_input_grads(self) -> None:
        """
        Freeze selected input dimensions of model.wi.weight.

        For nn.Linear(input_size, hidden_size), weight has shape:
            (hidden_size, input_size)

        Therefore input dimensions are columns.
        """
        if self.freeze_input_dims is None:
            return

        if self.model.wi.weight.grad is not None:
            self.model.wi.weight.grad[:, self.freeze_input_dims] = 0.0

    def _restore_frozen_input_weights(self) -> None:
        """
        Restore frozen input columns after optimizer.step().

        This prevents Adam momentum or weight decay from changing frozen weights.
        """
        if self.freeze_input_dims is None:
            return

        with torch.no_grad():
            self.model.wi.weight[:, self.freeze_input_dims] = (
                self._frozen_wi_weight[:, self.freeze_input_dims]
            )

    def _run_epoch(self, loader: DataLoader, train: bool) -> float:
        self.model.train(train)

        total_loss = 0.0
        total_examples = 0
        context = torch.enable_grad() if train else torch.no_grad()

        with context:
            for X, y in loader:
                X = X.to(self.device)
                y = y.to(self.device)
                batch_size = X.size(0)

                if train:
                    self.optimizer.zero_grad(set_to_none=True)

                y_pred = self._predict(X, y)
                loss = self.criterion(y_pred, y)

                if not torch.isfinite(loss):
                    return float("nan")

                if train:
                    loss.backward()

                    self._zero_low_rank_grads()
                    self._zero_input_grads()

                    if self.grad_clip_norm is not None:
                        torch.nn.utils.clip_grad_norm_(
                            self.model.parameters(),
                            self.grad_clip_norm,
                        )

                    self.optimizer.step()
                    self._restore_frozen_input_weights()

                total_loss += loss.item() * batch_size
                total_examples += batch_size

        if total_examples == 0:
            raise ValueError("Dataloader produced zero examples.")

        return total_loss / total_examples

    def _step_scheduler(self, val_loss: float) -> None:
        if self.scheduler is None:
            self.learning_rate = self.optimizer.param_groups[0]["lr"]
            return

        if isinstance(self.scheduler, torch.optim.lr_scheduler.ReduceLROnPlateau):
            self.scheduler.step(val_loss)
        else:
            self.scheduler.step()

        self.learning_rate = self.scheduler.get_last_lr()[0]

    def fit(self) -> tuple[list[float], list[float], list[float]]:
        self.train_losses.clear()
        self.val_losses.clear()
        self.learning_rates.clear()

        for epoch in range(1, self.num_epochs + 1):
            train_loss = self._run_epoch(self.train_loader, train=True)
            val_loss = self._run_epoch(self.val_loader, train=False)

            self._step_scheduler(val_loss)

            self.train_losses.append(train_loss)
            self.val_losses.append(val_loss)
            self.learning_rates.append(self.learning_rate)

            if self.verbose:
                print(
                    f"Epoch {epoch:03d}/{self.num_epochs} | "
                    f"lr: {self.learning_rate:.4f} | "
                    f"train: {train_loss:.4f} | val: {val_loss:.4f}"
                )

            if not math.isfinite(train_loss) or not math.isfinite(val_loss):
                print("Stopping: non-finite loss.")
                break

            if train_loss < self.stop_loss and val_loss < self.stop_loss:
                print(f"Stopping: losses below {self.stop_loss}.")
                break

            if val_loss > self.max_val_loss:
                print(f"Stopping: validation loss above {self.max_val_loss}.")
                break

        return self.train_losses, self.val_losses, self.learning_rates

import torch
import torch.nn as nn

class MaskedZeroLoss(nn.Module):
    """
    Loss for sequence targets where target==0 means the network should stay silent.

    For y != 0:
        penalize output against target.

    For y == 0:
        penalize output against zero.

    Assumes:
    - y_pred.shape == y.shape, e.g. [batch, time]
    - time is dim 1
    """

    def __init__(
        self,
        criterion: nn.Module | None = None,
        nonzero_weight: float = 1.0,
        zero_weight: float = 1.0,
        zero_tol: float = 0.0,
    ):
        super().__init__()
        self.criterion = criterion or nn.MSELoss()
        self.nonzero_weight = nonzero_weight
        self.zero_weight = zero_weight
        self.zero_tol = zero_tol

    def forward(self, y_pred: torch.Tensor, y: torch.Tensor) -> torch.Tensor:

        valid_mask = torch.isfinite(y_pred[..., -1]) & torch.isfinite(y) & (y!=0.5)
        bl_mask = y==0.5

        nonzero_mask = valid_mask & (y.abs() > self.zero_tol)
        zero_mask = valid_mask & (y.abs() <= self.zero_tol)

        loss_bl = 0
        for i in range(y_pred.shape[-1]):
            loss_bl = loss_bl + self.criterion(y_pred[..., i][bl_mask], torch.zeros_like(y[bl_mask]))

        # loss_bl = self.criterion(y_pred[..., -1][bl_mask], torch.zeros_like(y[bl_mask]))
        loss_nonzero = self.criterion(y_pred[..., -1][nonzero_mask], y[nonzero_mask])
        loss_zero = self.criterion(y_pred[..., -1][zero_mask], torch.zeros_like(y[zero_mask]))

        return 0.1 * loss_bl + self.nonzero_weight * loss_nonzero + self.zero_weight * loss_zero

* Low Rank RNNs
** description
The Vanilla RNN updates its hidden state based on the previous hidden state and the current input:

Let:
- \( x_t \): feedforward input at time step \( t \)
- \( h_t \): recurrent input at time \( t \)
- \( r_t \): rate at time \( t \)
- \( y_t \): output at time \( t \)

\[h_t = (1-\alpha)h_{t-1} + \alpha \frac{1}{N} m n^\top r_{t-1}.\]

\[
r_t = \tanh(W_{ih} x_t + b_{ih} + h_{t-1})
\]

\[
y_t = \sigma(W_{ho} r_t + b_o)
\]

def init_low_rank_rnn(
    model,
    eigval: float = 0.3,
    input_scale: float = 0.2,
    readout_scale: float = 0.5,
):
    """
    Initialize rank-1 or low-rank RNN with controlled recurrent eigenvalues.

    For rank 1:
        J = n m.T / N
        lambda = m.T n / N ≈ eigval

    Assumes model update uses:
        hidden = (rates @ n) @ m.T / N
    """
    hidden_size, rank = model.m.shape
    device = model.m.device

    with torch.no_grad():
        # Input weights.
        nn.init.normal_(
            model.wi.weight,
            mean=0.0,
            std=input_scale / np.sqrt(model.wi.in_features),
        )
        nn.init.zeros_(model.wi.bias)

        # Low-rank factors.
        m = torch.randn(hidden_size, rank, device=device)
        q = torch.randn(hidden_size, rank, device=device)

        # Normalize columns to sqrt(N).
        m = m / m.norm(dim=0, keepdim=True) * np.sqrt(hidden_size)

        # Orthogonalize q against corresponding m columns.
        q = q - m * (q * m).sum(dim=0, keepdim=True) / hidden_size
        q = q / q.norm(dim=0, keepdim=True) * np.sqrt(hidden_size)

        # Choose n so that m.T @ n / N ≈ eigval on the diagonal.
        # The orthogonal component keeps n norm roughly sqrt(N).
        noise_amp = np.sqrt(max(0.0, 1.0 - eigval**2))
        n = eigval * m + noise_amp * q

        model.m.copy_(m)
        model.n.copy_(n)

        if model.wo is not None:
            nn.init.normal_(
                model.wo.weight,
                mean=0.0,
                std=readout_scale / np.sqrt(hidden_size),
            )
            nn.init.zeros_(model.wo.bias)

class LowRankModel(nn.Module):
    def __init__(self, input_size, hidden_size, output_size, rank=1, alpha=0.01, rho=0.5, train_inputs=True, device='cpu'):
        super().__init__()

        self.device = torch.device(device)
        self.input_size = input_size

        print(f'initializing low rank network')
        # input layer
        self.wi = None
        if input_size>0:
            print(f'with input layer {input_size} x {hidden_size}')
            self.wi = nn.Linear(input_size, hidden_size, device=self.device)

            if not train_inputs:
                self.wi.weight.requires_grad_(False)
                self.wi.bias.requires_grad_(False)

                nn.init.constant_(self.wi.weight, 1.0)
                nn.init.constant_(self.wi.bias, 0.0)

        # low rank recurrent layer
        print(f'with low rank vectors {hidden_size} x {rank}')
        self.m = nn.Parameter(torch.randn(hidden_size, rank, device=self.device))
        self.n = nn.Parameter(torch.randn(hidden_size, rank, device=self.device))

        # output layer
        self.wo = None
        if output_size > 0:
            print(f'with output layer {hidden_size} x {output_size}')
            self.wo = nn.Linear(hidden_size, output_size, device=self.device)

        self.nonlinearity = torch.tanh
        self.alpha = torch.tensor(alpha, device=self.device)
        self.exp_alpha = torch.exp(-self.alpha)

        # with torch.no_grad():
        #     self.m.copy_(self.n)

        init_low_rank_rnn(self, rho)

    def get_readout(self, rates, rec_inputs, is_internal=False):
        if self.wo is not None:
            return self.wo(rates).squeeze(-1)
        if is_internal: # ostojic
            return rec_inputs @ self.m / torch.norm(self.m[:, -1])**2
        # vreeswijk
        return rates @ self.n / rates.shape[-1]


    def update_dynamics(self, ff_inputs, rec_inputs, rates):
        # updating the recurrent inputs
        hidden = (rates @ self.n) @ self.m.T / rates.shape[-1]
        # hidden = rates @ (self.m @ self.n.T) / rates.shape[-1]

        # rec_inputs = (1.0 - self.alpha) * rec_inputs + self.alpha * hidden
        rec_inputs = self.exp_alpha * rec_inputs + (1.0 - self.exp_alpha) * hidden

        # updating the rates
        rates = self.nonlinearity(self.wi(ff_inputs) + rec_inputs)

        return rates, rec_inputs


    def forward(self, ff_inputs, targets=None, ret_rates=False):
        # rates and rec_inputs are (n_trials, hidden_size)
        batch_size = ff_inputs.shape[0]
        hidden_size = self.m.shape[0]

        # rec_inputs = torch.randn(batch_size, hidden_size, device=self.device)
        #rates = self.nonlinearity(self.wi(ff_inputs[:, -1]) + rec_inputs)

        rec_inputs = torch.randn(batch_size, hidden_size, device=self.device)
        rates = self.nonlinearity(rec_inputs)

        # one pass to steady state
        for step in range(ff_inputs.shape[1]):
            rand_inputs = torch.randn(batch_size, self.input_size, device=self.device)
            rates, rec_inputs = self.update_dynamics(rand_inputs, rec_inputs, rates)

        # main pass
        readout_list = []
        rates_list = []
        inputs_list = []

        # Online reward delivered through last input channel on next timestep.
        rwd_next = torch.zeros(batch_size, device=self.device, dtype=ff_inputs.dtype)

        for step in range(ff_inputs.shape[1]):
            x_t = ff_inputs[:, step].clone()

            if targets is not None:
                x_t[:, -1] = x_t[:, -1] + rwd_next

            rates, rec_inputs = self.update_dynamics(x_t, rec_inputs, rates)
            readout = self.get_readout(rates, rec_inputs)

            rates_list.append(rates)
            inputs_list.append(rec_inputs)
            readout_list.append(readout)

            if targets is not None:
                rwd_mask = (targets[:, step]==1) & (readout[:, -1]>0.5)
                rwd_next = rwd_mask.to(dtype=ff_inputs.dtype)
            else:
                rwd_next.zero_()

        rates_list = torch.stack(rates_list, dim=1)
        inputs_list = torch.stack(inputs_list, dim=1)
        readout_list = torch.stack(readout_list, dim=1)

        if ret_rates:
            return readout_list, rates_list, inputs_list

        return readout_list

* Vector fields
** Vector field

The low-rank recurrent state lives approximately on the manifold

\[
h = M \kappa,\qquad M = [m_1,m_2]\in \mathbb{R}^{N\times 2}.
\]

For a fixed feedforward input \(x\), the induced rank-coordinate dynamics are

\[
\kappa^{+}
= (1-\alpha)\kappa + \alpha \Psi(\kappa; x),
\]

where

\[
\Psi(\kappa; x)
= \frac{1}{N} n^\top \tanh(W_i x + b_i + M\kappa).
\]

Thus the phase portrait in the \((\kappa_1,\kappa_2)\) plane is given by the vector field

\[
\Delta \kappa
= \kappa^{+}-\kappa
= \alpha\left[\Psi(\kappa;x)-\kappa\right].
\]

Below I plot the autonomous / delay-period field using \(x=0\). During stimulus epochs the vector field is different because \(x_t\neq 0\).

def low_rank_numpy_params(model):
    """
    Extract trained low-rank RNN parameters as NumPy arrays.

    Coordinate convention:
        h = M κ

    with M = [m_1, ..., m_R] ∈ R^{N × R}. For a fixed input x,

        Ψ(κ; x) = N^{-1} n^T tanh(W_i x + b_i + Mκ)

    and the induced discrete κ-map is

        κ⁺ = (1 - α)κ + α Ψ(κ; x).
    """
    alpha = float(model.alpha.detach().cpu()) if torch.is_tensor(model.alpha) else float(model.alpha)
    return {
        "M": model.m.detach().cpu().numpy().astype(np.float64),
        "Nvec": model.n.detach().cpu().numpy().astype(np.float64),
        "Wi": model.wi.weight.detach().cpu().numpy().astype(np.float64),
        "bi": model.wi.bias.detach().cpu().numpy().astype(np.float64),
        "alpha": alpha,
        "beta": 1.0 - np.exp(-alpha),
    }


def low_rank_field_np(params, kappa, ff_input=None, include_alpha=False):
    """
    Low-rank vector field in κ-coordinates.

    Returns

        F(κ; x) = Ψ(κ; x) - κ

    unless include_alpha=True, in which case it returns αF.

    Fixed points satisfy F(κ; x)=0.
    """
    M = params["M"]        # (N, rank)
    Nvec = params["Nvec"]  # (N, rank)
    Wi = params["Wi"]      # (N, input_size)
    bi = params["bi"]      # (N,)

    kappa = np.asarray(kappa, dtype=np.float64)
    orig_shape = kappa.shape
    rank = M.shape[1]

    kappa_flat = kappa.reshape(-1, rank)  # (B, rank)

    if ff_input is None:
        ff_input = np.zeros(Wi.shape[1], dtype=np.float64)
    else:
        ff_input = np.asarray(ff_input, dtype=np.float64)

    drive = ff_input @ Wi.T + bi                  # (N,)
    rec_inputs = kappa_flat @ M.T                 # (B, N)
    rates = np.tanh(rec_inputs + drive[None, :])  # (B, N)

    psi = rates @ Nvec / M.shape[0]               # (B, rank)
    field = psi - kappa_flat                      # (B, rank)

    if include_alpha:
        field = params["alpha"] * field

    return field.reshape(orig_shape)


def low_rank_jacobian_flow_np(params, kappa, ff_input=None):
    """
    Jacobian of F(κ; x) = Ψ(κ; x) - κ.

    With

        u_i = m_i · κ + w_i · x + b_i,

    we have

        J_ab = N^{-1} Σ_i n_{ia} φ'(u_i) m_{ib} - δ_ab.
    """
    M = params["M"]
    Nvec = params["Nvec"]
    Wi = params["Wi"]
    bi = params["bi"]

    kappa = np.asarray(kappa, dtype=np.float64).reshape(-1)

    if ff_input is None:
        ff_input = np.zeros(Wi.shape[1], dtype=np.float64)
    else:
        ff_input = np.asarray(ff_input, dtype=np.float64)

    u = M @ kappa + ff_input @ Wi.T + bi
    phi_prime = 1.0 - np.tanh(u) ** 2

    J = Nvec.T @ (phi_prime[:, None] * M) / M.shape[0]
    J -= np.eye(M.shape[1])

    return J


def low_rank_jacobian_map_np(params, kappa, ff_input=None):
    """
    Jacobian of the discrete κ-map:

        κ⁺ = κ + αF(κ).

    Discrete-time stability is determined by eigvals of this matrix.
    """

    J_flow = low_rank_jacobian_flow_np(params, kappa, ff_input=ff_input)
    return np.eye(J_flow.shape[0]) + params["beta"] * J_flow


def project_rec_inputs_to_kappa(model, rec_inputs):
    """
    Least-squares projection of recurrent inputs h onto the low-rank basis M.

        h ≈ Mκ.

    Parameters
    ----------
    rec_inputs : torch.Tensor
        Shape (..., hidden_size).

    Returns
    -------
    kappa : torch.Tensor
        Shape (..., rank).
    """
    device = next(model.parameters()).device
    dtype = model.m.dtype

    h = torch.as_tensor(rec_inputs, device=device, dtype=dtype)
    orig_shape = h.shape

    hidden_size = model.m.shape[0]
    rank = model.m.shape[1]
    assert orig_shape[-1] == hidden_size

    h_flat = h.reshape(-1, hidden_size)  # (B, N)

    with torch.no_grad():
        # Solve M κ_b ≈ h_b for each batch element b.
        sol = torch.linalg.lstsq(model.m, h_flat.T).solution  # (rank, B)
        kappa = sol.T.reshape(*orig_shape[:-1], rank)

    return kappa


def condition_indices_from_inputs(inputs):
    """
    Infer DPA condition labels from stimulus channels.

    Assumes:
        channel 0 = A
        channel 1 = B
        channel 2 = C
        channel 3 = D
    """
    sample_epoch = slice(int(N_STIM_ON[0]), int(N_STIM_OFF[0]))
    test_epoch = slice(int(N_STIM_ON[1]), int(N_STIM_OFF[1]))

    sample_drive = inputs[:, sample_epoch, :2].mean(dim=1)
    test_drive = inputs[:, test_epoch, 2:4].mean(dim=1)

    is_A = sample_drive[:, 0] > sample_drive[:, 1]
    is_B = ~is_A

    is_C = test_drive[:, 0] > test_drive[:, 1]
    is_D = ~is_C

    return {
        "AC": torch.where(is_A & is_C)[0],
        "AD": torch.where(is_A & is_D)[0],
        "BC": torch.where(is_B & is_C)[0],
        "BD": torch.where(is_B & is_D)[0],
    }


def auto_limits_from_trajs(kappa_traj, qlo=1, qhi=99, pad=1.35, symmetric=True):
    """
    Infer axis limits from κ-trajectories.
    """
    all_kappa = kappa_traj.reshape(-1, kappa_traj.shape[-1])

    lo = np.percentile(all_kappa, qlo, axis=0)
    hi = np.percentile(all_kappa, qhi, axis=0)

    if symmetric:
        zlim = float(pad * np.max(np.abs(np.concatenate([lo, hi]))))
        return (-zlim, zlim), (-zlim, zlim)

    center = 0.5 * (lo + hi)
    half_width = 0.5 * (hi - lo) * pad

    xlim = (float(center[0] - half_width[0]), float(center[0] + half_width[0]))
    ylim = (float(center[1] - half_width[1]), float(center[1] + half_width[1]))

    return xlim, ylim


def make_vector_field_grid(
    model,
    ff_input=None,
    xlim=None,
    ylim=None,
    kappa_traj=None,
    n_grid=251,
    qlo=1,
    qhi=99,
    pad=1.35,
    symmetric_auto=True,
    include_alpha=False,
):
    """
    Build vector-field grid.

    If xlim/ylim are None, limits are inferred from kappa_traj.
    """
    params = low_rank_numpy_params(model)

    if ff_input is None:
        ff_input_np = np.zeros(params["Wi"].shape[1], dtype=np.float64)
    else:
        ff_input_np = torch.as_tensor(ff_input).detach().cpu().numpy().astype(np.float64)

    if xlim is None or ylim is None:
        if kappa_traj is None:
            raise ValueError("Provide xlim/ylim or kappa_traj for auto limits.")

        auto_xlim, auto_ylim = auto_limits_from_trajs(
            kappa_traj,
            qlo=qlo,
            qhi=qhi,
            pad=pad,
            symmetric=symmetric_auto,
        )

        if xlim is None:
            xlim = auto_xlim
        if ylim is None:
            ylim = auto_ylim

    k1 = np.linspace(xlim[0], xlim[1], n_grid)
    k2 = np.linspace(ylim[0], ylim[1], n_grid)

    K1, K2 = np.meshgrid(k1, k2)
    K = np.stack([K1, K2], axis=-1)  # (n_grid, n_grid, 2)

    field = low_rank_field_np(
        params,
        K,
        ff_input=ff_input_np,
        include_alpha=include_alpha,
    )

    U = field[..., 0]
    V = field[..., 1]
    speed = np.sqrt(U**2 + V**2)

    return K1, K2, U, V, speed, xlim, ylim, params, ff_input_np


def merge_roots(roots, residuals, merge_tol=5e-2):
    """
    Merge duplicate fixed points found from different initial seeds.
    """
    if len(roots) == 0:
        return np.empty((0, 2)), np.empty((0,))

    roots = np.asarray(roots, dtype=np.float64)
    residuals = np.asarray(residuals, dtype=np.float64)

    order = np.argsort(residuals)
    kept = []
    kept_residuals = []

    for idx in order:
        r = roots[idx]
        if all(np.linalg.norm(r - q) > merge_tol for q in kept):
            kept.append(r)
            kept_residuals.append(residuals[idx])

    return np.stack(kept, axis=0), np.asarray(kept_residuals)


def find_all_fixed_points(
    model,
    xlim,
    ylim,
    ff_input=None,
    n_seeds=65,
    residual_tol=1e-8,
    merge_tol=5e-2,
):
    """
    Find roots of F(κ; x)=0 inside the plotting window.
    """
    params = low_rank_numpy_params(model)

    if ff_input is None:
        ff_input_np = np.zeros(params["Wi"].shape[1], dtype=np.float64)
    else:
        ff_input_np = torch.as_tensor(ff_input).detach().cpu().numpy().astype(np.float64)

    def fun(k):
        return low_rank_field_np(params, k, ff_input=ff_input_np).reshape(-1)

    def jac(k):
        return low_rank_jacobian_flow_np(params, k, ff_input=ff_input_np)

    k1_seeds = np.linspace(xlim[0], xlim[1], n_seeds)
    k2_seeds = np.linspace(ylim[0], ylim[1], n_seeds)

    roots = []
    residuals = []

    for k1_0 in k1_seeds:
        for k2_0 in k2_seeds:
            sol = root(
                fun,
                np.array([k1_0, k2_0], dtype=np.float64),
                jac=jac,
                method="hybr",
                options={"xtol": 1e-12, "maxfev": 1000},
            )

            if not sol.success:
                continue

            fp = sol.x
            residual = np.linalg.norm(fun(fp))

            in_bounds = (
                xlim[0] <= fp[0] <= xlim[1]
                and ylim[0] <= fp[1] <= ylim[1]
            )

            if residual <= residual_tol and in_bounds:
                roots.append(fp)
                residuals.append(residual)

    return merge_roots(roots, residuals, merge_tol=merge_tol)


def classify_fixed_points(model, fixed_points, ff_input=None, eig_tol=1e-5):
    """
    Classify fixed points by eigenvalues of the discrete κ-map Jacobian.
    """
    params = low_rank_numpy_params(model)

    if ff_input is None:
        ff_input_np = np.zeros(params["Wi"].shape[1], dtype=np.float64)
    else:
        ff_input_np = torch.as_tensor(ff_input).detach().cpu().numpy().astype(np.float64)

    labels = []
    eigvals = []

    for fp in fixed_points:
        J_map = low_rank_jacobian_map_np(params, fp, ff_input=ff_input_np)
        ev = np.linalg.eigvals(J_map)
        radii = np.abs(ev)

        n_stable = np.sum(radii < 1.0 - eig_tol)
        n_unstable = np.sum(radii > 1.0 + eig_tol)

        if n_stable == len(radii):
            label = "attractor"
        elif n_unstable == len(radii):
            label = "repeller"
        elif n_stable > 0 and n_unstable > 0:
            label = "saddle"
        else:
            label = "nonhyperbolic"

        labels.append(label)
        eigvals.append(ev)

    if len(eigvals) == 0:
        return np.asarray([], dtype=object), np.empty((0, 2), dtype=np.complex128)

    return np.asarray(labels, dtype=object), np.stack(eigvals, axis=0)


def select_task_relevant_attractors(
    fixed_points,
    fp_labels,
    condition_mean_trajs,
    n_keep=3,
    task_window=None,
):
    """
    Keep attracting fixed points closest to task trajectories.

    This is a visualization filter; other mathematical attractors may exist.
    """
    attractors = fixed_points[fp_labels == "attractor"]

    if attractors.shape[0] <= n_keep:
        return attractors

    if task_window is None:
        task_window = slice(int(N_STIM_OFF[0]), None)

    anchors = []
    for traj in condition_mean_trajs.values():
        anchors.append(traj[task_window])

    anchors = np.concatenate(anchors, axis=0)

    dists = np.linalg.norm(
        attractors[:, None, :] - anchors[None, :, :],
        axis=-1,
    )
    scores = dists.min(axis=1)

    keep = np.argsort(scores)[:n_keep]
    return attractors[keep]


def default_arrow_times(T, n_arrows=8, arrow_step=3):
    """
    Choose time indices for trajectory arrows.

    Arrows point from κ_t to κ_{t+arrow_step}.
    """
    if T <= arrow_step + 1:
        return np.array([], dtype=int)

    lo = 1
    hi = T - arrow_step - 1

    n_arrows = min(n_arrows, max(1, hi - lo + 1))
    return np.unique(np.linspace(lo, hi, n_arrows).astype(int))


def add_time_arrows(
    ax,
    xy,
    times=None,
    color="k",
    arrow_step=3,
    mutation_scale=15,
    lw=1.8,
    alpha=1.0,
    zorder=8,
    min_length=1e-4,
):
    """
    Draw directional arrows along a 2D trajectory.

    Parameters
    ----------
    xy : array, shape (T, 2)
        Trajectory in the κ-plane.
    times : array-like or None
        Starting indices t for arrows from xy[t] to xy[t+arrow_step].
    """
    xy = np.asarray(xy, dtype=np.float64)
    T = xy.shape[0]

    if times is None:
        times = default_arrow_times(T, n_arrows=8, arrow_step=arrow_step)

    for t in times:
        t = int(t)
        if t < 0:
            t = T + t

        if t < 0 or t >= T - 1:
            continue

        j0 = t
        j1 = min(t + arrow_step, T - 1)

        # If the selected segment is too short, search forward.
        while j1 < T and np.linalg.norm(xy[j1] - xy[j0]) < min_length:
            j0 += 1
            j1 = min(j0 + arrow_step, T - 1)
            if j0 >= T - 1:
                break

        if j0 >= T - 1:
            continue

        p0 = xy[j0]
        p1 = xy[j1]

        if not (np.all(np.isfinite(p0)) and np.all(np.isfinite(p1))):
            continue

        if np.linalg.norm(p1 - p0) < min_length:
            continue

        ax.annotate(
            "",
            xy=p1,
            xytext=p0,
            arrowprops=dict(
                arrowstyle="-|>",
                color=color,
                lw=lw,
                alpha=alpha,
                shrinkA=0,
                shrinkB=0,
                mutation_scale=mutation_scale,
            ),
            zorder=zorder,
        )

    """

    candidate_marks = [
        (0, "o", "start"),

        # (int(N_STIM_ON[1]), "^", "test on"),

    ]

    marks = []
    for t, marker, label in candidate_marks:
        if 0 <= t < T:
            marks.append((t, marker, label))

    return marks

def make_input(input_size, active_dims=None, value=1.0, device=None, dtype=None):
    """Construct a fixed frozen input vector."""
    x = torch.zeros(input_size, device=device, dtype=dtype)

def low_rank_numpy_params(model):
    """
    Extract low-rank parameters.

    Uses signed_n() if present, so this also works with Dale/EI low-rank models.

    Coordinate convention:
        h = M κ
        Ψ(κ; x) = N^{-1} n^T tanh(W_i x + b_i + Mκ)
        κ⁺ = κ + β[Ψ(κ; x) - κ]
        β = 1 - exp(-alpha)
    """
    alpha = (
        float(model.alpha.detach().cpu())
        if torch.is_tensor(model.alpha)
        else float(model.alpha)
    )

    if hasattr(model, "signed_n"):
        n_eff = model.signed_n()
    else:
        n_eff = model.n

    return {
        "M": model.m.detach().cpu().numpy().astype(np.float64),
        "Nvec": n_eff.detach().cpu().numpy().astype(np.float64),
        "Wi": model.wi.weight.detach().cpu().numpy().astype(np.float64),
        "bi": model.wi.bias.detach().cpu().numpy().astype(np.float64),
        "alpha": alpha,
        "beta": 1.0 - np.exp(-alpha),
    }


def low_rank_field_np(params, kappa, ff_input=None, include_alpha=False):
    """
    Low-rank vector field in κ-coordinates.

        F(κ; x) = Ψ(κ; x) - κ

    If include_alpha=True, returns βF, matching the model update
    rec_inputs <- exp(-alpha) rec_inputs + (1-exp(-alpha)) hidden.
    """
    M = params["M"]
    Nvec = params["Nvec"]
    Wi = params["Wi"]
    bi = params["bi"]

    kappa = np.asarray(kappa, dtype=np.float64)
    orig_shape = kappa.shape
    rank = M.shape[1]
    kappa_flat = kappa.reshape(-1, rank)

    if ff_input is None:
        ff_input = np.zeros(Wi.shape[1], dtype=np.float64)
    else:
        ff_input = np.asarray(ff_input, dtype=np.float64)

    drive = ff_input @ Wi.T + bi                  # [N]
    rec_inputs = kappa_flat @ M.T                 # [B, N]
    rates = np.tanh(rec_inputs + drive[None, :])  # [B, N]

    psi = rates @ Nvec / M.shape[0]               # [B, rank]
    field = psi - kappa_flat

    if include_alpha:
        field = params["beta"] * field

    return field.reshape(orig_shape)


def low_rank_jacobian_flow_np(params, kappa, ff_input=None):
    """Jacobian of F(κ; x) = Ψ(κ; x) - κ."""
    M = params["M"]
    Nvec = params["Nvec"]
    Wi = params["Wi"]
    bi = params["bi"]

    kappa = np.asarray(kappa, dtype=np.float64).reshape(-1)

    if ff_input is None:
        ff_input = np.zeros(Wi.shape[1], dtype=np.float64)
    else:
        ff_input = np.asarray(ff_input, dtype=np.float64)

    u = M @ kappa + ff_input @ Wi.T + bi
    phi_prime = 1.0 - np.tanh(u) ** 2

    J = Nvec.T @ (phi_prime[:, None] * M) / M.shape[0]
    J -= np.eye(M.shape[1])

    return J


def low_rank_jacobian_map_np(params, kappa, ff_input=None):
    """Jacobian of the discrete κ-map κ⁺ = κ + βF(κ)."""
    J_flow = low_rank_jacobian_flow_np(params, kappa, ff_input=ff_input)
    return np.eye(J_flow.shape[0]) + params["beta"] * J_flow


def project_rec_inputs_to_kappa(model, rec_inputs):
    """
    Least-squares projection h ≈ Mκ.

    This is preferred over dividing by ||m|| because columns of M need not stay
    orthogonal after training.
    """
    device = _model_device(model)
    dtype = _model_dtype(model)

    h = torch.as_tensor(rec_inputs, device=device, dtype=dtype)
    orig_shape = h.shape

    hidden_size = model.m.shape[0]
    rank = model.m.shape[1]

    if orig_shape[-1] != hidden_size:
        raise ValueError(f"Expected last dim {hidden_size}, got {orig_shape[-1]}.")

    h_flat = h.reshape(-1, hidden_size)

    with torch.no_grad():
        sol = torch.linalg.lstsq(model.m, h_flat.T).solution  # [rank, B]
        kappa = sol.T.reshape(*orig_shape[:-1], rank)

    return kappa


def auto_limits_from_trajs(kappa_traj, qlo=1, qhi=99, pad=1.35, symmetric=True):
    """Infer plotting limits from κ-trajectories."""
    all_kappa = kappa_traj.reshape(-1, kappa_traj.shape[-1])

    lo = np.percentile(all_kappa, qlo, axis=0)
    hi = np.percentile(all_kappa, qhi, axis=0)

    if symmetric:
        zlim = float(pad * np.max(np.abs(np.concatenate([lo[:2], hi[:2]]))))
        zlim = max(zlim, 1e-6)
        return (-zlim, zlim), (-zlim, zlim)

    center = 0.5 * (lo + hi)
    half_width = 0.5 * (hi - lo) * pad

    return (
        (float(center[0] - half_width[0]), float(center[0] + half_width[0])),
        (float(center[1] - half_width[1]), float(center[1] + half_width[1])),
    )


def make_vector_field_grid(
    model,
    ff_input=None,
    xlim=None,
    ylim=None,
    kappa_traj=None,
    n_grid=180,
    qlo=1,
    qhi=99,
    pad=1.35,
    symmetric_auto=True,
    include_alpha=False,
):
    """Build vector-field grid."""
    params = low_rank_numpy_params(model)

    if ff_input is None:
        ff_input_np = np.zeros(params["Wi"].shape[1], dtype=np.float64)
    else:
        ff_input_np = torch.as_tensor(ff_input).detach().cpu().numpy().astype(np.float64)

    if xlim is None or ylim is None:
        if kappa_traj is None:
            raise ValueError("Provide xlim/ylim or kappa_traj.")

        auto_xlim, auto_ylim = auto_limits_from_trajs(
            kappa_traj,
            qlo=qlo,
            qhi=qhi,
            pad=pad,
            symmetric=symmetric_auto,
        )

        xlim = auto_xlim if xlim is None else xlim
        ylim = auto_ylim if ylim is None else ylim

    k1 = np.linspace(xlim[0], xlim[1], n_grid)
    k2 = np.linspace(ylim[0], ylim[1], n_grid)

    K1, K2 = np.meshgrid(k1, k2)
    K = np.stack([K1, K2], axis=-1)

    field = low_rank_field_np(
        params,
        K,
        ff_input=ff_input_np,
        include_alpha=include_alpha,
    )

    U = field[..., 0]
    V = field[..., 1]
    speed = np.sqrt(U**2 + V**2)

    return K1, K2, U, V, speed, xlim, ylim, params, ff_input_np


def merge_roots(roots, residuals, merge_tol=5e-2):
    """Merge duplicate fixed points."""
    if len(roots) == 0:
        return np.empty((0, 2)), np.empty((0,))

    roots = np.asarray(roots, dtype=np.float64)
    residuals = np.asarray(residuals, dtype=np.float64)

    order = np.argsort(residuals)
    kept = []
    kept_residuals = []

    for idx in order:
        r = roots[idx]
        if all(np.linalg.norm(r - q) > merge_tol for q in kept):
            kept.append(r)
            kept_residuals.append(residuals[idx])

    return np.stack(kept, axis=0), np.asarray(kept_residuals)


def find_all_fixed_points(
    model,
    xlim,
    ylim,
    ff_input=None,
    n_seeds=45,
    residual_tol=1e-8,
    merge_tol=5e-2,
):
    """Find roots of F(κ; x)=0 inside the plotting window."""
    params = low_rank_numpy_params(model)

    if ff_input is None:
        ff_input_np = np.zeros(params["Wi"].shape[1], dtype=np.float64)
    else:
        ff_input_np = torch.as_tensor(ff_input).detach().cpu().numpy().astype(np.float64)

    def fun(k):
        return low_rank_field_np(params, k, ff_input=ff_input_np).reshape(-1)

    def jac(k):
        return low_rank_jacobian_flow_np(params, k, ff_input=ff_input_np)

    k1_seeds = np.linspace(xlim[0], xlim[1], n_seeds)
    k2_seeds = np.linspace(ylim[0], ylim[1], n_seeds)

    roots = []
    residuals = []

    for k1_0 in k1_seeds:
        for k2_0 in k2_seeds:
            sol = root(
                fun,
                np.array([k1_0, k2_0], dtype=np.float64),
                jac=jac,
                method="hybr",
                options={"xtol": 1e-12, "maxfev": 1000},
            )

            if not sol.success:
                continue

            fp = sol.x
            residual = np.linalg.norm(fun(fp))

            in_bounds = (
                xlim[0] <= fp[0] <= xlim[1]
                and ylim[0] <= fp[1] <= ylim[1]
            )

            if residual <= residual_tol and in_bounds:
                roots.append(fp)
                residuals.append(residual)

    return merge_roots(roots, residuals, merge_tol=merge_tol)


def classify_fixed_points(model, fixed_points, ff_input=None, eig_tol=1e-5):
    """Classify fixed points using discrete-time map eigenvalues."""
    params = low_rank_numpy_params(model)

    if ff_input is None:
        ff_input_np = np.zeros(params["Wi"].shape[1], dtype=np.float64)
    else:
        ff_input_np = torch.as_tensor(ff_input).detach().cpu().numpy().astype(np.float64)

    labels = []
    eigvals = []

    for fp in fixed_points:
        J_map = low_rank_jacobian_map_np(params, fp, ff_input=ff_input_np)
        ev = np.linalg.eigvals(J_map)
        radii = np.abs(ev)

        n_stable = np.sum(radii < 1.0 - eig_tol)
        n_unstable = np.sum(radii > 1.0 + eig_tol)

        if n_stable == len(radii):
            label = "attractor"
        elif n_unstable == len(radii):
            label = "repeller"
        elif n_stable > 0 and n_unstable > 0:
            label = "saddle"
        else:
            label = "nonhyperbolic"

        labels.append(label)
        eigvals.append(ev)

    if len(eigvals) == 0:
        return np.asarray([], dtype=object), np.empty((0, 2), dtype=np.complex128)

    return np.asarray(labels, dtype=object), np.stack(eigvals, axis=0)


def default_arrow_times(T, n_arrows=4, arrow_step=3):
    """Choose time indices for trajectory arrows."""
    if T <= arrow_step + 1:
        return np.array([], dtype=int)

    lo = 1
    hi = T - arrow_step - 1
    n_arrows = min(n_arrows, max(1, hi - lo + 1))

    return np.unique(np.linspace(lo, hi, n_arrows).astype(int))


def add_time_arrows(
    ax,
    xy,
    times=None,
    color="k",
    arrow_step=3,
    mutation_scale=13,
    lw=1.5,
    alpha=0.95,
    zorder=8,
    min_length=1e-4,
):
    """Draw directional arrows along a 2D trajectory."""
    xy = np.asarray(xy, dtype=np.float64)
    T = xy.shape[0]

    if times is None:
        times = default_arrow_times(T, n_arrows=4, arrow_step=arrow_step)

    for t in times:
        t = int(t)

        if t < 0:
            t = T + t

        if t < 0 or t >= T - 1:
            continue

        j0 = t
        j1 = min(t + arrow_step, T - 1)

        while j1 < T and np.linalg.norm(xy[j1] - xy[j0]) < min_length:
            j0 += 1
            j1 = min(j0 + arrow_step, T - 1)

            if j0 >= T - 1:
                break

        if j0 >= T - 1:
            continue

        p0 = xy[j0]
        p1 = xy[j1]

        if not (np.all(np.isfinite(p0)) and np.all(np.isfinite(p1))):
            continue

        if np.linalg.norm(p1 - p0) < min_length:
            continue

        ax.annotate(
            "",
            xy=p1,
            xytext=p0,
            arrowprops=dict(
                arrowstyle="-|>",
                color=color,
                lw=lw,
                alpha=alpha,
                shrinkA=0,
                shrinkB=0,
                mutation_scale=mutation_scale,
            ),
            zorder=zorder,
        )


def _as_epoch_list(epochs):
    """Normalize one slice or list of slices to a list."""
    if isinstance(epochs, slice):
        return [epochs]
    return list(epochs)


def _no_input_epochs(timing):
    """
    Epochs where the mean task input is zero.

    This avoids overlaying stimulus-driven trajectory segments on the autonomous
    vector field.
    """
    n_on = [int(x) for x in timing.n_stim_on]
    n_off = [int(x) for x in timing.n_stim_off]
    n_steps = int(timing.n_steps)

    epochs = []

    # Before first stimulus.
    if n_on[0] > 0:
        epochs.append(slice(0, n_on[0]))

    # Gaps between stimuli.
    for j in range(len(n_on) - 1):
        if n_off[j] < n_on[j + 1]:
            epochs.append(slice(n_off[j], n_on[j + 1]))

    # After last stimulus.
    if n_off[-1] < n_steps:
        epochs.append(slice(n_off[-1], n_steps))

    return epochs


def condition_indices_for_task(inputs, timing, task, condition_names=None):
    """
    Build task-specific condition groups.

    DPA labels:
        AC, AD, BC, BD

    GNG labels:
        Go, NoGo

    Dual labels:
        AC, AD, BC, BD, Go, NoGo, DPAOnly
    """
    task = task.lower()
    n_on = timing.n_stim_on
    n_off = timing.n_stim_off

    if task == "dpa":
        sample_epoch = slice(int(n_on[0]), int(n_off[0]))
        test_epoch = slice(int(n_on[1]), int(n_off[1]))

        sample_drive = inputs[:, sample_epoch, :2].mean(dim=1)
        test_drive = inputs[:, test_epoch, 2:4].mean(dim=1)

        is_A = sample_drive[:, 0] > sample_drive[:, 1]
        is_B = ~is_A
        is_C = test_drive[:, 0] > test_drive[:, 1]
        is_D = ~is_C

        return {
            "AC": torch.where(is_A & is_C)[0],
            "AD": torch.where(is_A & is_D)[0],
            "BC": torch.where(is_B & is_C)[0],
            "BD": torch.where(is_B & is_D)[0],
        }

    if task == "gng":
        stim_epoch = slice(int(n_on[0]), int(n_off[0]))
        stim_drive = inputs[:, stim_epoch, 4:6].mean(dim=1)

        is_go = stim_drive[:, 0] > stim_drive[:, 1]
        is_nogo = ~is_go

        return {
            "Go": torch.where(is_go)[0],
            "NoGo": torch.where(is_nogo)[0],
        }

    if task == "dual":
        if condition_names is not None:
            names = np.asarray(condition_names).astype(str)

            groups = {
                "AC": [],
                "AD": [],
                "BC": [],
                "BD": [],
                "Go": [],
                "NoGo": [],
                "DPAOnly": [],
            }

            for i, name in enumerate(names):
                parts = name.split("_")

                if len(parts) == 2:
                    sample, test = parts
                    gng = "none"
                elif len(parts) == 3:
                    sample, gng, test = parts
                else:
                    raise ValueError(f"Unexpected condition name: {name}")

                pair_label = f"{sample}{test}"
                if pair_label in groups:
                    groups[pair_label].append(i)

                if gng == "go":
                    groups["Go"].append(i)
                elif gng == "nogo":
                    groups["NoGo"].append(i)
                elif gng == "none":
                    groups["DPAOnly"].append(i)

            return {
                key: torch.as_tensor(val, dtype=torch.long)
                for key, val in groups.items()
            }

        # Fallback: infer from input channels.
        sample_epoch = slice(int(n_on[0]), int(n_off[0]))
        gng_epoch = slice(int(n_on[1]), int(n_off[1]))
        test_epoch = slice(int(n_on[3]), int(n_off[3]))

        sample_drive = inputs[:, sample_epoch, :2].mean(dim=1)
        test_drive = inputs[:, test_epoch, 2:4].mean(dim=1)
        gng_drive = inputs[:, gng_epoch, 4:6].mean(dim=1)

        is_A = sample_drive[:, 0] > sample_drive[:, 1]
        is_B = ~is_A
        is_C = test_drive[:, 0] > test_drive[:, 1]
        is_D = ~is_C
        is_go = gng_drive[:, 0] > 0.25
        is_nogo = gng_drive[:, 1] > 0.25
        is_dpa_only = ~(is_go | is_nogo)

        return {
            "AC": torch.where(is_A & is_C)[0],
            "AD": torch.where(is_A & is_D)[0],
            "BC": torch.where(is_B & is_C)[0],
            "BD": torch.where(is_B & is_D)[0],
            "Go": torch.where(is_go)[0],
            "NoGo": torch.where(is_nogo)[0],
            "DPAOnly": torch.where(is_dpa_only)[0],
        }

    raise ValueError(f"Unknown task: {task}")


def flow_specs_for_task(timing, task):
    """
    Return task-correct frozen-input flow panels.

    Each spec contains:
        name   : panel title
        dims   : active input channels, or None for autonomous
        conds  : condition-mean trajectories to overlay
        epochs : trajectory epoch(s) consistent with the frozen input
    """
    task = task.lower()
    n_on = [int(x) for x in timing.n_stim_on]
    n_off = [int(x) for x in timing.n_stim_off]

    auto_epochs = _no_input_epochs(timing)

    if task == "dpa":
        return [
            dict(name="Autonomous", dims=None, conds=["AC", "AD", "BC", "BD"], epochs=auto_epochs),
            dict(name="A", dims=[0], conds=["AC", "AD"], epochs=slice(n_on[0], n_off[0])),
            dict(name="B", dims=[1], conds=["BC", "BD"], epochs=slice(n_on[0], n_off[0])),
            dict(name="C", dims=[2], conds=["AC", "BC"], epochs=slice(n_on[1], n_off[1])),
            dict(name="D", dims=[3], conds=["AD", "BD"], epochs=slice(n_on[1], n_off[1])),
        ]

    if task == "gng":
        return [
            dict(name="Autonomous", dims=None, conds=["Go", "NoGo"], epochs=auto_epochs),
            dict(name="Go", dims=[4], conds=["Go"], epochs=slice(n_on[0], n_off[0])),
            dict(name="NoGo", dims=[5], conds=["NoGo"], epochs=slice(n_on[0], n_off[0])),
            dict(name="Cue", dims=[6], conds=["Go", "NoGo"], epochs=slice(n_on[1], n_off[1])),
        ]

    if task == "dual":
        return [
            dict(name="Autonomous", dims=None, conds=["AC", "AD", "BC", "BD"], epochs=auto_epochs),
            dict(name="A", dims=[0], conds=["AC", "AD"], epochs=slice(n_on[0], n_off[0])),
            dict(name="B", dims=[1], conds=["BC", "BD"], epochs=slice(n_on[0], n_off[0])),
            dict(name="Go", dims=[4], conds=["Go"], epochs=slice(n_on[1], n_off[1])),
            dict(name="NoGo", dims=[5], conds=["NoGo"], epochs=slice(n_on[1], n_off[1])),
            dict(name="Cue", dims=[6], conds=["Go", "NoGo"], epochs=slice(n_on[2], n_off[2])),
            dict(name="C", dims=[2], conds=["AC", "BC"], epochs=slice(n_on[3], n_off[3])),
            dict(name="D", dims=[3], conds=["AD", "BD"], epochs=slice(n_on[3], n_off[3])),
        ]

    raise ValueError(f"Unknown task: {task}")


def plot_task_flow_fields(
    model,
    inputs,
    timing,
    task,
    condition_names=None,
    xlim=None,
    ylim=None,
    n_grid=180,
    n_fp_seeds=45,
    include_alpha_in_field=False,
    qlo=1,
    qhi=99,
    pad=1.35,
    symmetric_auto=True,
    speed_percentile=98,
    figsize_per_panel=5.0,
):
    """
    Plot task-correct low-rank vector fields and matching trajectory segments.

    Parameters
    ----------
    model : LowRankModel-like
        Must expose m, n or signed_n(), wi, alpha, and forward(..., ret_rates=True).
    inputs : torch.Tensor
        Shape [trials, time, input_size].
    timing : TaskTiming
        Explicit task timing object.
    task : str
        One of {"dpa", "gng", "dual"}.
    condition_names : array-like or None
        Recommended for dual task; names such as "A_C", "A_go_D", etc.
    """
    task = task.lower()

    if model.m.shape[1] < 2:
        raise ValueError("Flow-field plotting requires rank >= 2.")

    model.eval()

    device = _model_device(model)
    dtype = _model_dtype(model)
    input_size = inputs.shape[-1]

    specs = flow_specs_for_task(timing, task)

    max_dim = max(
        [d for spec in specs if spec["dims"] is not None for d in spec["dims"]],
        default=-1,
    )
    if input_size <= max_dim:
        raise ValueError(f"Task {task} requires input_size > {max_dim}, got {input_size}.")

    # ------------------------------------------------------------------
    # Run trajectories and project recurrent currents to κ coordinates.
    # ------------------------------------------------------------------
    with torch.no_grad():
        _, _, rec_inputs = model(inputs.to(device=device, dtype=dtype), ret_rates=True)

    kappa_traj = (
        project_rec_inputs_to_kappa(model, rec_inputs.detach())
        .detach()
        .cpu()
        .numpy()
    )

    cond_idx = condition_indices_for_task(
        inputs.detach().cpu(),
        timing=timing,
        task=task,
        condition_names=condition_names,
    )

    condition_mean_trajs = {}
    for name, idx in cond_idx.items():
        if len(idx) == 0:
            continue
        idx_np = idx.detach().cpu().numpy()
        condition_mean_trajs[name] = kappa_traj[idx_np].mean(axis=0)

    colors = {
        "AC": "tab:blue",
        "AD": "tab:orange",
        "BC": "tab:green",
        "BD": "tab:purple",
        "Go": "tab:red",
        "NoGo": "tab:brown",
        "DPAOnly": "tab:gray",
    }

    # ------------------------------------------------------------------
    # Precompute fields and fixed points, with common speed scale.
    # ------------------------------------------------------------------
    field_cache = {}
    all_speeds = []

    for spec in specs:
        ff_input = make_input(
            input_size,
            active_dims=spec["dims"],
            value=1.0,
            device=device,
            dtype=dtype,
        )

        K1, K2, U, V, speed, xlim_flow, ylim_flow, params, ff_input_np = make_vector_field_grid(
            model,
            ff_input=ff_input,
            xlim=xlim,
            ylim=ylim,
            kappa_traj=kappa_traj[..., :2],
            n_grid=n_grid,
            qlo=qlo,
            qhi=qhi,
            pad=pad,
            symmetric_auto=symmetric_auto,
            include_alpha=include_alpha_in_field,
        )

        fixed_points, fixed_residuals = find_all_fixed_points(
            model,
            xlim=xlim_flow,
            ylim=ylim_flow,
            ff_input=ff_input,
            n_seeds=n_fp_seeds,
            residual_tol=1e-8,
            merge_tol=5e-2,
        )

        fp_labels, fp_eigvals = classify_fixed_points(
            model,
            fixed_points,
            ff_input=ff_input,
            eig_tol=1e-5,
        )

        field_cache[spec["name"]] = dict(
            spec=spec,
            ff_input=ff_input,
            K1=K1,
            K2=K2,
            U=U,
            V=V,
            speed=speed,
            xlim=xlim_flow,
            ylim=ylim_flow,
            fixed_points=fixed_points,
            fp_labels=fp_labels,
            fp_eigvals=fp_eigvals,
            fixed_residuals=fixed_residuals,
        )

        all_speeds.append(speed[np.isfinite(speed)].reshape(-1))

    speed_vmax = np.percentile(np.concatenate(all_speeds), speed_percentile)

    # ------------------------------------------------------------------
    # Plot.
    # ------------------------------------------------------------------
    n_panels = len(specs)
    fig, axes = plt.subplots(
        1,
        n_panels,
        figsize=(n_panels * figsize_per_panel, figsize_per_panel),
        sharex=True,
        sharey=True,
    )

    axes = np.atleast_1d(axes).ravel()
    last_hm = None

    for ax, spec in zip(axes, specs):
        cache = field_cache[spec["name"]]

        K1 = cache["K1"]
        K2 = cache["K2"]
        U = cache["U"]
        V = cache["V"]
        speed = cache["speed"]
        fixed_points = cache["fixed_points"]
        fp_labels = cache["fp_labels"]

        k1 = K1[0, :]
        k2 = K2[:, 0]

        last_hm = ax.pcolormesh(
            K1,
            K2,
            speed,
            shading="auto",
            cmap="magma",
            vmax=speed_vmax,
            rasterized=True,
            zorder=0,
        )

        ax.streamplot(
            k1,
            k2,
            U,
            V,
            color="white",
            density=1.15,
            linewidth=0.7,
            arrowsize=0.8,
            zorder=2,
        )

        # Fixed points for this frozen input.
        for label, marker in [
            ("attractor", "o"),
            ("saddle", "x"),

# Corrected low-rank flow-field plotting.
#
# Main fixes:
# 1. Trajectories are projected with the same coordinate convention used by the
#    vector field: h ≈ M κ, via least squares.
# 2. The trajectory runner records the *effective* input used by LowRankModel,
#    including the online reward feedback through the last input channel when
#    targets are supplied.
# 3. Autonomous trajectory overlays exclude time points where effective input is
#    not actually zero. This prevents reward-driven segments from being plotted
#    on the zero-input vector field.
# 4. The field uses beta = 1 - exp(-alpha), matching the model update.

import numpy as np
import torch
import matplotlib.pyplot as plt

def _model_device(model):
    return next(model.parameters()).device


def _model_dtype(model):
    return next(model.parameters()).dtype


def make_input(input_size, active_dims=None, value=1.0, device=None, dtype=None):
    x = torch.zeros(input_size, device=device, dtype=dtype)
    if active_dims is not None:
        x[active_dims] = value
    return x


def low_rank_numpy_params(model):
    alpha = (
        float(model.alpha.detach().cpu())
        if torch.is_tensor(model.alpha)
        else float(model.alpha)
    )

    if hasattr(model, "signed_n"):
        n_eff = model.signed_n()
    else:
        n_eff = model.n

    return {
        "M": model.m.detach().cpu().numpy().astype(np.float64),       # [N, R]
        "Nvec": n_eff.detach().cpu().numpy().astype(np.float64),      # [N, R]
        "Wi": model.wi.weight.detach().cpu().numpy().astype(np.float64),
        "bi": model.wi.bias.detach().cpu().numpy().astype(np.float64),
        "beta": 1.0 - np.exp(-alpha),
    }


def low_rank_field_np(params, kappa, ff_input=None, include_beta=False):
    """
    Vector field in coordinates h = Mκ.

        F(κ; x) = Ψ(κ; x) - κ
        Ψ(κ; x) = N^{-1} n^T tanh(W_i x + b_i + Mκ)

    If include_beta=True, returns βF, exactly matching the discrete update
    increment κ⁺ - κ for fixed x.
    """
    M = params["M"]
    Nvec = params["Nvec"]
    Wi = params["Wi"]
    bi = params["bi"]

    kappa = np.asarray(kappa, dtype=np.float64)
    orig_shape = kappa.shape
    rank = M.shape[1]
    kappa_flat = kappa.reshape(-1, rank)

    if ff_input is None:
        ff_input = np.zeros(Wi.shape[1], dtype=np.float64)
    else:
        ff_input = np.asarray(ff_input, dtype=np.float64)

    drive = ff_input @ Wi.T + bi                  # [N]
    h = kappa_flat @ M.T                          # [B, N]
    r = np.tanh(h + drive[None, :])               # [B, N]
    psi = r @ Nvec / M.shape[0]                   # [B, R]

    field = psi - kappa_flat
    if include_beta:
        field = params["beta"] * field

    return field.reshape(orig_shape)


def low_rank_jacobian_flow_np(params, kappa, ff_input=None):
    """Jacobian of F(κ; x) = Ψ(κ; x) - κ."""
    M = params["M"]
    Nvec = params["Nvec"]
    Wi = params["Wi"]
    bi = params["bi"]

    kappa = np.asarray(kappa, dtype=np.float64).reshape(-1)

    if ff_input is None:
        ff_input = np.zeros(Wi.shape[1], dtype=np.float64)
    else:
        ff_input = np.asarray(ff_input, dtype=np.float64)

    u = M @ kappa + ff_input @ Wi.T + bi
    phi_prime = 1.0 - np.tanh(u) ** 2

    J = Nvec.T @ (phi_prime[:, None] * M) / M.shape[0]
    J -= np.eye(M.shape[1])
    return J


def low_rank_jacobian_map_np(params, kappa, ff_input=None):
    """Jacobian of κ⁺ = κ + βF(κ)."""
    J_flow = low_rank_jacobian_flow_np(params, kappa, ff_input=ff_input)
    return np.eye(J_flow.shape[0]) + params["beta"] * J_flow


def project_rec_inputs_to_kappa(model, rec_inputs):
    """
    Least-squares coordinates for h ≈ Mκ.

    This is the coordinate system used by low_rank_field_np. Do not replace this
    by h@m/||m||^2 unless M columns are known orthogonal.
    """
    device = _model_device(model)
    dtype = _model_dtype(model)

    h = torch.as_tensor(rec_inputs, device=device, dtype=dtype)
    orig_shape = h.shape
    hidden_size = model.m.shape[0]

    if orig_shape[-1] != hidden_size:
        raise ValueError(f"Expected last dim {hidden_size}, got {orig_shape[-1]}.")

    h_flat = h.reshape(-1, hidden_size)  # [B, N]

    with torch.no_grad():
        sol = torch.linalg.lstsq(model.m, h_flat.T).solution  # [R, B]
        kappa = sol.T.reshape(*orig_shape[:-1], model.m.shape[1])

    return kappa


def run_low_rank_with_effective_inputs(model, ff_inputs, targets=None):
    """
    Run a LowRankModel-like network while recording the actual input x_t used.

    For your LowRankModel, this mirrors forward(), including online reward
    feedback through the last input channel when targets is not None.

    Returns
    -------
    readouts      : [B, T, R_or_output]
    rates         : [B, T, N]
    rec_inputs    : [B, T, N]
    effective_x   : [B, T, input_size]
    """
    device = _model_device(model)
    dtype = _model_dtype(model)

    ff_inputs = ff_inputs.to(device=device, dtype=dtype)
    if targets is not None:
        targets = targets.to(device=device)

    batch_size, T, input_size = ff_inputs.shape
    hidden_size = model.m.shape[0]

    rec_inputs = torch.randn(batch_size, hidden_size, device=device, dtype=dtype)
    rates = model.nonlinearity(rec_inputs)

    # Match LowRankModel warmup.
    for _ in range(T):
        rand_inputs = torch.randn(batch_size, input_size, device=device, dtype=dtype)
        rates, rec_inputs = model.update_dynamics(rand_inputs, rec_inputs, rates)

    readout_list = []
    rates_list = []
    rec_list = []
    eff_input_list = []

    rwd_next = torch.zeros(batch_size, device=device, dtype=dtype)

    for step in range(T):
        x_t = ff_inputs[:, step].clone()

        if targets is not None:
            x_t[:, -1] = x_t[:, -1] + rwd_next

        rates, rec_inputs = model.update_dynamics(x_t, rec_inputs, rates)
        readout = model.get_readout(rates, rec_inputs)

        readout_list.append(readout)
        rates_list.append(rates)
        rec_list.append(rec_inputs)
        eff_input_list.append(x_t)

        if targets is not None:
            rwd_mask = (targets[:, step] == 1) & (readout[:, -1] > 0.5)
            rwd_next = rwd_mask.to(dtype=dtype)
        else:
            rwd_next.zero_()

    return (
        torch.stack(readout_list, dim=1),
        torch.stack(rates_list, dim=1),
        torch.stack(rec_list, dim=1),
        torch.stack(eff_input_list, dim=1),
    )


def auto_limits_from_trajs(kappa_traj, qlo=1, qhi=99, pad=1.35, symmetric=True):
    z = kappa_traj.reshape(-1, kappa_traj.shape[-1])
    lo = np.percentile(z[:, :2], qlo, axis=0)
    hi = np.percentile(z[:, :2], qhi, axis=0)

    if symmetric:
        lim = float(pad * np.max(np.abs(np.concatenate([lo, hi]))))
        lim = max(lim, 1e-6)
        return (-lim, lim), (-lim, lim)

    center = 0.5 * (lo + hi)
    half = 0.5 * (hi - lo) * pad
    return (
        (float(center[0] - half[0]), float(center[0] + half[0])),
        (float(center[1] - half[1]), float(center[1] + half[1])),
    )


def make_vector_field_grid(
    model,
    ff_input,
    xlim,
    ylim,
    n_grid=180,
    include_beta=False,
):
    params = low_rank_numpy_params(model)
    ff_input_np = torch.as_tensor(ff_input).detach().cpu().numpy().astype(np.float64)

    k1 = np.linspace(xlim[0], xlim[1], n_grid)
    k2 = np.linspace(ylim[0], ylim[1], n_grid)
    K1, K2 = np.meshgrid(k1, k2)
    K = np.stack([K1, K2], axis=-1)

    field = low_rank_field_np(
        params,
        K,
        ff_input=ff_input_np,
        include_beta=include_beta,
    )

    U = field[..., 0]
    V = field[..., 1]
    speed = np.sqrt(U**2 + V**2)

    return K1, K2, U, V, speed, params, ff_input_np


def merge_roots(roots, residuals, merge_tol=5e-2):
    if len(roots) == 0:
        return np.empty((0, 2)), np.empty((0,))

    roots = np.asarray(roots, dtype=np.float64)
    residuals = np.asarray(residuals, dtype=np.float64)

    order = np.argsort(residuals)
    kept = []
    kept_res = []

    for idx in order:
        r = roots[idx]
        if all(np.linalg.norm(r - q) > merge_tol for q in kept):
            kept.append(r)
            kept_res.append(residuals[idx])

    return np.stack(kept, axis=0), np.asarray(kept_res)


def find_all_fixed_points(
    model,
    xlim,
    ylim,
    ff_input,
    n_seeds=45,
    residual_tol=1e-8,
    merge_tol=5e-2,
):
    params = low_rank_numpy_params(model)
    ff_input_np = torch.as_tensor(ff_input).detach().cpu().numpy().astype(np.float64)

    def fun(k):
        return low_rank_field_np(params, k, ff_input=ff_input_np).reshape(-1)

    def jac(k):
        return low_rank_jacobian_flow_np(params, k, ff_input=ff_input_np)

    k1_seeds = np.linspace(xlim[0], xlim[1], n_seeds)
    k2_seeds = np.linspace(ylim[0], ylim[1], n_seeds)

    roots = []
    residuals = []

    for k1_0 in k1_seeds:
        for k2_0 in k2_seeds:
            sol = root(
                fun,
                np.array([k1_0, k2_0], dtype=np.float64),
                jac=jac,
                method="hybr",
                options={"xtol": 1e-12, "maxfev": 1000},
            )
            if not sol.success:
                continue

            fp = sol.x
            residual = np.linalg.norm(fun(fp))
            in_bounds = xlim[0] <= fp[0] <= xlim[1] and ylim[0] <= fp[1] <= ylim[1]

            if residual <= residual_tol and in_bounds:
                roots.append(fp)
                residuals.append(residual)

    return merge_roots(roots, residuals, merge_tol=merge_tol)


def classify_fixed_points(model, fixed_points, ff_input, eig_tol=1e-5):
    params = low_rank_numpy_params(model)
    ff_input_np = torch.as_tensor(ff_input).detach().cpu().numpy().astype(np.float64)

    labels = []
    eigvals = []

    for fp in fixed_points:
        J = low_rank_jacobian_map_np(params, fp, ff_input=ff_input_np)
        ev = np.linalg.eigvals(J)
        rad = np.abs(ev)

        n_stable = np.sum(rad < 1.0 - eig_tol)
        n_unstable = np.sum(rad > 1.0 + eig_tol)

        if n_stable == len(rad):
            label = "attractor"
        elif n_unstable == len(rad):
            label = "repeller"
        elif n_stable > 0 and n_unstable > 0:
            label = "saddle"
        else:
            label = "nonhyperbolic"

        labels.append(label)
        eigvals.append(ev)

    if len(eigvals) == 0:
        return np.asarray([], dtype=object), np.empty((0, 2), dtype=np.complex128)

    return np.asarray(labels, dtype=object), np.stack(eigvals, axis=0)


def condition_indices_for_task(inputs, timing, task, condition_names=None):
    task = task.lower()
    n_on = timing.n_stim_on
    n_off = timing.n_stim_off

    if task == "dpa":
        sample_epoch = slice(int(n_on[0]), int(n_off[0]))
        test_epoch = slice(int(n_on[1]), int(n_off[1]))

        sample_drive = inputs[:, sample_epoch, :2].mean(dim=1)
        test_drive = inputs[:, test_epoch, 2:4].mean(dim=1)

        is_A = sample_drive[:, 0] > sample_drive[:, 1]
        is_B = ~is_A
        is_C = test_drive[:, 0] > test_drive[:, 1]
        is_D = ~is_C

        return {
            "AC": torch.where(is_A & is_C)[0],
            "AD": torch.where(is_A & is_D)[0],
            "BC": torch.where(is_B & is_C)[0],
            "BD": torch.where(is_B & is_D)[0],
        }

    if task == "gng":
        stim_epoch = slice(int(n_on[0]), int(n_off[0]))
        stim_drive = inputs[:, stim_epoch, 4:6].mean(dim=1)

        is_go = stim_drive[:, 0] > stim_drive[:, 1]
        is_nogo = ~is_go

        return {
            "Go": torch.where(is_go)[0],
            "NoGo": torch.where(is_nogo)[0],
        }

    if task == "dual":
        if condition_names is not None:
            names = np.asarray(condition_names).astype(str)
            groups = {k: [] for k in ["AC", "AD", "BC", "BD", "Go", "NoGo", "DPAOnly"]}

                parts = name.split("_")
                if len(parts) == 2:
                    sample, test = parts
                    gng = "none"
                elif len(parts) == 3:
                    sample, gng, test = parts
                else:

                pair = f"{sample}{test}"
                if pair in groups:

                if gng == "go":

                elif gng == "nogo":
                    groups["NoGo"].append(i)
                elif gng == "none":
                    groups["DPAOnly"].append(i)

        sample_epoch = slice(int(n_on[0]), int(n_off[0]))
        gng_epoch = slice(int(n_on[1]), int(n_off[1]))
        test_epoch = slice(int(n_on[3]), int(n_off[3]))

        sample_drive = inputs[:, sample_epoch, :2].mean(dim=1)

def _canonical_input_mask(effective_x, dims, threshold=0.35, atol_inactive=0.35):
    """
    Boolean [B, T] mask selecting time points whose effective input matches a
    frozen-input panel.

    For dims=None, selects approximately zero-input points.
    For dims=[k], selects points where channel k is active and all other task
    channels are inactive. Noise is tolerated by thresholds.
    """
    x = effective_x.detach().cpu()

    if dims is None:
        return x.abs().amax(dim=-1) < atol_inactive

    dims = list(dims)
    active = torch.ones(x.shape[:2], dtype=torch.bool)
    for d in dims:
        active &= x[..., d] > threshold

    inactive_dims = [d for d in range(x.shape[-1]) if d not in dims]
    inactive = x[..., inactive_dims].abs().amax(dim=-1) < atol_inactive

    return active & inactive


def _segments_from_mask(mask_1d, min_len=2):
    idx = np.where(mask_1d)[0]
    if idx.size == 0:
        return []

    breaks = np.where(np.diff(idx) > 1)[0] + 1
    chunks = np.split(idx, breaks)

    out = []
    for c in chunks:
        if len(c) >= min_len:
            out.append(slice(int(c[0]), int(c[-1]) + 1))
    return out


def default_arrow_times(T, n_arrows=4, arrow_step=3):
    if T <= arrow_step + 1:
        return np.array([], dtype=int)

    lo = 1
    hi = T - arrow_step - 1
    n_arrows = min(n_arrows, max(1, hi - lo + 1))
    return np.unique(np.linspace(lo, hi, n_arrows).astype(int))


def add_time_arrows(
    ax,
    xy,
    times=None,
    color="k",
    arrow_step=3,
    mutation_scale=13,
    lw=1.5,
    alpha=0.95,
    zorder=8,
    min_length=1e-4,
):
    xy = np.asarray(xy, dtype=np.float64)
    T = xy.shape[0]

    if times is None:
        times = default_arrow_times(T, n_arrows=4, arrow_step=arrow_step)

    for t in times:
        t = int(t)
        if t < 0:
            t += T
        if t < 0 or t >= T - 1:
            continue

        j0 = t
        j1 = min(t + arrow_step, T - 1)

        while j1 < T and np.linalg.norm(xy[j1] - xy[j0]) < min_length:
            j0 += 1
            j1 = min(j0 + arrow_step, T - 1)
            if j0 >= T - 1:
                break

        if j0 >= T - 1:
            continue

        p0 = xy[j0]
        p1 = xy[j1]

        if np.linalg.norm(p1 - p0) < min_length:
            continue

        ax.annotate(
            "",
            xy=p1,
            xytext=p0,
            arrowprops=dict(
                arrowstyle="-|>",
                color=color,
                lw=lw,
                alpha=alpha,
                shrinkA=0,
                shrinkB=0,
                mutation_scale=mutation_scale,
            ),
            zorder=zorder,
        )

@dataclass(frozen=True)
class TaskTiming:
    stim_on: list[float]
    stim_off: list[float]
    t_steps: float
    dt: float

    @property
    def n_stim_on(self) -> torch.Tensor:
        return torch.tensor([int(t / self.dt) for t in self.stim_on], dtype=torch.long)

    @property
    def n_stim_off(self) -> torch.Tensor:
        return torch.tensor([int(t / self.dt) for t in self.stim_off], dtype=torch.long)

    @property
    def n_steps(self) -> int:
        return int(self.t_steps / self.dt)

    @property

        return np.linspace(0, self.n_steps * self.dt, self.n_steps)

def generate_dpa_trials(
    n_trials: int,
    timing: TaskTiming,
    input_size: int = 7,
    noise: float = 0.1,
    baseline_value: float = 0.5,
):
    n_steps = timing.n_steps
    n_on = timing.n_stim_on
    n_off = timing.n_stim_off

    inputs = noise * torch.randn(n_trials, n_steps, input_size)
    targets = torch.zeros(n_trials, n_steps)

    # First cue: A vs B.
    idx_A = torch.rand(n_trials) > 0.5
    idx_B = ~idx_A

    # Second cue: C vs D.
    idx_C = torch.rand(n_trials) > 0.5
    idx_D = ~idx_C

    # sample input
    inputs[idx_A, n_on[0]:n_off[0], 0] += 1.0
    inputs[idx_B, n_on[0]:n_off[0], 1] += 1.0

    # test input
    inputs[idx_C, n_on[1]:n_off[1], 2] += 1.0
    inputs[idx_D, n_on[1]:n_off[1], 3] += 1.0

    # Paired: AC or BD.
    idx_pair = (idx_A & idx_C) | (idx_B & idx_D)

    if baseline_value is not None:
        targets[:, :n_on[0]] = baseline_value

    targets[idx_pair, n_off[1]:] = 1.0
    targets[~idx_pair, n_off[1]:] = -1.0
    targets[:, n_on[1]:n_off[1]] = torch.nan

    return inputs, targets

def generate_dual_trials(
    n_trials: int,
    timing: TaskTiming,
    input_size: int = 8,
    noise: float = 0.1,
    baseline_value: float = 0.5,
):
    n_steps = timing.n_steps
    n_on = timing.n_stim_on
    n_off = timing.n_stim_off

    s_sample = slice(int(n_on[0]), int(n_off[0]))
    s_gng    = slice(int(n_on[1]), int(n_off[1]))
    s_cue    = slice(int(n_on[2]), int(n_off[2]))
    # s_rwd    = slice(int(n_off[2]), int(n_off[2] + (n_on[3] - n_off[2]) /2))
    s_rwd    = slice(int(n_off[2]), int(n_on[3]))
    s_test   = slice(int(n_on[3]), int(n_off[3]))

    specs = []
    for sample in ["A", "B"]:
        for gng in ["none", "go", "nogo"]:
            for test in ["C", "D"]:
                specs.append((sample, gng, test))

    n_types = len(specs)

    reps = int(np.ceil(n_trials / n_types))
    trial_type = torch.arange(n_types).repeat(reps)[:n_trials]
    trial_type = trial_type[torch.randperm(n_trials)]

    sample_code = torch.tensor(
        [0 if s == "A" else 1 for s, _, _ in specs],
        dtype=torch.long,
    )
    gng_code = torch.tensor(
        [0 if g == "none" else 1 if g == "go" else 2 for _, g, _ in specs],
        dtype=torch.long,
    )
    test_code = torch.tensor(
        [0 if t == "C" else 1 for _, _, t in specs],
        dtype=torch.long,
    )

    sample = sample_code[trial_type]
    gng = gng_code[trial_type]
    test = test_code[trial_type]

    idx_A = sample == 0
    idx_B = sample == 1
    idx_C = test == 0
    idx_D = test == 1

    idx_none = gng == 0
    idx_go = gng == 1
    idx_nogo = gng == 2
    idx_gng = idx_go | idx_nogo

    inputs = noise * torch.randn(n_trials, n_steps, input_size)
    targets = torch.zeros(n_trials, n_steps)

    if baseline_value is not None:
        targets[:, :s_sample.start] = baseline_value

    # DPA sample.
    inputs[idx_A, s_sample, 0] += 1.0
    inputs[idx_B, s_sample, 1] += 1.0

    # GNG stimulus.
    inputs[idx_go, s_gng, 4] += 1.0
    inputs[idx_nogo, s_gng, 5] += 1.0

    # Cue only on GNG trials.
    inputs[idx_gng, s_cue, 6] += 1.0

    # DPA test.
    inputs[idx_C, s_test, 2] += 1.0
    inputs[idx_D, s_test, 3] += 1.0

    # DPA final target.
    idx_pair = (idx_A & idx_C) | (idx_B & idx_D)
    targets[idx_pair, s_test.stop:] = 1.0
    targets[~idx_pair, s_test.stop:] = -1.0

    # GNG cue-period target.
    targets[idx_go, s_rwd] = 1.0
    targets[idx_nogo, s_rwd] = -1.0

    condition_names = np.array([
        f"{s}_{g}_{t}" if g != "none" else f"{s}_{t}"
        for s, g, t in specs
    ])[trial_type.numpy()]

    return inputs, targets, trial_type, condition_names

import math
import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F


class EILowRankModel(nn.Module):
    """
    Dale-constrained low-rank RNN, matched to LowRankModel's interface.

    Recurrent convention
    --------------------
    Effective recurrent matrix:

        W_rec[i, j] = N^{-1} sum_r m[i, r] * n[j, r] * dale_sign[j]

    where dale_sign[j] = +1 for excitatory presynaptic neurons and -1 for
    inhibitory presynaptic neurons. Since m,n >= 0, each presynaptic neuron has
    fixed outgoing sign.

    Dynamics
    --------
        hidden_t = (rates_{t-1} @ signed_n) @ m.T / N
        h_t      = exp(-alpha) h_{t-1} + (1 - exp(-alpha)) hidden_t
        r_t      = tanh(W_i x_t + b_i + h_t)

    Forward interface mirrors LowRankModel:
        model(ff_inputs, targets=None, ret_rates=False)

    If targets is supplied, online reward feedback is added through the final
    input channel on the next timestep, as in LowRankModel.
    """

    def __init__(
        self,
        input_size: int,
        hidden_size: int,
        output_size: int,
        rank: int = 1,
        alpha: float = 0.01,
        p_exc: float = 0.8,
        rho: float = 0.5,
        input_scale: float = 0.2,
        readout_scale: float = 0.5,
        train_inputs: bool = True,
        init_state_scale: float = 1.0,
        warmup_noise: bool = True,
        device: str | torch.device = "cpu",
    ):
        super().__init__()

        self.device = torch.device(device)
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.output_size = output_size
        self.rank = rank
        self.p_exc = p_exc
        self.init_state_scale = init_state_scale
        self.warmup_noise = warmup_noise

        print("initializing Dale-constrained EI low-rank network")

        # Input layer, kept compatible with LowRankModel/Optimization.
        self.wi = None
        if input_size > 0:
            print(f"with input layer {input_size} x {hidden_size}")
            self.wi = nn.Linear(input_size, hidden_size, device=self.device)

            if not train_inputs:
                self.wi.weight.requires_grad_(False)
                self.wi.bias.requires_grad_(False)
                nn.init.constant_(self.wi.weight, 1.0)
                nn.init.constant_(self.wi.bias, 0.0)

        # Raw low-rank factors. Positive factors are exposed through m,n
        # properties below.
        print(f"with positive EI low-rank vectors {hidden_size} x {rank}")
        self.m_raw = nn.Parameter(torch.randn(hidden_size, rank, device=self.device))
        self.n_raw = nn.Parameter(torch.randn(hidden_size, rank, device=self.device))

        # Output layer.
        self.wo = None
        if output_size > 0:
            print(f"with output layer {hidden_size} x {output_size}")
            self.wo = nn.Linear(hidden_size, output_size, device=self.device)

        self.nonlinearity = torch.tanh

        self.register_buffer("alpha", torch.tensor(float(alpha), device=self.device))
        self.register_buffer("exp_alpha", torch.exp(-self.alpha))

        # Dale signs: columns of W_rec have fixed signs.
        n_exc = int(round(p_exc * hidden_size))
        n_inh = hidden_size - n_exc

        dale_sign = torch.cat(
            [
                torch.ones(n_exc, device=self.device),
                -torch.ones(n_inh, device=self.device),
            ],
            dim=0,
        )
        dale_sign = dale_sign[torch.randperm(hidden_size, device=self.device)]

        self.register_buffer("dale_sign", dale_sign)
        self.register_buffer("is_exc", dale_sign > 0)
        self.register_buffer("is_inh", dale_sign < 0)

        self.reset_parameters(
            rho=rho,
            input_scale=input_scale,
            readout_scale=readout_scale,
            train_inputs=train_inputs,
        )

    @staticmethod
    def _inv_softplus(x: float) -> float:
        """Scalar inverse softplus, stable for small positive x."""
        x = max(float(x), 1e-8)
        return math.log(math.expm1(x))

    @property
    def m(self) -> torch.Tensor:
        """Nonnegative postsynaptic factors, shape [hidden_size, rank]."""
        return F.softplus(self.m_raw)

    @property
    def n(self) -> torch.Tensor:
        """Nonnegative presynaptic magnitudes, shape [hidden_size, rank]."""
        return F.softplus(self.n_raw)

    def signed_n(self) -> torch.Tensor:
        """
        Dale-signed presynaptic factors, shape [hidden_size, rank].

        This is the effective n to use in vector-field code.
        """
        return self.dale_sign[:, None] * self.n

    def recurrent_matrix(self) -> torch.Tensor:
        """
        Effective recurrent matrix W_rec, shape [post, pre].

        Columns have fixed sign:
            W[:, excitatory] >= 0
            W[:, inhibitory] <= 0
        """
        return self.m @ self.signed_n().T / self.hidden_size

    def reset_parameters(
        self,
        rho: float = 0.5,
        input_scale: float = 0.2,
        readout_scale: float = 0.5,
        train_inputs: bool = True,
    ) -> None:
        """
        Initialize parameters.

        rho is a factor scale, not an exact spectral radius/eigenvalue under
        Dale positivity constraints.
        """
        with torch.no_grad():
            if self.wi is not None and train_inputs:
                nn.init.normal_(
                    self.wi.weight,
                    mean=0.0,
                    std=input_scale / np.sqrt(max(1, self.input_size)),
                )
                nn.init.zeros_(self.wi.bias)

            # Initialize softplus(raw) around rho/sqrt(rank), keeping recurrent
            # drive moderate as rank increases.
            target = max(float(rho) / np.sqrt(max(1, self.rank)), 1e-4)
            raw_mean = self._inv_softplus(target)

            self.m_raw.normal_(mean=raw_mean, std=0.2)
            self.n_raw.normal_(mean=raw_mean, std=0.2)

            if self.wo is not None:
                nn.init.normal_(
                    self.wo.weight,
                    mean=0.0,
                    std=readout_scale / np.sqrt(self.hidden_size),
                )
                nn.init.zeros_(self.wo.bias)

    def get_readout(
        self,
        rates: torch.Tensor,
        rec_inputs: torch.Tensor,
        is_internal: bool = False,
    ) -> torch.Tensor:
        if self.wo is not None:
            return self.wo(rates)

        if is_internal:
            # Ostojic-style coordinate along final m column.
            return rec_inputs @ self.m / (torch.norm(m_last) ** 2 + 1e-12)

        # Vreeswijk-style readout from final signed n column.
        return rates @ self.signed_n() / rates.shape[-1]

    def update_dynamics(
        self,
        ff_inputs: torch.Tensor,
        rec_inputs: torch.Tensor,
        rates: torch.Tensor,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """
        One recurrent update.

        ff_inputs : [batch, input_size]
        rec_inputs: [batch, hidden_size]
        rates     : [batch, hidden_size]
        """
        hidden = (rates @ self.signed_n()) @ self.m.T / rates.shape[-1]

        rec_inputs = self.exp_alpha * rec_inputs + (1.0 - self.exp_alpha) * hidden

        if self.wi is None:
            drive = rec_inputs
        else:
            drive = self.wi(ff_inputs) + rec_inputs

        rates = self.nonlinearity(drive)

        return rates, rec_inputs

    @staticmethod
    def _reward_readout_scalar(readout: torch.Tensor) -> torch.Tensor:
        """
        Return scalar decision readout per batch for online reward feedback.

        Handles both:
            readout shape [batch]
            readout shape [batch, output_size]
        """
        if readout.ndim == 1:
            return readout
        return readout[..., -1]

    def forward(
        self,
        ff_inputs: torch.Tensor,
        targets: torch.Tensor | None = None,
        ret_rates: bool = False,
    ):
        """
        Run sequence.

        Parameters
        ----------
        ff_inputs : torch.Tensor
            Shape [batch, time, input_size].
        targets : torch.Tensor or None
            If supplied, online reward feedback is delivered through the final
            input channel on the next timestep when target==1 and readout>0.5.
        ret_rates : bool
            If True, return (readouts, rates, rec_inputs).

        Returns
        -------
        readout_list : torch.Tensor
            Shape [batch, time] for scalar readout, or [batch, time, output_size].
        """
        device = next(self.parameters()).device
        dtype = ff_inputs.dtype

        ff_inputs = ff_inputs.to(device=device, dtype=dtype)
        if targets is not None:
            targets = targets.to(device=device)

        batch_size, T, input_size = ff_inputs.shape

        rec_inputs = self.init_state_scale * torch.randn(
            batch_size,
            self.hidden_size,
            device=device,
            dtype=dtype,
        )
        rates = self.nonlinearity(rec_inputs)

        # Match LowRankModel: one random-input warmup pass to a typical state.
        if self.warmup_noise:
            for _ in range(T):
                rand_inputs = torch.randn(
                    batch_size,
                    input_size,
                    device=device,
                    dtype=dtype,
                )
                rates, rec_inputs = self.update_dynamics(
                    rand_inputs,
                    rec_inputs,
                    rates,
                )

        readout_list = []
        rates_list = []
        inputs_list = []

        # Online reward delivered through final input channel at next timestep.
        rwd_next = torch.zeros(batch_size, device=device, dtype=dtype)

        for step in range(T):
            x_t = ff_inputs[:, step].clone()

            if targets is not None:
                x_t[:, -1] = x_t[:, -1] + rwd_next

            rates, rec_inputs = self.update_dynamics(x_t, rec_inputs, rates)
            readout = self.get_readout(rates, rec_inputs)

            rates_list.append(rates)
            inputs_list.append(rec_inputs)
            readout_list.append(readout)

            if targets is not None:
                decision = self._reward_readout_scalar(readout)
                rwd_mask = (targets[:, step] == 1) & (decision > 0.5)
                rwd_next = rwd_mask.to(dtype=dtype)
            else:
                rwd_next.zero_()

        rates_list = torch.stack(rates_list, dim=1)
        inputs_list = torch.stack(inputs_list, dim=1)
        readout_list = torch.stack(readout_list, dim=1)

        if ret_rates:
            return readout_list, rates_list, inputs_list

        return readout_list

def plot_dual_kappa_targets(
    kappa,
    targets,
    condition_names,
    x_time,
    dim: int = 0,
    kappa_label: str = r"\kappa",
    T_STIM_ON=None,
    T_STIM_OFF=None,
    width: float = 6,
    height: float | None = None,
    condition_colors=("tab:blue", "tab:orange"),
    condition_styles=("-", "--"),
    qlim: float = 99,
):
    """
    Plot dual-task condition-averaged latent coordinate and targets.

    Layout:
        columns = DPA-only, Go, NoGo
        rows    = paired (AC, BD), unpaired (AD, BC)

    Parameters
    ----------
    kappa : array-like
        Shape [n_trials, time, rank]. Can be NumPy or torch.Tensor.
    targets : array-like
        Shape [n_trials, time, 2].
        targets[..., 0] = DPA target
        targets[..., 1] = GNG target
    condition_names : array-like of str
        Names like "A_C", "B_go_D", "A_nogo_C".
    x_time : array-like
        Time axis, shape [time].
    dim : int
        Which latent dimension to plot. dim=0 plots κ_1.
    kappa_label : str
        Label base, e.g. r"\\kappa", r"\\kappa^{rec}".
    """
    if height is None:
        golden_ratio = (5**0.5 - 1) / 2
        height = width * golden_ratio

    if torch.is_tensor(kappa):
        kappa_np = kappa.detach().cpu().numpy()
    else:
        kappa_np = np.asarray(kappa)

    if torch.is_tensor(targets):
        targets_np = targets.detach().cpu().numpy()
    else:
        targets_np = np.asarray(targets)

    condition_names = np.asarray(condition_names).astype(str)

    if kappa_np.ndim != 3:
        raise ValueError(f"kappa must have shape [trials, time, rank], got {kappa_np.shape}")
    if dim < 0 or dim >= kappa_np.shape[-1]:
        raise ValueError(f"dim={dim} outside available rank={kappa_np.shape[-1]}")

    def concrete_conditions(pair_status: str, gng_status: str) -> list[str]:
        if pair_status == "pair":
            base = [("A", "C"), ("B", "D")]
        elif pair_status == "unpair":
            base = [("A", "D"), ("B", "C")]
        else:
            raise ValueError(pair_status)

        if gng_status == "none":
            return [f"{sample}_{test}" for sample, test in base]
        return [f"{sample}_{gng_status}_{test}" for sample, test in base]

    row_specs = {
        "Pair\nAC, BD": "pair",
        "Unpair\nAD, BC": "unpair",
    }

    col_specs = {
        "DPA only": "none",
        "Go": "go",
        "NoGo": "nogo",
    }

    fig, ax = plt.subplots(
        2,
        3,
        figsize=(3 * width, 2 * height),
        sharex=True,
        sharey=True,
    )

    twin_axes = []

    for row, (row_title, pair_status) in enumerate(row_specs.items()):
        for col, (col_title, gng_status) in enumerate(col_specs.items()):
            this_ax = ax[row, col]
            kax = this_ax.twinx()
            twin_axes.append(kax)

            conds = concrete_conditions(pair_status, gng_status)

            for cond_i, cond_name in enumerate(conds):
                idxs = np.where(condition_names == cond_name)[0]

                if len(idxs) == 0:
                    this_ax.text(
                        0.5,
                        0.5 - 0.12 * cond_i,
                        f"Missing {cond_name}",
                        ha="center",
                        va="center",
                        transform=this_ax.transAxes,
                        color="crimson",
                    )
                    continue

                k_mean = kappa_np[idxs, :, dim].mean(axis=0)
                tgt_mean = targets_np[idxs].mean(axis=0)

                ls = condition_styles[cond_i]
                color = condition_colors[cond_i]

                # Targets on left axis.
                this_ax.plot(
                    x_time,
                    tgt_mean,
                    color="k",
                    ls=ls,
                    lw=2.4,
                    alpha=0.85,
                )

                # Latent coordinate on right axis.
                kax.plot(
                    x_time,
                    k_mean,
                    color=color,
                    ls=ls,
                    lw=3.0,
                    alpha=0.9,
                )

            if T_STIM_ON is not None and T_STIM_OFF is not None:
                for t_on, t_off in zip(T_STIM_ON, T_STIM_OFF):
                    this_ax.axvspan(
                        float(t_on),
                        float(t_off),
                        alpha=0.12,
                        color="tab:blue",
                        lw=0,
                    )

            this_ax.set_ylim([-1.5, 1.5])
            this_ax.set_title(f"{col_title}")

            if col == 0:
                this_ax.set_ylabel(row_title + "\nTargets")
            if col == 2:
                kax.set_ylabel(rf"${kappa_label}_{dim + 1}$")
            else:
                kax.set_yticklabels([])

            if row == 1:
                this_ax.set_xlabel("Time (s)")

    # Common latent y-limits.
    k_vals = kappa_np[:, :, dim].reshape(-1)
    k_lim = np.nanpercentile(np.abs(k_vals), qlim)
    k_lim = 1.1 * max(float(k_lim), 1e-6)

    for kax in twin_axes:
        kax.set_ylim([-k_lim, k_lim])

    legend_handles = [
        Line2D([0], [0], color="k", lw=2.4, label="DPA target"),
        Line2D([0], [0], color="crimson", lw=2.4, label="GNG target"),
        Line2D([0], [0], color=condition_colors[0], lw=3.0, label=rf"${kappa_label}_{dim + 1}$ condition 1"),
