* Low Rank RNNs

The Vanilla RNN updates its hidden state based on the previous hidden state and the current input:

Let:
- \( x_t \): feedforward input at time step \( t \)
- \( h_t \): recurrent input at time \( t \)
- \( r_t \): rate at time \( t \)
- \( y_t \): output at time \( t \)

\[h_t = (1-\alpha)h_{t-1} + \alpha \frac{1}{N} m n^\top r_{t-1}.\]

\[
r_t = \tanh(W_{ih} x_t + b_{ih} + h_{t-1})
\]

\[
y_t = \sigma(W_{ho} r_t + b_o)
\]

def init_low_rank_rnn(
    model,
    eigval: float = 0.3,
    input_scale: float = 0.2,
    readout_scale: float = 0.5,
):
    """
    Initialize rank-1 or low-rank RNN with controlled recurrent eigenvalues.

    For rank 1:
        J = n m.T / N
        lambda = m.T n / N ≈ eigval

    Assumes model update uses:
        hidden = (rates @ n) @ m.T / N
    """
    hidden_size, rank = model.m.shape
    device = model.m.device

    with torch.no_grad():
        # Input weights.
        nn.init.normal_(
            model.wi.weight,
            mean=0.0,
            std=input_scale / np.sqrt(model.wi.in_features),
        )
        nn.init.zeros_(model.wi.bias)

        # Low-rank factors.
        m = torch.randn(hidden_size, rank, device=device)
        q = torch.randn(hidden_size, rank, device=device)

        # Normalize columns to sqrt(N).
        m = m / m.norm(dim=0, keepdim=True) * np.sqrt(hidden_size)

        # Orthogonalize q against corresponding m columns.
        q = q - m * (q * m).sum(dim=0, keepdim=True) / hidden_size
        q = q / q.norm(dim=0, keepdim=True) * np.sqrt(hidden_size)

        # Choose n so that m.T @ n / N ≈ eigval on the diagonal.
        # The orthogonal component keeps n norm roughly sqrt(N).
        noise_amp = np.sqrt(max(0.0, 1.0 - eigval**2))
        n = eigval * m + noise_amp * q

        model.m.copy_(m)
        model.n.copy_(n)

        if model.wo is not None:
            nn.init.normal_(
                model.wo.weight,
                mean=0.0,
                std=readout_scale / np.sqrt(hidden_size),
            )
            nn.init.zeros_(model.wo.bias)

class LowRankModel(nn.Module):
    def __init__(self, input_size, hidden_size, output_size, rank=1, alpha=0.01, rho=0.5, device='cpu'):
        super().__init__()

        self.device = torch.device(device)
        self.input_size = input_size
        self.wi = nn.Linear(input_size, hidden_size, device=self.device)

        self.m = nn.Parameter(torch.randn(hidden_size, rank, device=self.device))
        self.n = nn.Parameter(torch.randn(hidden_size, rank, device=self.device))

        if output_size > 0:
            print('linear readout')
            self.wo = nn.Linear(hidden_size, output_size, device=self.device)
        else:
            self.wo = None

        self.nonlinearity = torch.tanh
        self.alpha = torch.tensor(alpha, device=self.device)
        self.exp_alpha = torch.exp(-self.alpha)

        # with torch.no_grad():
        #     self.m.copy_(self.n)

        init_low_rank_rnn(self, rho)

    def get_readout(self, rates, rec_inputs, is_internal=False):
        if self.wo is not None:
            return self.wo(rates).squeeze(-1)
        if is_internal: # ostojic
            return rec_inputs @ self.m / torch.norm(self.m[:, -1])
        # vreeswijk
        return rates @ self.n / rates.shape[-1]


    def update_dynamics(self, ff_inputs, rec_inputs, rates):
        # updating the recurrent inputs
        hidden = (rates @ self.n) @ self.m.T / rates.shape[-1]
        # hidden = rates @ (self.m @ self.n.T) / rates.shape[-1]
        # rec_inputs = (1.0 - self.alpha) * rec_inputs + self.alpha * hidden
        rec_inputs = self.exp_alpha * rec_inputs + (1.0 - self.exp_alpha) * hidden
        # updating the rates
        rates = self.nonlinearity(self.wi(ff_inputs) + rec_inputs)

        return rates, rec_inputs


    def forward(self, ff_inputs, ret_rates=False):
        # rates and rec_inputs are (n_trials, hidden_size)
        batch_size = ff_inputs.shape[0]
        hidden_size = self.m.shape[0]

        # rec_inputs = torch.randn(batch_size, hidden_size, device=self.device)
        rec_inputs = torch.randn(batch_size, hidden_size, device=self.device)
        rates = self.nonlinearity(rec_inputs)
        #rates = self.nonlinearity(self.wi(ff_inputs[:, -1]) + rec_inputs)

        step = 0
        while step < ff_inputs.shape[1]:
            rand_inputs = torch.randn(batch_size, self.input_size, device=self.device)
            rates, rec_inputs = self.update_dynamics(rand_inputs, rec_inputs, rates)
            step = step + 1

        readout_list = []
        rates_list = []
        inputs_list = []

        for step in range(ff_inputs.shape[1]):
            rates, rec_inputs = self.update_dynamics(ff_inputs[:, step], rec_inputs, rates)
            readout = self.get_readout(rates, rec_inputs)

            rates_list.append(rates)
            inputs_list.append(rec_inputs)
            readout_list.append(readout)

        rates_list = torch.stack(rates_list, dim=1)
        inputs_list = torch.stack(inputs_list, dim=1)
        readout_list = torch.stack(readout_list, dim=1)


        if ret_rates:
            return readout_list, rates_list, inputs_list

        return readout_list

** In the loss I use the recurrent readout. Should I use the internal readout instead

#+begin_reasoning

#+end_reasoning



/Short answer:/ usually *no*—for the loss, use the readout that corresponds to the task output you want the network to produce. Use the “internal” readout mainly as a diagnostic / latent-coordinate probe, unless your hypothesis is specifically that the target should be represented directly in the recurrent current \(h_t\).

In your code, with =wo is None=:

#+begin_src python
# default currently used in forward()
return rates @ self.n[:, -1] / rates.shape[-1]
#+end_src

#+RESULTS:

This is the low-rank coordinate-like readout

\[
\kappa_r(t) = \frac{1}{N} r_t^\top n_r,
\]

whereas

#+begin_src python
return rec_inputs @ self.m[:, -1] / torch.norm(self.m[:, -1])
#+end_src

#+RESULTS:

reads out the recurrent input/current along \(m_r\).

*Interpretation*

| readout | code | what it measures | best use |
|---|---|---|---|
| rate / overlap readout | =rates @ n[:, -1] / N= | instantaneous low-rank drive \(\kappa_t\) generated by rates | task loss if you want output tied to the low-rank coordinate |
| internal readout | =rec_inputs @ m[:, -1] / ||m[:, -1]||= | projection of recurrent current \(h_t\) onto \(m_r\) | diagnostics, phase portraits, latent state analysis |
| trainable output | =wo(rates)= | arbitrary linear decoding from population rates | best default for behavioral output |

For your model,

\[
h_{t+1}
= e^{-\alpha} h_t + (1-e^{-\alpha}) M \kappa_t,
\qquad
\kappa_t = \frac{1}{N} r_t^\top n.
\]

So the internal projection is a /filtered recurrent state/, while =rates @ n / N= is the instantaneous low-rank drive. They are related, but not identical. In rank 1, if \(\|m\|=\sqrt{N}\),

\[
\frac{h_t^\top m}{\|m\|}
\approx \sqrt{N}\,\bar{\kappa}_t,
\]

where \(\bar{\kappa}_t\) is the low-pass filtered coordinate. So the internal readout has a different scale and temporal lag. Using it in an MSE loss can unintentionally train the norm/scale of \(m\), not just the task dynamics.

*Recommendation*

For a task-output loss, I would use one of these:

1. *Best general choice:* use a trainable readout

   #+begin_src python
   output_size = 1
   readout = self.wo(rates).squeeze(-1)
   #+end_src

   #+RESULTS:

   This separates “solve the task” from “where is the latent stored”.

2. *If you want the task output to be the low-rank coordinate:* keep your current default

   #+begin_src python
   readout = rates @ self.n[:, -1] / rates.shape[-1]
   #+end_src

   #+RESULTS:

   This is most consistent with the low-rank mean-field coordinate \(\kappa\).

3. *Use internal readout only if the target is meant to supervise the recurrent current itself*, e.g. you want \(h_t\) projected along \(m_r\) to behave like the decision variable.

If you do use the internal readout, I would normalize it more consistently:

#+begin_src python
# rank-1 approximate κ from recurrent current h ≈ m κ
readout = rec_inputs @ self.m[:, -1] / torch.sum(self.m[:, -1] ** 2)
#+end_src

#+RESULTS:

because if \(h = m\kappa\), then

\[
\hat{\kappa}
= \frac{h^\top m}{m^\top m}.
\]

Your current version divides by \(\|m\|\), giving \(\sqrt{N}\kappa\) when \(\|m\|=\sqrt{N}\).

So: *for the loss, keep the recurrent/rate overlap readout or use =wo=; do not switch to the internal readout unless you explicitly want to supervise the latent recurrent current.*

def train_val_split(X, y, batch_size=32, frac=0.8):
    # 1. Wrap tensors into a dataset
    dataset = TensorDataset(X.float(), y.float())

    # 2. Train/validation split
    n_total = len(dataset)
    n_train = int(frac * n_total)
    n_val = n_total - n_train

    generator = torch.Generator().manual_seed(0)
    train_dataset, val_dataset = random_split(dataset, lengths=[n_train, n_val], generator=generator)

    # 3. Create DataLoaders
    train_loader = DataLoader(train_dataset, batch_size, shuffle=True)
    val_loader = DataLoader(val_dataset, batch_size, shuffle=False)

    return train_loader, val_loader

from typing import Any

import torch
from torch.utils.data import DataLoader


class Optimization:
    """Small PyTorch train/validation loop."""

    def __init__(
        self,
        model: torch.nn.Module,
        train_loader: DataLoader,
        val_loader: DataLoader,
        criterion: torch.nn.Module,
        optimizer: torch.optim.Optimizer,
        scheduler: Any | None = None,
        grad_clip_norm: float | None = None,
        num_epochs: int = 100,
        stop_loss: float = 0.005,
        max_val_loss: float = 100.0,
        zero_low_rank_grad: Any | None = None,
        device: str | torch.device | None = None,
        verbose: bool = True,
    ):
        self.model = model
        self.train_loader = train_loader
        self.val_loader = val_loader
        self.criterion = criterion
        self.optimizer = optimizer
        self.num_epochs = num_epochs
        self.stop_loss = stop_loss
        self.max_val_loss = max_val_loss
        self.scheduler = scheduler
        self.zero_low_rank_grad = zero_low_rank_grad
        self.grad_clip_norm = grad_clip_norm
        self.verbose = verbose

        self.device = torch.device(device) if device is not None else self._infer_device()
        self.model.to(self.device)
        self.model.device = self.device  # useful for models expecting `self.device`

        self.train_losses: list[float] = []
        self.val_losses: list[float] = []
        self.learning_rates: list[float] = []
        self.learning_rate = 1.0

    def _infer_device(self) -> torch.device:
        if hasattr(self.model, "device"):
            return torch.device(self.model.device)
        return next(self.model.parameters()).device

    def _predict(self, X: torch.Tensor) -> torch.Tensor:
        output = self.model(X)
        return self.model.readout if hasattr(self.model, "readout") else output

    def _zero_low_rank_grads(self) -> None:
        if self.zero_low_rank_grad is None:
            return

        for param in (self.model.m, self.model.n):
            if param.grad is None:
                continue

            if self.zero_low_rank_grad == "all":
                param.grad.zero_()
            else:
                param.grad[:, self.zero_low_rank_grad] = 0

    def _run_epoch(self, loader: DataLoader, train: bool) -> float:
        self.model.train(train)

        total_loss = 0.0
        total_examples = 0
        context = torch.enable_grad() if train else torch.no_grad()

        with context:
            for X, y in loader:
                X = X.to(self.device)
                y = y.to(self.device)
                batch_size = X.size(0)

                if train:
                    self.optimizer.zero_grad(set_to_none=True)

                y_pred = self._predict(X)

                if y.dim()==3:
                    loss = 0
                    for i in range(y.shape[-1]):
                        loss += self.criterion(y_pred, y[..., i])
                else:
                    loss = self.criterion(y_pred, y)

                if not torch.isfinite(loss):
                    return float("nan")

                if train:
                    loss.backward()
                    self._zero_low_rank_grads()

                    if self.grad_clip_norm is not None:
                        torch.nn.utils.clip_grad_norm_(
                            self.model.parameters(),
                            self.grad_clip_norm,
                        )

                    self.optimizer.step()

                total_loss += loss.item() * batch_size
                total_examples += batch_size

        if total_examples == 0:
            raise ValueError("Dataloader produced zero examples.")

        return total_loss / total_examples

    def _step_scheduler(self, val_loss: float) -> None:
        if self.scheduler is None:
            self.learning_rate = learning_rate
            return

        if isinstance(self.scheduler, torch.optim.lr_scheduler.ReduceLROnPlateau):
            self.scheduler.step(val_loss)
        else:
            self.scheduler.step()

        self.learning_rate = self.scheduler.get_last_lr()[0]

    def fit(self) -> tuple[list[float], list[float]]:
        self.train_losses.clear()
        self.val_losses.clear()

        for epoch in range(1, self.num_epochs + 1):
            train_loss = self._run_epoch(self.train_loader, train=True)
            val_loss = self._run_epoch(self.val_loader, train=False)

            self._step_scheduler(val_loss)

            self.train_losses.append(train_loss)
            self.val_losses.append(val_loss)
            self.learning_rates.append(self.learning_rate)

            if self.verbose:
                print(
                    f"Epoch {epoch:03d}/{self.num_epochs} | "
                    f"lr: {self.learning_rate:.4f} |"
                    f"train: {train_loss:.4f} | val: {val_loss:.4f}"
                )

            if not math.isfinite(train_loss) or not math.isfinite(val_loss):
                print("Stopping: non-finite loss.")
                break

            if train_loss < self.stop_loss and val_loss < self.stop_loss:
                print(f"Stopping: losses below {self.stop_loss}.")
                break

            if val_loss > self.max_val_loss:
                print(f"Stopping: validation loss above {self.max_val_loss}.")
                break

        return self.train_losses, self.val_losses, self.learning_rates

import torch
import torch.nn as nn

class MaskedZeroLoss(nn.Module):
    """
    Loss for sequence targets where target==0 means the network should stay silent.

    For y != 0:
        penalize output against target.

    For y == 0:
        penalize output against zero.

    Assumes:
    - y_pred.shape == y.shape, e.g. [batch, time]
    - time is dim 1
    """

    def __init__(
        self,
        criterion: nn.Module | None = None,
        nonzero_weight: float = 1.0,
        zero_weight: float = 1.0,
        zero_tol: float = 0.0,
    ):
        super().__init__()
        self.criterion = criterion or nn.MSELoss()
        self.nonzero_weight = nonzero_weight
        self.zero_weight = zero_weight
        self.zero_tol = zero_tol

    def forward(self, y_pred: torch.Tensor, y: torch.Tensor) -> torch.Tensor:

        valid_mask = torch.isfinite(y_pred[..., -1]) & torch.isfinite(y) & (y!=0.5)
        bl_mask = y==0.5

        nonzero_mask = valid_mask & (y.abs() > self.zero_tol)
        zero_mask = valid_mask & (y.abs() <= self.zero_tol)

        loss_bl = 0
        for i in range(y_pred.shape[-1]):
            loss_bl = loss_bl + self.criterion(y_pred[..., i][bl_mask], torch.zeros_like(y[bl_mask]))

        # loss_bl = self.criterion(y_pred[..., -1][bl_mask], torch.zeros_like(y[bl_mask]))
        loss_nonzero = self.criterion(y_pred[..., -1][nonzero_mask], y[nonzero_mask])
        loss_zero = self.criterion(y_pred[..., -1][zero_mask], torch.zeros_like(y[zero_mask]))

        return 0.1 * loss_bl + self.nonzero_weight * loss_nonzero + self.zero_weight * loss_zero

* Vector fields
** Vector field

The low-rank recurrent state lives approximately on the manifold

\[
h = M \kappa,\qquad M = [m_1,m_2]\in \mathbb{R}^{N\times 2}.
\]

For a fixed feedforward input \(x\), the induced rank-coordinate dynamics are

\[
\kappa^{+}
= (1-\alpha)\kappa + \alpha \Psi(\kappa; x),
\]

where

\[
\Psi(\kappa; x)
= \frac{1}{N} n^\top \tanh(W_i x + b_i + M\kappa).
\]

Thus the phase portrait in the \((\kappa_1,\kappa_2)\) plane is given by the vector field

\[
\Delta \kappa
= \kappa^{+}-\kappa

def make_input(input_size, active_dims=None, value=1.0, device=None, dtype=None):

def low_rank_numpy_params(model):
    """
    Extract trained low-rank RNN parameters as NumPy arrays.

    Coordinate convention:
        h = M κ

    with M = [m_1, ..., m_R] ∈ R^{N × R}. For a fixed input x,

        Ψ(κ; x) = N^{-1} n^T tanh(W_i x + b_i + Mκ)

    and the induced discrete κ-map is

        κ⁺ = (1 - α)κ + α Ψ(κ; x).
    """
    return {
        "M": model.m.detach().cpu().numpy().astype(np.float64),          # (N, rank)
        "Nvec": model.n.detach().cpu().numpy().astype(np.float64),       # (N, rank)
        "Wi": model.wi.weight.detach().cpu().numpy().astype(np.float64), # (N, input_size)
        "bi": model.wi.bias.detach().cpu().numpy().astype(np.float64),   # (N,)
        "alpha": float(model.alpha),
    }


def low_rank_field_np(params, kappa, ff_input=None, include_alpha=False):
    """
    Low-rank vector field in κ-coordinates.

    Returns

        F(κ; x) = Ψ(κ; x) - κ

    unless include_alpha=True, in which case it returns αF.

    Fixed points satisfy F(κ; x)=0.
    """
    M = params["M"]        # (N, rank)
    Nvec = params["Nvec"]  # (N, rank)
    Wi = params["Wi"]      # (N, input_size)
    bi = params["bi"]      # (N,)

    kappa = np.asarray(kappa, dtype=np.float64)
    orig_shape = kappa.shape
    rank = M.shape[1]

    kappa_flat = kappa.reshape(-1, rank)  # (B, rank)

    if ff_input is None:
        ff_input = np.zeros(Wi.shape[1], dtype=np.float64)
    else:
        ff_input = np.asarray(ff_input, dtype=np.float64)

    drive = ff_input @ Wi.T + bi                  # (N,)
    rec_inputs = kappa_flat @ M.T                 # (B, N)
    rates = np.tanh(rec_inputs + drive[None, :])  # (B, N)

    psi = rates @ Nvec / M.shape[0]               # (B, rank)
    field = psi - kappa_flat                      # (B, rank)

    if include_alpha:
        field = params["alpha"] * field

    return field.reshape(orig_shape)


def low_rank_jacobian_flow_np(params, kappa, ff_input=None):
    """
    Jacobian of F(κ; x) = Ψ(κ; x) - κ.

    With

        u_i = m_i · κ + w_i · x + b_i,

    we have

        J_ab = N^{-1} Σ_i n_{ia} φ'(u_i) m_{ib} - δ_ab.
    """
    M = params["M"]
    Nvec = params["Nvec"]
    Wi = params["Wi"]
    bi = params["bi"]

    kappa = np.asarray(kappa, dtype=np.float64).reshape(-1)

    if ff_input is None:
        ff_input = np.zeros(Wi.shape[1], dtype=np.float64)
    else:
        ff_input = np.asarray(ff_input, dtype=np.float64)

    u = M @ kappa + ff_input @ Wi.T + bi
    phi_prime = 1.0 - np.tanh(u) ** 2

    J = Nvec.T @ (phi_prime[:, None] * M) / M.shape[0]
    J -= np.eye(M.shape[1])

    return J


def low_rank_jacobian_map_np(params, kappa, ff_input=None):
    """
    Jacobian of the discrete κ-map:

        κ⁺ = κ + αF(κ).

    Discrete-time stability is determined by eigvals of this matrix.
    """
    J_flow = low_rank_jacobian_flow_np(params, kappa, ff_input=ff_input)
    return np.eye(J_flow.shape[0]) + params["alpha"] * J_flow


def project_rec_inputs_to_kappa(model, rec_inputs):
    """
    Least-squares projection of recurrent inputs h onto the low-rank basis M.

        h ≈ Mκ.

    Parameters
    ----------
    rec_inputs : torch.Tensor
        Shape (..., hidden_size).

    Returns
    -------
    kappa : torch.Tensor
        Shape (..., rank).
    """
    device = next(model.parameters()).device
    dtype = model.m.dtype

    h = torch.as_tensor(rec_inputs, device=device, dtype=dtype)
    orig_shape = h.shape

    hidden_size = model.m.shape[0]
    rank = model.m.shape[1]
    assert orig_shape[-1] == hidden_size

    h_flat = h.reshape(-1, hidden_size)  # (B, N)

    with torch.no_grad():
        # Solve M κ_b ≈ h_b for each batch element b.
        sol = torch.linalg.lstsq(model.m, h_flat.T).solution  # (rank, B)
        kappa = sol.T.reshape(*orig_shape[:-1], rank)

    return kappa


def condition_indices_from_inputs(inputs):
    """
    Infer DPA condition labels from stimulus channels.

    Assumes:
        channel 0 = A
        channel 1 = B
        channel 2 = C
        channel 3 = D
    """
    sample_epoch = slice(int(N_STIM_ON[0]), int(N_STIM_OFF[0]))
    test_epoch = slice(int(N_STIM_ON[1]), int(N_STIM_OFF[1]))

    sample_drive = inputs[:, sample_epoch, :2].mean(dim=1)
    test_drive = inputs[:, test_epoch, 2:4].mean(dim=1)

    is_A = sample_drive[:, 0] > sample_drive[:, 1]
    is_B = ~is_A

    is_C = test_drive[:, 0] > test_drive[:, 1]
    is_D = ~is_C

    return {
        "AC": torch.where(is_A & is_C)[0],
        "AD": torch.where(is_A & is_D)[0],
        "BC": torch.where(is_B & is_C)[0],
        "BD": torch.where(is_B & is_D)[0],
    }


def auto_limits_from_trajs(kappa_traj, qlo=1, qhi=99, pad=1.35, symmetric=True):
    """
    Infer axis limits from κ-trajectories.
    """
    all_kappa = kappa_traj.reshape(-1, kappa_traj.shape[-1])

    lo = np.percentile(all_kappa, qlo, axis=0)
    hi = np.percentile(all_kappa, qhi, axis=0)

    if symmetric:
        zlim = float(pad * np.max(np.abs(np.concatenate([lo, hi]))))
        return (-zlim, zlim), (-zlim, zlim)

    center = 0.5 * (lo + hi)
    half_width = 0.5 * (hi - lo) * pad

    xlim = (float(center[0] - half_width[0]), float(center[0] + half_width[0]))
    ylim = (float(center[1] - half_width[1]), float(center[1] + half_width[1]))

    return xlim, ylim


def make_vector_field_grid(
    model,
    ff_input=None,
    xlim=None,
    ylim=None,
    kappa_traj=None,
    n_grid=251,
    qlo=1,
    qhi=99,
    pad=1.35,
    symmetric_auto=True,
    include_alpha=False,
):
    """
    Build vector-field grid.

    If xlim/ylim are None, limits are inferred from kappa_traj.
    """
    params = low_rank_numpy_params(model)

    if ff_input is None:
        ff_input_np = np.zeros(params["Wi"].shape[1], dtype=np.float64)
    else:
        ff_input_np = torch.as_tensor(ff_input).detach().cpu().numpy().astype(np.float64)

    if xlim is None or ylim is None:
        if kappa_traj is None:
            raise ValueError("Provide xlim/ylim or kappa_traj for auto limits.")

        auto_xlim, auto_ylim = auto_limits_from_trajs(
            kappa_traj,
            qlo=qlo,
            qhi=qhi,
            pad=pad,
            symmetric=symmetric_auto,
        )

        if xlim is None:
            xlim = auto_xlim
        if ylim is None:
            ylim = auto_ylim

    k1 = np.linspace(xlim[0], xlim[1], n_grid)
    k2 = np.linspace(ylim[0], ylim[1], n_grid)

    K1, K2 = np.meshgrid(k1, k2)
    K = np.stack([K1, K2], axis=-1)  # (n_grid, n_grid, 2)

    field = low_rank_field_np(
        params,
        K,
        ff_input=ff_input_np,
        include_alpha=include_alpha,
    )

    U = field[..., 0]
    V = field[..., 1]
    speed = np.sqrt(U**2 + V**2)

    return K1, K2, U, V, speed, xlim, ylim, params, ff_input_np


def merge_roots(roots, residuals, merge_tol=5e-2):
    """
    Merge duplicate fixed points found from different initial seeds.
    """
    if len(roots) == 0:
        return np.empty((0, 2)), np.empty((0,))

    roots = np.asarray(roots, dtype=np.float64)
    residuals = np.asarray(residuals, dtype=np.float64)

    order = np.argsort(residuals)
    kept = []
    kept_residuals = []

    for idx in order:
        r = roots[idx]
        if all(np.linalg.norm(r - q) > merge_tol for q in kept):
            kept.append(r)
            kept_residuals.append(residuals[idx])

    return np.stack(kept, axis=0), np.asarray(kept_residuals)

def default_arrow_times(T, n_arrows=8, arrow_step=3):
    """
    Choose time indices for trajectory arrows.

    Arrows point from κ_t to κ_{t+arrow_step}.
    """
    if T <= arrow_step + 1:
        return np.array([], dtype=int)

    lo = 1
    hi = T - arrow_step - 1

    n_arrows = min(n_arrows, max(1, hi - lo + 1))
    return np.unique(np.linspace(lo, hi, n_arrows).astype(int))


def add_time_arrows(
    ax,
    xy,
    times=None,
    color="k",
    arrow_step=3,
    mutation_scale=15,
    lw=1.8,
    alpha=1.0,
    zorder=8,
    min_length=1e-4,
):
    """
    Draw directional arrows along a 2D trajectory.

    Parameters
    ----------
    xy : array, shape (T, 2)
        Trajectory in the κ-plane.
    times : array-like or None
        Starting indices t for arrows from xy[t] to xy[t+arrow_step].
    """
    xy = np.asarray(xy, dtype=np.float64)
    T = xy.shape[0]

    if times is None:
        times = default_arrow_times(T, n_arrows=8, arrow_step=arrow_step)

    for t in times:
        t = int(t)
        if t < 0:
            t = T + t

        if t < 0 or t >= T - 1:
            continue

        j0 = t
        j1 = min(t + arrow_step, T - 1)

        # If the selected segment is too short, search forward.
        while j1 < T and np.linalg.norm(xy[j1] - xy[j0]) < min_length:
            j0 += 1
            j1 = min(j0 + arrow_step, T - 1)
            if j0 >= T - 1:
                break

        if j0 >= T - 1:
            continue

        p0 = xy[j0]
        p1 = xy[j1]

        if not (np.all(np.isfinite(p0)) and np.all(np.isfinite(p1))):
            continue

        if np.linalg.norm(p1 - p0) < min_length:
            continue

        ax.annotate(
            "",
            xy=p1,
            xytext=p0,
            arrowprops=dict(
                arrowstyle="-|>",
                color=color,
                lw=lw,
                alpha=alpha,
                shrinkA=0,
                shrinkB=0,
                mutation_scale=mutation_scale,
            ),
            zorder=zorder,
        )

Below is a drop-in /Dale-constrained E/I low-rank RNN/ compatible with your existing =Optimization=, vector-field, and trajectory utilities as much as possible.

Key convention: recurrent weight from presynaptic neuron \(j\) to postsynaptic neuron \(i\) is

\[
W_{ij} = \frac{1}{N}\sum_{r=1}^R m_{ir}\, n_{jr}\, s_j,
\]

where \(s_j=+1\) for excitatory neurons and \(s_j=-1\) for inhibitory neurons. Because =m= and =n= are constrained nonnegative, each presynaptic neuron has a fixed outgoing sign, satisfying Dale’s law.

import torch
import torch.nn as nn
import torch.nn.functional as F
import numpy as np


class EILowRankModel(nn.Module):
    """
    Dale-constrained excitatory/inhibitory low-rank RNN.

    Recurrent matrix convention
    ---------------------------
    The effective recurrent matrix is

        W_rec[i, j] = sum_r m_pos[i, r] * n_pos[j, r] * dale_sign[j] / N

    so the sign of all outgoing synapses from presynaptic neuron j is fixed by
    dale_sign[j]:

        +1 excitatory
        -1 inhibitory

    Dynamics
    --------
        h_t = decay * h_{t-1} + (1 - decay) * W_rec r_{t-1}
        r_t = tanh(W_in x_t + b_in + h_t)
        y_t = W_out r_t + b_out

    Shapes
    ------
    ff_inputs : [batch, time, input_size]
    rates     : [batch, hidden_size]
    rec_inputs: [batch, hidden_size]
    readout   : [batch, time] if output_size == 1
    """

    def __init__(
        self,
        input_size: int,
        hidden_size: int,
        output_size: int,
        rank: int = 1,
        alpha: float = 0.01,
        p_exc: float = 0.8,
        rho: float = 0.5,
        input_scale: float = 0.2,
        readout_scale: float = 0.5,
        init_state_scale: float = 1.0,
        warmup_noise: bool = True,
        device: str | torch.device = "cpu",
    ):
        super().__init__()

        self.device = torch.device(device)
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.output_size = output_size
        self.rank = rank
        self.p_exc = p_exc
        self.init_state_scale = init_state_scale
        self.warmup_noise = warmup_noise

        self.wi = nn.Linear(input_size, hidden_size, device=self.device)

        # Raw factors; exposed `m= and =n= properties below return positive factors.
        self.m_raw = nn.Parameter(torch.randn(hidden_size, rank, device=self.device))
        self.n_raw = nn.Parameter(torch.randn(hidden_size, rank, device=self.device))

        if output_size > 0:
            print("linear readout")
            self.wo = nn.Linear(hidden_size, output_size, device=self.device)
        else:
            self.wo = None

        self.nonlinearity = torch.tanh
        self.alpha = torch.tensor(alpha, device=self.device)
        self.exp_alpha = torch.exp(-self.alpha)

        n_exc = int(round(p_exc * hidden_size))
        n_inh = hidden_size - n_exc

        dale_sign = torch.cat(
            [
                torch.ones(n_exc, device=self.device),
                -torch.ones(n_inh, device=self.device),
            ],
            dim=0,
        )

        # Shuffle so E/I identity is not tied to index order.
        perm = torch.randperm(hidden_size, device=self.device)
        dale_sign = dale_sign[perm]

        self.register_buffer("dale_sign", dale_sign)
        self.register_buffer("is_exc", dale_sign > 0)
        self.register_buffer("is_inh", dale_sign < 0)

        self.reset_parameters(
            rho=rho,
            input_scale=input_scale,
            readout_scale=readout_scale,
        )

    @property
    def m(self) -> torch.Tensor:
        """Nonnegative postsynaptic low-rank factors, shape [N, rank]."""
        return F.softplus(self.m_raw)

    @property
    def n(self) -> torch.Tensor:
        """Nonnegative presynaptic low-rank factors, shape [N, rank]."""
        return F.softplus(self.n_raw)

    def signed_n(self) -> torch.Tensor:
        """
        Dale-signed presynaptic factors.

        Shape: [N, rank]
        """
        return self.dale_sign[:, None] * self.n

    def recurrent_matrix(self) -> torch.Tensor:
        """
        Effective recurrent matrix W with shape [N_post, N_pre].

        W[i, j] has sign fixed by dale_sign[j].
        """
        return self.m @ self.signed_n().T / self.hidden_size

    def reset_parameters(
        self,
        rho: float = 0.5,
        input_scale: float = 0.2,
        readout_scale: float = 0.5,
    ) -> None:
        """
        Initialize input, readout, and positive low-rank factors.

        =rho= roughly controls recurrent factor scale. Because Dale constraints and
        softplus positivity change the spectrum, treat rho as a scale parameter, not
        an exact eigenvalue.
        """
        with torch.no_grad():
            nn.init.normal_(
                self.wi.weight,
                mean=0.0,
                std=input_scale / np.sqrt(self.input_size),
            )
            nn.init.zeros_(self.wi.bias)

            # Initialize raw factors so softplus(raw) is O(rho).
            target = max(float(rho), 1e-4)
            raw_mean = np.log(np.exp(target) - 1.0)

            self.m_raw.normal_(mean=raw_mean, std=0.2)
            self.n_raw.normal_(mean=raw_mean, std=0.2)

            if self.wo is not None:
                nn.init.normal_(
                    self.wo.weight,
                    mean=0.0,
                    std=readout_scale / np.sqrt(self.hidden_size),
                )
                nn.init.zeros_(self.wo.bias)

    def get_readout(
        self,
        rates: torch.Tensor,
        rec_inputs: torch.Tensor,
        is_internal: bool = False,
    ) -> torch.Tensor:
        if self.wo is not None:
            return self.wo(rates).squeeze(-1)

        if is_internal:
            return rec_inputs @ self.m[:, -1] / torch.norm(self.m[:, -1])

        return rates @ self.signed_n()[:, -1] / rates.shape[-1]

    def update_dynamics(
        self,
        ff_inputs: torch.Tensor,
        rec_inputs: torch.Tensor,
        rates: torch.Tensor,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """
        One recurrent update.

        ff_inputs : [batch, input_size]
        rec_inputs: [batch, hidden_size]
        rates     : [batch, hidden_size]
        """
        hidden = (rates @ self.signed_n()) @ self.m.T / rates.shape[-1]

        rec_inputs = self.exp_alpha * rec_inputs + (1.0 - self.exp_alpha) * hidden
        rates = self.nonlinearity(self.wi(ff_inputs) + rec_inputs)

        return rates, rec_inputs

    def forward(
        self,
        ff_inputs: torch.Tensor,
        ret_rates: bool = False,
        init_rec_inputs: torch.Tensor | None = None,
    ):
        """
        Run the network over a sequence.

        ff_inputs : [batch, time, input_size]
        """
        batch_size, T, _ = ff_inputs.shape

        if init_rec_inputs is None:
            rec_inputs = self.init_state_scale * torch.randn(
                batch_size,
                self.hidden_size,
                device=self.device,
                dtype=ff_inputs.dtype,
            )
        else:
            rec_inputs = init_rec_inputs.to(self.device)

        rates = self.nonlinearity(rec_inputs)

        # Optional warmup using random inputs, matching your LowRankModel behavior.
        if self.warmup_noise:
            for _ in range(T):
                rand_inputs = torch.randn(
                    batch_size,
                    self.input_size,
                    device=self.device,
                    dtype=ff_inputs.dtype,
                )
                rates, rec_inputs = self.update_dynamics(rand_inputs, rec_inputs, rates)

        readout_list = []
        rates_list = []
        inputs_list = []

        for step in range(T):
            rates, rec_inputs = self.update_dynamics(
                ff_inputs[:, step],
                rec_inputs,
                rates,
            )
            readout = self.get_readout(rates, rec_inputs)

            rates_list.append(rates)
            inputs_list.append(rec_inputs)
            readout_list.append(readout)

        rates_list = torch.stack(rates_list, dim=1)
        inputs_list = torch.stack(inputs_list, dim=1)
        readout_list = torch.stack(readout_list, dim=1)

        if ret_rates:
            return readout_list, rates_list, inputs_list

        return readout_list
#+end_src

*** Here I use a exponential euler integration scheme. How should I scale the noise with DT or alpha

#+begin_reasoning

#+end_reasoning



/Short answer/: if your exponential Euler coefficient is

\[
a = e^{-\Delta t/\tau} = e^{-\alpha},
\qquad \alpha = \Delta t/\tau,
\]

then additive white process noise should scale like

\[
\sqrt{1-a^2}=\sqrt{1-e^{-2\alpha}},
\]

up to the convention-dependent noise amplitude.

For small \(\alpha\),

\[
\sqrt{1-e^{-2\alpha}} \approx \sqrt{2\alpha}
= \sqrt{2\Delta t/\tau}.
\]

So: /noise scales like \(\sqrt{\Delta t}\), not \(\Delta t\)./

*Mode: Explain*

If you want noisy recurrent inputs \(h_t\), a common continuous-time model is

\[
dh = \frac{-h + H(r,x)}{\tau}\,dt + \frac{\sigma}{\sqrt{\tau}}\,dW_t,
\]

where \(H(r,x)\) is your low-rank recurrent drive and \(\sigma\) controls diffusion strength. The exact discretization over one step, assuming \(H\) is constant during the step, is

\[
h_{t+\Delta t}
= e^{-\Delta t/\tau}h_t
+ \left(1-e^{-\Delta t/\tau}\right)H_t
+ \sigma \sqrt{\frac{1-e^{-2\Delta t/\tau}}{2}}\,\xi_t,
\]

with

\[
\xi_t \sim \mathcal N(0,I).
\]

In your notation:

\[
h_{t+1}
= e^{-\alpha}h_t
+ \left(1-e^{-\alpha}\right)H_t
+ \sigma \sqrt{\frac{1-e^{-2\alpha}}{2}}\,\xi_t.
\]

So your update would become something like:

#+begin_src jupyter-python
def update_dynamics(self, ff_inputs, rec_inputs, rates, noise_std=0.0):
    hidden = (rates @ self.n) @ self.m.T / rates.shape[-1]

    a = self.exp_alpha  # exp(-dt/tau)
    rec_inputs = a * rec_inputs + (1.0 - a) * hidden

    if noise_std > 0:
        noise_scale = noise_std * torch.sqrt((1.0 - a**2) / 2.0)
        rec_inputs = rec_inputs + noise_scale * torch.randn_like(rec_inputs)

    rates = self.nonlinearity(self.wi(ff_inputs) + rec_inputs)

    return rates, rec_inputs
#+end_src

If instead you want a discrete-time AR(1) process whose stationary variance is exactly \(\sigma^2\) when \(H=0\), use

\[
h_{t+1} = a h_t + \sigma \sqrt{1-a^2}\,\xi_t.
\]

Then:

#+begin_src jupyter-python
noise_scale = noise_std * torch.sqrt(1.0 - a**2)
#+end_src

The difference is only a convention for what =noise_std= means.

| Desired meaning of =noise_std= | Use scale |
|---|---|
| Diffusion amplitude in \(dh = \cdots + \sigma/\sqrt{\tau}\,dW_t\) | \(\sigma\sqrt{(1-e^{-2\alpha})/2}\) |
| Stationary standard deviation of OU noise around zero | \(\sigma\sqrt{1-e^{-2\alpha}}\) |
| Euler-Maruyama small-\(\alpha\) approximation | \(\sigma\sqrt{\alpha}\) or \(\sigma\sqrt{2\alpha}\), depending on convention |

For your code, I would probably use the /stationary-variance convention/ because it is easiest to tune:

#+begin_src jupyter-python
a = self.exp_alpha
rec_inputs = a * rec_inputs + (1.0 - a) * hidden
rec_inputs = rec_inputs + noise_std * torch.sqrt(1.0 - a**2) * torch.randn_like(rec_inputs)
#+end_src

Then =noise_std=0.1= means roughly “the noise alone would give recurrent inputs with stationary std \(0.1\).”

One more note: your current initialization

#+begin_src jupyter-python
rec_inputs = torch.randn(batch_size, hidden_size, device=self.device)
#+end_src

implicitly gives initial recurrent inputs with std \(1\), independent of \(\alpha\). If you want consistency with the same OU noise scale, initialize instead as

#+begin_src jupyter-python
rec_inputs = noise_std * torch.randn(batch_size, hidden_size, device=self.device)
#+end_src

or simply zero-initialize:

#+begin_src jupyter-python
rec_inputs = torch.zeros(batch_size, hidden_size, device=self.device)
#+end_src

depending on whether you want random initial conditions or noise-driven variability.

Abstract

Neural computations are currently investigated using two separate approaches: sorting neurons into functional subpopulations or
examining the low-dimensional dynamics of collective activity. Whether and how these two aspects interact to shape computations is
currently unclear. Using a novel approach to extract computational mechanisms from networks trained on neuroscience tasks, here we
show that the dimensionality of the dynamics and subpopulation structure play fundamentally complementary roles. Although various
tasks can be implemented by increasing the dimensionality in networks with fully random population structure, flexible input–output
mappings instead require a non-random population structure that can be described in terms of multiple subpopulations. Our analyses
revealed that such a subpopulation structure enables flexible computations through a mechanism based on gain-controlled modulations
that flexibly shape the collective dynamics. Our results lead to task-specific predictions for the structure of neural selectivity, for
inactivation experiments and for the implication of different neurons in multi-tasking.

- Summarize — Dubreuil et al. 2022

**** Citation

Dubreuil, A., Valente, A., Beiran, M., Mastrogiuseppe, F. & Ostojic, S. /The role of population structure in computations through neural dynamics/. Nature Neuroscience 25, 783–794 (2022). DOI: [[https://doi.org/10.1038/s41593-022-01088-4][10.1038/s41593-022-01088-4]]

**** Problem and gap

The paper asks how two common views of neural computation relate:

1. /population/subpopulation coding/: neurons are organized into functional groups with distinct selectivity/connectivity statistics;
2. /low-dimensional dynamics/: neural populations compute through trajectories in a low-dimensional latent state space.

The gap is that low-dimensional dynamics describe /what the population activity does/, but not always /which connectivity/statistical structure makes those dynamics implement flexible computations/.

**** Key idea in one sentence

Flexible cognitive computations in low-rank RNNs often require not only sufficiently high-dimensional latent dynamics, but also non-random population structure that modulates the effective low-dimensional dynamics in an input- and task-dependent way.

**** Method sketch

They train recurrent neural networks on neuroscience-inspired tasks, especially low-rank RNNs where recurrent connectivity has form approximately

\[
J = \frac{1}{N}\sum_{r=1}^R m^{(r)} n^{(r)\top}.
\]

The low-rank structure induces latent variables

\[
\kappa_r = \frac{1}{N} n^{(r)\top} \phi(h),
\]

or equivalent coordinates depending on convention, yielding a reduced dynamical system in \(R\)-dimensional latent space.

They then analyze:

- the geometry of trained connectivity vectors;
- whether neurons form one or multiple Gaussian-like subpopulations in connectivity/selectivity space;
- whether resampling connectivity from fitted single-population or multi-population distributions preserves task performance;
- how inputs reshape effective latent dynamics through gain modulation;
- predictions for inactivation and neural selectivity.

**** Main result with quantitative anchor

A central empirical anchor: for each task and rank \(R=1,\dots,5\), they trained ten rank-\(R\) networks to determine minimal rank requirements. Extended Data Fig. 2 reports this rank sweep. Some tasks can be solved by increasing rank with largely random population structure, whereas flexible input–output mapping tasks require multiple non-random subpopulations even when the rank is sufficient.

For the delayed match-to-sample task, Extended Data Fig. 6 shows that rank-two networks can solve the task, but resampling from a single-population connectivity distribution fails to preserve performance, whereas resampling from a two-population distribution preserves high performance.

**** Mechanistic claim

The core mechanism is /gain-controlled modulation of latent dynamics/.

For a low-rank RNN, the effective latent dynamics can be written schematically as

\[
\dot{\kappa}_r
=
-\kappa_r
+
\frac{1}{N}\sum_i n_i^{(r)}
\phi\left(
\sum_s m_i^{(s)}\kappa_s
+
I_i(x)
\right).
\]

Inputs \(x\) shift each neuron's operating point. Because different subpopulations have different joint statistics of \(m_i,n_i,I_i,w_i\), the same external input can selectively change the gains

\[
\phi_i' =
\phi'\left(
\sum_s m_i^{(s)}\kappa_s + I_i(x)
\right),
\]

thereby changing the effective coupling between latent variables:

\[
\frac{\partial \dot{\kappa}_r}{\partial \kappa_s}
\approx
- \delta_{rs}
+
\frac{1}{N}\sum_i n_i^{(r)} \phi_i' m_i^{(s)}.
\]

Thus inputs can reshape the phase portrait: moving fixed points, changing stability, inducing or removing feedback loops, and in some cases generating oscillatory/limit-cycle-like dynamics.

**** Limitations

- The paper analyzes trained artificial RNNs, not direct biological circuit measurements.
- Population structure is inferred from connectivity/selectivity statistics in models; mapping this to biological cell types or anatomical populations is nontrivial.
- The low-rank assumption provides analytical tractability but may miss computations relying on high-dimensional transients.
- Resampling from Gaussian or Gaussian-mixture distributions tests whether certain low-order statistics are sufficient, but does not prove they are the unique mechanism.
- The trained solutions may depend on task setup, rank, initialization, optimizer, regularization and input/output conventions.

1. Plot vector fields under multiple fixed inputs:
   - zero input / delay;
   - sample A;

- Implement — Dubreuil-style population low-rank RNN

Below is a proposed replacement class aimed at /both/ training and Dubreuil-style post-hoc analysis.

Design goals:

1. low-rank recurrent connectivity
   \[
   J = \frac{1}{N} M N^\top
   \]
   with \(M,N\in\mathbb{R}^{N\times R}\);
2. explicit per-neuron connectivity vectors
   \[
   z_i =
   [m_i^{(1:R)}, n_i^{(1:R)}, w_{i,\mathrm{in}}^{(1:D)}, w_{i,\mathrm{out}}^{(1:O)}]
   \]
   for GMM / population analysis;
3. methods for:
   - extracting connectivity space;
   - fitting population labels;
   - resampling from one- or multi-population Gaussian statistics;
   - computing latent vector fields and gain-controlled Jacobians.

This is closest in spirit to Dubreuil et al. because the /population structure is statistical/, not hard-wired as tied parameters.

#+begin_src jupyter-python
import copy
from dataclasses import dataclass
from typing import Literal

import numpy as np
import torch
import torch.nn as nn


@dataclass
class PopulationFit:
    labels: np.ndarray
    means: np.ndarray
    covs: np.ndarray
    weights: np.ndarray
    feature_names: list[str]


class PopulationLowRankRNN(nn.Module):
    """
    Low-rank RNN with utilities for Dubreuil-style population analysis.

    State update
    ------------
    Let:
        x_t ∈ R^D
        h_t ∈ R^N
        r_t = tanh(W_i x_t + b_i + h_t)

    The recurrent input evolves as

        h_{t+1}
        =
        exp(-alpha) h_t
        +
        (1 - exp(-alpha)) M κ_t

    where

        κ_t = N^T r_t / N_hidden.

    Equivalently,

        M κ_t = M N^T r_t / N_hidden.

    Low-dimensional latent dynamics
    -------------------------------
    Under the approximation h = M κ,

        κ^+ = exp(-alpha) κ
              + (1 - exp(-alpha)) Ψ(κ; x)

    with

        Ψ(κ; x) = N^T tanh(Mκ + W_i x + b_i) / N_hidden.

    This matches the vector-field convention in your notebook.
    """

    def __init__(
        self,
        input_size: int,
        hidden_size: int,
        output_size: int,
        rank: int = 2,
        alpha: float = 0.05,
        rho: float = 0.5,
        init_noise_std: float = 0.0,
        device: str | torch.device = "cpu",
    ):
        super().__init__()

        self.input_size = input_size
        self.hidden_size = hidden_size
        self.output_size = output_size
        self.rank = rank
        self.device = torch.device(device)

        self.wi = nn.Linear(input_size, hidden_size, device=self.device)

        self.m = nn.Parameter(torch.empty(hidden_size, rank, device=self.device))
        self.n = nn.Parameter(torch.empty(hidden_size, rank, device=self.device))

        if output_size > 0:
            self.wo = nn.Linear(hidden_size, output_size, device=self.device)
        else:
            self.wo = None

        self.alpha = torch.tensor(float(alpha), device=self.device)
        self.exp_alpha = torch.exp(-self.alpha)
        self.nonlinearity = torch.tanh
        self.init_noise_std = float(init_noise_std)

        self.reset_parameters(rho=rho)

    def reset_parameters(
        self,
        rho: float = 0.5,
        input_scale: float = 0.2,
        readout_scale: float = 0.5,
    ) -> None:
        """
        Initialize low-rank factors with approximate recurrent gain rho.

        For rank R, this initializes M columns with norm sqrt(N), and sets
        N ≈ rho M plus orthogonal noise column-wise.
        """
        N = self.hidden_size
        R = self.rank

        with torch.no_grad():
            nn.init.normal_(
                self.wi.weight,
                mean=0.0,
                std=input_scale / np.sqrt(self.input_size),
            )
            nn.init.zeros_(self.wi.bias)

            m = torch.randn(N, R, device=self.device)
            q = torch.randn(N, R, device=self.device)

            m = m / m.norm(dim=0, keepdim=True).clamp_min(1e-8) * np.sqrt(N)

            q = q - m * (q * m).sum(dim=0, keepdim=True) / N
            q = q / q.norm(dim=0, keepdim=True).clamp_min(1e-8) * np.sqrt(N)

            noise_amp = np.sqrt(max(0.0, 1.0 - rho**2))
            n = rho * m + noise_amp * q

            self.m.copy_(m)
            self.n.copy_(n)

            if self.wo is not None:
                nn.init.normal_(
                    self.wo.weight,
                    mean=0.0,
                    std=readout_scale / np.sqrt(N),
                )
                nn.init.zeros_(self.wo.bias)

    @property
    def beta(self) -> torch.Tensor:
        """Effective Euler step: beta = 1 - exp(-alpha)."""
        return 1.0 - self.exp_alpha

    def initial_state(self, batch_size: int) -> tuple[torch.Tensor, torch.Tensor]:
        """
        Initialize recurrent input h and rate r.

        For clean dynamical-systems analysis, default is zero initial condition.
        Set init_noise_std > 0 during training if you want noisy initial states.
        """
        h = torch.zeros(batch_size, self.hidden_size, device=self.device)

        if self.init_noise_std > 0:
            h = h + self.init_noise_std * torch.randn_like(h)

        r = self.nonlinearity(h)
        return h, r

    def update_dynamics(
        self,
        x_t: torch.Tensor,
        h_t: torch.Tensor,
        r_t: torch.Tensor,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """
        One recurrent update.

        Shapes:
            x_t : [B, input_size]
            h_t : [B, hidden_size]
            r_t : [B, hidden_size]
        """
        kappa_t = r_t @ self.n / self.hidden_size          # [B, rank]
        recurrent_drive = kappa_t @ self.m.T              # [B, hidden_size]

        h_next = self.exp_alpha * h_t + self.beta * recurrent_drive
        r_next = self.nonlinearity(self.wi(x_t) + h_next)

        return h_next, r_next

    def get_readout(
        self,
        r_t: torch.Tensor,
        h_t: torch.Tensor,
        mode: Literal["linear", "n_last", "m_last"] = "linear",
    ) -> torch.Tensor:
        """
        Readout choices:
        - linear : trainable linear readout W_o r + b_o
        - n_last : projection onto last n vector
        - m_last : projection of h onto last m vector
        """
        if mode == "linear":
            if self.wo is None:
                raise ValueError("mode='linear' requires output_size > 0.")
            y = self.wo(r_t)
            return y.squeeze(-1) if y.shape[-1] == 1 else y

        if mode == "n_last":
            return r_t @ self.n[:, -1] / self.hidden_size

        if mode == "m_last":
            return h_t @ self.m[:, -1] / self.m[:, -1].norm().clamp_min(1e-8)

        raise ValueError(f"Unknown readout mode: {mode}")

    def forward(
        self,
        ff_inputs: torch.Tensor,
        ret_rates: bool = False,
        ret_inputs: bool = False,
        readout_mode: Literal["linear", "n_last", "m_last"] = "linear",
    ):
        """
        Parameters
        ----------
        ff_inputs : torch.Tensor
            Shape [B, T, input_size].
        """
        B, T, _ = ff_inputs.shape

        h_t, r_t = self.initial_state(B)

        readouts = []
        rates = []
        rec_inputs = []

        for t in range(T):
            h_t, r_t = self.update_dynamics(ff_inputs[:, t], h_t, r_t)
            y_t = self.get_readout(r_t, h_t, mode=readout_mode)

            readouts.append(y_t)
            rates.append(r_t)
            rec_inputs.append(h_t)

        readouts = torch.stack(readouts, dim=1)

        if ret_rates or ret_inputs:
            out = [readouts]
            if ret_rates:
                out.append(torch.stack(rates, dim=1))
            if ret_inputs:
                out.append(torch.stack(rec_inputs, dim=1))
            return tuple(out)

        return readouts

    # ------------------------------------------------------------------
    # Dubreuil-style population analysis utilities
    # ------------------------------------------------------------------

    def connectivity_vectors(
        self,
        include_input: bool = True,
        include_output: bool = True,
        include_biases: bool = False,
        standardize: bool = False,
    ) -> tuple[np.ndarray, list[str]]:
        """
        Return per-neuron connectivity vectors z_i.

        Default feature vector:

            z_i = [m_i, n_i, W_i[i, :], W_o[:, i]]

        Shape:
            Z : [hidden_size, n_features]
        """
        parts = []
        names = []

        m = self.m.detach().cpu().numpy()
        n = self.n.detach().cpu().numpy()

        parts.append(m)
        names += [f"m{r+1}" for r in range(self.rank)]

        parts.append(n)
        names += [f"n{r+1}" for r in range(self.rank)]

        if include_input:
            wi = self.wi.weight.detach().cpu().numpy()  # [N, input_size]
            parts.append(wi)
            names += [f"wi{d+1}" for d in range(self.input_size)]

        if include_biases:
            bi = self.wi.bias.detach().cpu().numpy()[:, None]
            parts.append(bi)
            names += ["bi"]

        if include_output and self.wo is not None:
            # wo.weight shape: [output_size, hidden_size]
            wo = self.wo.weight.detach().cpu().numpy().T  # [N, output_size]
            parts.append(wo)
            names += [f"wo{o+1}" for o in range(self.output_size)]

        Z = np.concatenate(parts, axis=1)

        if standardize:
            mu = Z.mean(axis=0, keepdims=True)
            sd = Z.std(axis=0, keepdims=True) + 1e-8
            Z = (Z - mu) / sd

        return Z, names

    def fit_populations_gmm(
        self,
        n_components: int = 2,
        covariance_type: str = "full",
        random_state: int = 0,
        standardize: bool = True,
        **kwargs,
    ) -> PopulationFit:
        """
        Fit Gaussian mixture model to per-neuron connectivity vectors.

        Requires scikit-learn.
        """
        from sklearn.mixture import GaussianMixture

        Z, names = self.connectivity_vectors(standardize=standardize)

        gmm = GaussianMixture(
            n_components=n_components,
            covariance_type=covariance_type,
            random_state=random_state,
            **kwargs,
        )
        labels = gmm.fit_predict(Z)

        if covariance_type != "full":
            raise ValueError("For resampling, use covariance_type='full'.")

        return PopulationFit(
            labels=labels,
            means=gmm.means_,
            covs=gmm.covariances_,
            weights=gmm.weights_,
            feature_names=names,
        )

    def population_summary(
        self,
        labels: np.ndarray,
        standardize: bool = False,
    ) -> dict[int, dict[str, np.ndarray]]:
        """
        Return mean and covariance of connectivity vectors for each population.
        """
        Z, names = self.connectivity_vectors(standardize=standardize)

        labels = np.asarray(labels)
        summary = {}

        for c in np.unique(labels):
            Zc = Z[labels == c]
            summary[int(c)] = {
                "n": Zc.shape[0],
                "mean": Zc.mean(axis=0),
                "cov": np.cov(Zc, rowvar=False),
                "feature_names": names,
            }

        return summary

    def resampled_copy(
        self,
        labels: np.ndarray | None = None,
        n_components: int | None = None,
        standardize_for_gmm: bool = False,
        cov_jitter: float = 1e-5,
        random_state: int = 0,
    ) -> "PopulationLowRankRNN":
        """
        Create a new network by resampling neuron-wise connectivity vectors.

        Two use cases:
        1. labels given:
             resample each population from its empirical Gaussian.
        2. n_components given:
             fit a GMM first, then resample from fitted populations.

        Important:
        If standardize_for_gmm=True, GMM is fit in standardized space, but this
        method currently resamples empirical unstandardized Gaussian statistics
        within the resulting labels. This is usually what you want for preserving
        physical parameter scales.
        """
        rng = np.random.default_rng(random_state)

        if labels is None:
            if n_components is None:
                raise ValueError("Provide labels or n_components.")
            fit = self.fit_populations_gmm(
                n_components=n_components,
                random_state=random_state,
                standardize=standardize_for_gmm,
            )
            labels = fit.labels

        labels = np.asarray(labels)

        Z, names = self.connectivity_vectors(
            include_input=True,
            include_output=True,
            include_biases=False,
            standardize=False,
        )

        Z_new = np.empty_like(Z)

        for c in np.unique(labels):
            idx = np.where(labels == c)[0]
            Zc = Z[idx]

            mu = Zc.mean(axis=0)
            cov = np.cov(Zc, rowvar=False)
            cov = cov + cov_jitter * np.eye(cov.shape[0])

            Z_new[idx] = rng.multivariate_normal(mu, cov, size=len(idx))

        new_model = copy.deepcopy(self)
        new_model.load_connectivity_vectors(Z_new, names)

        return new_model

    def load_connectivity_vectors(
        self,
        Z: np.ndarray,
        names: list[str],
    ) -> None:
        """
        Load m, n, input weights, and output weights from connectivity vectors.

        Assumes feature names produced by connectivity_vectors().
        """
        Z = np.asarray(Z, dtype=np.float32)

        def cols(prefix: str) -> list[int]:
            return [i for i, name in enumerate(names) if name.startswith(prefix)]

        m_cols = cols("m")
        n_cols = cols("n")
        wi_cols = cols("wi")
        wo_cols = cols("wo")

        with torch.no_grad():
            self.m.copy_(torch.tensor(Z[:, m_cols], device=self.device))
            self.n.copy_(torch.tensor(Z[:, n_cols], device=self.device))

            if len(wi_cols) > 0:
                self.wi.weight.copy_(torch.tensor(Z[:, wi_cols], device=self.device))

            if self.wo is not None and len(wo_cols) > 0:
                # Z[:, wo_cols] shape [N, output_size]
                self.wo.weight.copy_(torch.tensor(Z[:, wo_cols].T, device=self.device))

    # ------------------------------------------------------------------
    # Latent dynamics / gain modulation
    # ------------------------------------------------------------------

    def latent_psi(
        self,
        kappa: torch.Tensor,
        ff_input: torch.Tensor | None = None,
    ) -> torch.Tensor:
        """
        Ψ(κ; x) = N^{-1} N^T tanh(Mκ + W_i x + b_i).

        kappa shape:
            [..., rank]
        ff_input shape:
            [input_size] or [..., input_size]
        """
        if ff_input is None:
            ff_input = torch.zeros(self.input_size, device=self.device)

        kappa = torch.as_tensor(kappa, device=self.device, dtype=self.m.dtype)
        ff_input = torch.as_tensor(ff_input, device=self.device, dtype=self.m.dtype)

        h = kappa @ self.m.T

        drive = ff_input @ self.wi.weight.T + self.wi.bias

        rates = torch.tanh(h + drive)
        psi = rates @ self.n / self.hidden_size

        return psi

    def latent_field(
        self,
        kappa: torch.Tensor,
        ff_input: torch.Tensor | None = None,
        include_beta: bool = False,
    ) -> torch.Tensor:
        """
        F(κ; x) = Ψ(κ; x) - κ.

        If include_beta=True, return beta * F, where beta = 1-exp(-alpha).
        """
        kappa = torch.as_tensor(kappa, device=self.device, dtype=self.m.dtype)
        field = self.latent_psi(kappa, ff_input=ff_input) - kappa

        if include_beta:
            field = self.beta * field

        return field

    def latent_jacobian_flow(
        self,
        kappa: torch.Tensor,
        ff_input: torch.Tensor | None = None,
    ) -> torch.Tensor:
        """
        Jacobian of F(κ; x) = Ψ(κ; x) - κ.

        J_ab = N^{-1} sum_i n_i^a phi'_i m_i^b - δ_ab.
        """
        if ff_input is None:
            ff_input = torch.zeros(self.input_size, device=self.device)

        kappa = torch.as_tensor(kappa, device=self.device, dtype=self.m.dtype).reshape(-1)
        ff_input = torch.as_tensor(ff_input, device=self.device, dtype=self.m.dtype)

        u = self.m @ kappa + self.wi(ff_input)
        phi_prime = 1.0 - torch.tanh(u).pow(2)

        J = self.n.T @ (phi_prime[:, None] * self.m) / self.hidden_size
        J = J - torch.eye(self.rank, device=self.device, dtype=self.m.dtype)

        return J

    def latent_jacobian_map(
        self,
        kappa: torch.Tensor,
        ff_input: torch.Tensor | None = None,
    ) -> torch.Tensor:
        """
        Jacobian of discrete latent map:

            κ^+ = exp(-alpha)κ + beta Ψ(κ; x)

        Equivalent:

            J_map = I + beta J_flow.
        """
        J_flow = self.latent_jacobian_flow(kappa, ff_input=ff_input)
        I = torch.eye(self.rank, device=self.device, dtype=self.m.dtype)
        return I + self.beta * J_flow

    def neural_gains(
        self,
        kappa: torch.Tensor,
        ff_input: torch.Tensor | None = None,
    ) -> torch.Tensor:
        """
        Return per-neuron gains phi'(Mκ + W_i x + b_i).

        Shape:
            [hidden_size]
        """
        if ff_input is None:
            ff_input = torch.zeros(self.input_size, device=self.device)

        kappa = torch.as_tensor(kappa, device=self.device, dtype=self.m.dtype).reshape(-1)
        ff_input = torch.as_tensor(ff_input, device=self.device, dtype=self.m.dtype)

        u = self.m @ kappa + self.wi(ff_input)
        return 1.0 - torch.tanh(u).pow(2)

    def gains_by_population(
        self,
        labels: np.ndarray,
        kappa: torch.Tensor,
        ff_input: torch.Tensor | None = None,
    ) -> dict[int, np.ndarray]:
        """
        Return gain distributions split by population labels.
        """
        labels = np.asarray(labels)
        gains = self.neural_gains(kappa, ff_input=ff_input).detach().cpu().numpy()

        return {
            int(c): gains[labels == c]
            for c in np.unique(labels)
        }

|---|---|

| =latent_jacobian_flow()= | Quantify input-dependent gain modulation |

def generate_dpa_trials(n_trials, seq_len, input_size=2, noise=0.1):
    inputs = noise * torch.randn(n_trials, seq_len, input_size)
    targets = torch.zeros(n_trials, seq_len)

    # first cue: A vs B
    idx_A = torch.rand(n_trials) > 0.5
    idx_B = ~idx_A

    # second cue: C vs D
    idx_C = torch.rand(n_trials) > 0.5
    idx_D = ~idx_C

    # cue 1 encoding
    inputs[idx_A, N_STIM_ON[0]:N_STIM_OFF[0], 0] += 1.0
    inputs[idx_B, N_STIM_ON[0]:N_STIM_OFF[0], 1] += 1.0

    # cue 2 encoding
    inputs[idx_C, N_STIM_ON[1]:N_STIM_OFF[1], 2] += 1.0
    inputs[idx_D, N_STIM_ON[1]:N_STIM_OFF[1], 3] += 1.0

    # rewarded pairs: AC and BD
    idx_pair = (idx_A & idx_C) | (idx_B & idx_D)

    targets[:, :N_STIM_ON[0]] = 0.5

    targets[idx_pair, N_STIM_ON[1]:] = 1.0
    targets[~idx_pair, N_STIM_ON[1]:] = -1.0
    # targets[:, N_STIM_ON[1]:N_STIM_OFF[1]] = torch.nan

    return inputs, targets
#+end_src

def generate_gng_trials(n_trials, seq_len, input_size=3, noise=0.1):
    inputs = noise * torch.randn(n_trials, seq_len, input_size)
    targets = torch.zeros(n_trials, seq_len)
    mask = torch.zeros(1, seq_len)

    # Go no go
    idx_go = torch.rand(n_trials) > 0.5
    idx_nogo = ~idx_go

    # Go noGo
    inputs[idx_go, N_STIM_ON[0]:N_STIM_OFF[0], 4] += 1.0
    inputs[idx_nogo, N_STIM_ON[0]:N_STIM_OFF[0], 5] += 1.0

    # cue
    inputs[:, N_STIM_ON[1]:N_STIM_OFF[1], 6] += 1.0
    targets[:, N_STIM_ON[1]:N_STIM_OFF[1]] = torch.nan

    # rewarded go
    targets[idx_go, N_STIM_OFF[1]:] = 1.0
    targets[idx_nogo, N_STIM_OFF[1]:] = -1.0

    return inputs, targets, mask

def generate_dual_trials(n_trials, seq_len, input_size=7, noise=0.1):
    """
    Sequence:
      A/B -> None/Go/NoGo -> None/cue -> C/D

    Outputs:
      targets[0]: DPA decision after C/D
      targets[1]: GNG decision only for Go/NoGo trials

    Trial types balanced across 6 concrete conditions:
      0: DPA-only, A, C
      1: DPA-only, A, D
      2: Go,       A, C
      3: Go,       A, D
      4: NoGo,     B, C
      5: NoGo,     B, D
    """
    inputs = noise * torch.randn(n_trials, seq_len, input_size)
    targets = torch.zeros(n_trials, seq_len, 2)
    mask = torch.zeros(2, 1, seq_len)

    perm = torch.randperm(n_trials)
    trial_type = torch.empty(n_trials, dtype=torch.long)

    n_types = 6
    base = n_trials // n_types
    rem = n_trials % n_types

    start = 0
    for k in range(n_types):
        nk = base + (1 if k < rem else 0)
        trial_type[perm[start:start+nk]] = k
        start += nk

    idx0 = trial_type == 0
    idx1 = trial_type == 1
    idx2 = trial_type == 2
    idx3 = trial_type == 3
    idx4 = trial_type == 4
    idx5 = trial_type == 5

    idx_A = idx0 | idx1 | idx2 | idx3
    idx_B = idx4 | idx5
    idx_C = idx0 | idx2 | idx4
    idx_D = idx1 | idx3 | idx5

    idx_dpa_only = idx0 | idx1
    idx_go = idx2 | idx3
    idx_nogo = idx4 | idx5
    idx_gng = idx_go | idx_nogo

    # A/B
    inputs[idx_A, N_STIM_ON[0]:N_STIM_OFF[0], 0] += 1.0
    inputs[idx_B, N_STIM_ON[0]:N_STIM_OFF[0], 1] += 1.0

    # None / Go / NoGo
    inputs[idx_go,   N_STIM_ON[1]:N_STIM_OFF[1], 4] += 1.0
    inputs[idx_nogo, N_STIM_ON[1]:N_STIM_OFF[1], 5] += 1.0

    # None / cue
    inputs[idx_gng, N_STIM_ON[2]:N_STIM_OFF[2], 6] += 1.0

    # C/D
    inputs[idx_C, N_STIM_ON[3]:N_STIM_OFF[3], 2] += 1.0
    inputs[idx_D, N_STIM_ON[3]:N_STIM_OFF[3], 3] += 1.0

    # DPA target after final stimulus
    idx_pair = (idx_A & idx_C) | (idx_B & idx_D)
    targets[:, N_STIM_ON[3]:N_STIM_OFF[3], 0] = torch.nan
    targets[idx_pair, N_STIM_OFF[3]:, 0] = 1.0
    targets[~idx_pair, N_STIM_OFF[3]:, 0] = -1.0

    # GNG target only for Go/NoGo trials, during cue epoch
    targets[idx_go,   N_STIM_ON[2]:N_STIM_OFF[2], 1] = 1.0
    targets[idx_nogo, N_STIM_ON[2]:N_STIM_OFF[2], 1] = -1.0

    return inputs, targets, mask, trial_type

def plot_dynamics_by_trial_type(trial_type, trial_ids, kappa, predictions, targets, x_time, title=''):

        ax[2].plot(x_time, targets[idx0, :, 0].cpu().numpy(), 'k', lw=2, label='DPA target')

model.eval()

        "epoch": slice(0, N_STEPS),

        "conds": ["AC", "AD"],
        "epoch": slice(int(N_STIM_ON[0]), int(N_STIM_OFF[0])),
    },
    "B": {
        "dims": [1],
        "conds": ["BC", "BD"],
        "epoch": slice(int(N_STIM_ON[0]), int(N_STIM_OFF[0])),
    },
    "C": {
        "dims": [2],
        "conds": ["AC", "BC"],
        "epoch": slice(int(N_STIM_ON[1]), int(N_STIM_OFF[1])),
    },
    "D": {
        "dims": [3],
        "conds": ["AD", "BD"],
        "epoch": slice(int(N_STIM_ON[1]), int(N_STIM_OFF[1])),
    },
}

    "AC": "tab:blue",
    "AD": "tab:orange",
    "BC": "tab:green",
    "BD": "tab:purple",

with torch.no_grad():
    predictions, rates, rec_inputs = model(inputs.to(model.device), ret_rates=True)

rec_inputs_cpu = rec_inputs.detach().cpu()

kappa_traj = (
    project_rec_inputs_to_kappa(model, rec_inputs_cpu)
    .detach()
    .cpu()
    .numpy()
)

cond_idx = condition_indices_from_inputs(inputs)

condition_mean_trajs = {}
for name, idx in cond_idx.items():
    if len(idx) > 0:
        condition_mean_trajs[name] = kappa_traj[idx.detach().cpu().numpy()].mean(axis=0)

# ---------------------------------------------------------------------
# Precompute all fields first, so all panels use one common color scale
# ---------------------------------------------------------------------

field_cache = {}
all_speeds = []

for flow_name, spec in flow_specs.items():
    ff_input = make_input(
        input_size,
        active_dims=spec["dims"],
        value=1.0,
        device=device,
        dtype=dtype,
    )

    K1, K2, U, V, speed, xlim_flow, ylim_flow, params, ff_input_np = make_vector_field_grid(
        model,
        ff_input=ff_input,
        xlim=FLOW_GRID_XLIM,
        ylim=FLOW_GRID_YLIM,
        kappa_traj=kappa_traj,
        n_grid=N_GRID_FLOW,
        include_alpha=INCLUDE_ALPHA_IN_FIELD,
    )

    fixed_points, fixed_residuals = find_all_fixed_points(
        model,
        xlim=xlim_flow,
        ylim=ylim_flow,
        ff_input=ff_input,
        n_seeds=N_FP_SEEDS_FLOW,
        residual_tol=1e-8,
        merge_tol=5e-2,
    )

    fp_labels, fp_eigvals = classify_fixed_points(
        model,
        fixed_points,
        ff_input=ff_input,
        eig_tol=1e-5,
    )

    field_cache[flow_name] = dict(
        ff_input=ff_input,
        K1=K1,
        K2=K2,
        U=U,
        V=V,
        speed=speed,
        xlim=xlim_flow,
        ylim=ylim_flow,
        fixed_points=fixed_points,
        fp_labels=fp_labels,
