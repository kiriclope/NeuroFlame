* Low Rank RNNs

The Vanilla RNN updates its hidden state based on the previous hidden state and the current input:

Let:
- \( x_t \): feedforward input at time step \( t \)
- \( h_t \): recurrent input at time \( t \)
- \( r_t \): rate at time \( t \)
- \( y_t \): output at time \( t \)

\[h_t = (1-\alpha)h_{t-1} + \alpha \frac{1}{N} m n^\top r_{t-1}.\]

\[
r_t = \tanh(W_{ih} x_t + b_{ih} + h_{t-1})
\]

\[
y_t = \sigma(W_{ho} r_t + b_o)
\]

#+begin_src jupyter-python
class LowRankModel(nn.Module):
    def __init__(self, input_size, hidden_size, output_size, rank=1, alpha=0.01, device='cpu'):
        super().__init__()
        # input layer
        self.wi = nn.Linear(input_size, hidden_size)

        # low rank vectors
        self.m = nn.Parameter(torch.randn((hidden_size, rank)))
        self.n = nn.Parameter(torch.randn((hidden_size, rank)))

        with torch.no_grad():
            self.m.copy_(self.n)

        # output layer
        if output_size>0:
            print('linear readout')
            self.wo = nn.Linear(hidden_size, output_size)
        else:
            self.wo = None

        # non linearity
        self.nonlinearity = torch.tanh

        self.device = device
        # time constants
        self.alpha = alpha

    def get_readout(self, rates, rec_inputs, is_internal=False):
        if self.wo is not None:
            return self.wo(rates).squeeze(-1)
        if is_internal: # ostojic
            return rec_inputs @ self.m[:, -1] / torch.norm(self.m[:, -1])
        # vreeswijk
        return rates @ self.n[:, -1] / rates.shape[-1]


    def update_dynamics(self, ff_inputs, rec_inputs, rates):
        # updating the recurrent inputs
        hidden = (rates @ self.n) @ self.m.T / rates.shape[-1]
        # hidden = rates @ (self.m @ self.n.T) / rates.shape[-1]
        rec_inputs = (1.0 - self.alpha) * rec_inputs + self.alpha * hidden
        # updating the rates
        rates = self.nonlinearity(self.wi(ff_inputs) + rec_inputs)

        return rates, rec_inputs


    def forward(self, ff_inputs, ret_rates=False):
        # rates and rec_inputs are (n_trials, hidden_size)
        rec_inputs = torch.randn(ff_inputs.shape[0], self.m.shape[0])
        rates = self.nonlinearity(self.wi(ff_inputs[:, 0]) + rec_inputs)

        readout_list = []
        rates_list = []
        inputs_list = []

        for step in range(ff_inputs.shape[1]):
            rates, rec_inputs = self.update_dynamics(ff_inputs[:, step], rec_inputs, rates)
            readout = self.get_readout(rates, rec_inputs)

            rates_list.append(rates)
            inputs_list.append(rec_inputs)
            readout_list.append(readout)

        rates_list = torch.stack(rates_list, dim=1)
        inputs_list = torch.stack(inputs_list, dim=1)
        readout_list = torch.stack(readout_list, dim=1)


        if ret_rates:
            return readout_list, rates_list, inputs_list

        return readout_list

def generate_dpa_trials(n_trials, seq_len, input_size=2, noise=0.1):
    inputs = noise * torch.randn(n_trials, seq_len, input_size)
    targets = torch.zeros(n_trials, seq_len)
    mask = torch.zeros(1, seq_len)

    # first cue: A vs B
    idx_A = torch.rand(n_trials) > 0.5
    idx_B = ~idx_A

    # second cue: C vs D
    idx_C = torch.rand(n_trials) > 0.5
    idx_D = ~idx_C

    # cue 1 encoding
    inputs[idx_A, N_STIM_ON[0]:N_STIM_OFF[0], 0] += 1.0
    inputs[idx_B, N_STIM_ON[0]:N_STIM_OFF[0], 1] += 1.0

    # cue 2 encoding
    inputs[idx_C, N_STIM_ON[1]:N_STIM_OFF[1], 2] += 1.0
    inputs[idx_D, N_STIM_ON[1]:N_STIM_OFF[1], 3] += 1.0

    # rewarded pairs: AC and BD
    idx_pair = (idx_A & idx_C) | (idx_B & idx_D)
    targets[idx_pair, N_STIM_OFF[1]:] = 1.0
    targets[~idx_pair, N_STIM_OFF[1]:] = -1.0
    mask[0, N_STIM_ON[1]:] = 1.0

    return inputs, targets, mask

def optimization(model, X, Y, criterion, optimizer, mask=1, epochs=30, batch_size=32, verbose=1):

    loss_list = []
    for epoch in range(epochs):
        permutation = torch.randperm(X.size(0))
        total_loss = 0
        for i in range(0, X.size(0), batch_size):
            indices = permutation[i:i+batch_size]
            batch_x, batch_y = X[indices], Y[indices]

            output = model(batch_x)

            loss = torch.sum(criterion(output, batch_y) * mask) / torch.sum(mask) / 2
            loss += torch.sum(criterion(output, batch_y*0) * (1-mask)) / torch.sum(1-mask) / 2

            loss.backward()
            optimizer.step()
            optimizer.zero_grad()
            total_loss += loss.item()

        loss_list.append(total_loss / batch_size)

        if (verbose) and (((epoch+1) % 5==0) or (epoch==epochs-1) or (epoch==0)):
            print(f'Epoch {epoch+1}/{epochs}, Training Loss: {loss:.4f}')

    return loss_list
#+end_src

def generate_gng_trials(n_trials, seq_len, input_size=3, noise=0.1):
    inputs = noise * torch.randn(n_trials, seq_len, input_size)
    targets = torch.zeros(n_trials, seq_len)
    mask = torch.zeros(1, seq_len)

    # Go no go
    idx_go = torch.rand(n_trials) > 0.5
    idx_nogo = ~idx_go

    # Go noGo
    inputs[idx_go, N_STIM_ON[0]:N_STIM_OFF[0], 4] += 1.0

def optimization(model, X, Y, criterion, optimizer, mask=1, epochs=30, batch_size=32, verbose=1):

    loss_list = []
    for epoch in range(epochs):
        permutation = torch.randperm(X.size(0))
        total_loss = 0
        for i in range(0, X.size(0), batch_size):
            indices = permutation[i:i+batch_size]
            batch_x, batch_y = X[indices], Y[indices]

            output = model(batch_x)

            loss = torch.sum(criterion(output, batch_y) * mask) / torch.sum(mask) / 2
            loss += torch.sum(criterion(output, batch_y*0) * (1-mask)) / torch.sum(1-mask) / 2

            loss.backward()
            optimizer.step()
            optimizer.zero_grad()
            total_loss += loss.item()

        loss_list.append(total_loss / batch_size)

        if (verbose) and (((epoch+1) % 5==0) or (epoch==epochs-1) or (epoch==0)):
            print(f'Epoch {epoch+1}/{epochs}, Training Loss: {loss:.4f}')

    return loss_list

def generate_dual_trials(n_trials, seq_len, input_size=7, noise=0.1):
    """
    Sequence:
      A/B -> None/Go/NoGo -> None/cue -> C/D

    Outputs:
      targets[0]: DPA decision after C/D
      targets[1]: GNG decision only for Go/NoGo trials

    Trial types balanced across 6 concrete conditions:
      0: DPA-only, A, C
      1: DPA-only, A, D
      2: Go,       A, C
      3: Go,       A, D
      4: NoGo,     B, C
      5: NoGo,     B, D
    """
    inputs = noise * torch.randn(n_trials, seq_len, input_size)
    targets = torch.zeros(2, n_trials, seq_len)
    mask = torch.zeros(2, 1, seq_len)

    perm = torch.randperm(n_trials)
    trial_type = torch.empty(n_trials, dtype=torch.long)

    n_types = 6
    base = n_trials // n_types
    rem = n_trials % n_types

    start = 0
    for k in range(n_types):
        nk = base + (1 if k < rem else 0)
        trial_type[perm[start:start+nk]] = k
        start += nk

    idx0 = trial_type == 0
    idx1 = trial_type == 1
    idx2 = trial_type == 2
    idx3 = trial_type == 3
    idx4 = trial_type == 4
    idx5 = trial_type == 5

    idx_A = idx0 | idx1 | idx2 | idx3
    idx_B = idx4 | idx5
    idx_C = idx0 | idx2 | idx4
    idx_D = idx1 | idx3 | idx5

    idx_dpa_only = idx0 | idx1
    idx_go = idx2 | idx3
    idx_nogo = idx4 | idx5
    idx_gng = idx_go | idx_nogo

    # A/B
    inputs[idx_A, N_STIM_ON[0]:N_STIM_OFF[0], 0] += 1.0
    inputs[idx_B, N_STIM_ON[0]:N_STIM_OFF[0], 1] += 1.0

    # None / Go / NoGo
    inputs[idx_go,   N_STIM_ON[1]:N_STIM_OFF[1], 4] += 1.0
    inputs[idx_nogo, N_STIM_ON[1]:N_STIM_OFF[1], 5] += 1.0

    # None / cue
    inputs[idx_gng, N_STIM_ON[2]:N_STIM_OFF[2], 6] += 1.0

    # C/D
    inputs[idx_C, N_STIM_ON[3]:N_STIM_OFF[3], 2] += 1.0
    inputs[idx_D, N_STIM_ON[3]:N_STIM_OFF[3], 3] += 1.0

    # DPA target after final stimulus
    idx_pair = (idx_A & idx_C) | (idx_B & idx_D)
    targets[0, idx_pair, N_STIM_OFF[3]:] = 1.0
    targets[0, ~idx_pair, N_STIM_OFF[3]:] = -1.0
    mask[0, 0, N_STIM_ON[3]:] = 1.0

    # GNG target only for Go/NoGo trials, during cue epoch
    targets[1, idx_go,   N_STIM_ON[2]:N_STIM_OFF[2]] = 1.0
    targets[1, idx_nogo, N_STIM_ON[2]:N_STIM_OFF[2]] = -1.0
    mask[1, 0, N_STIM_ON[2]:N_STIM_OFF[2]] = 1.0

    return inputs, targets, mask, trial_type
#+end_src

def optimization(model, X, Y, criterion, optimizer, mask, epochs=30, batch_size=32, verbose=1):
    """
    Shared readout trained on multiple task targets.

    X:    (n_trials, seq_len, input_size)
    Y:    (n_tasks, n_trials, seq_len)
    mask: (n_tasks, 1, seq_len)
    model(X) -> (n_trials, seq_len)
    """
    loss_list = []
    n_tasks = Y.shape[0]

    for epoch in range(epochs):
        permutation = torch.randperm(X.size(0))
        total_loss = 0.0

        for i in range(0, X.size(0), batch_size):
            indices = permutation[i:i+batch_size]
            batch_x = X[indices]      # (B, T, in)
            batch_y = Y[:, indices]   # (n_tasks, B, T)

            output = model(batch_x)   # (B, T)

            loss = 0.0
            for task in range(n_tasks):
                task_loss = torch.sum(criterion(output, batch_y[task]) * mask[task]) / torch.sum(mask[task]) / 2
                base_loss = torch.sum(criterion(output, torch.zeros_like(batch_y[task])) * (1 - mask[task])) / torch.sum(1 - mask[task]) / 2
                loss = loss + task_loss + base_loss

            optimizer.zero_grad()
            loss.backward()
            optimizer.step()

            total_loss += loss.item()

        loss_list.append(total_loss / max(1, X.size(0) // batch_size))

        if verbose and (((epoch + 1) % 5 == 0) or (epoch == 0) or (epoch == epochs - 1)):
            print(f'Epoch {epoch+1}/{epochs}, Training Loss: {loss:.4f}')

    return loss_list
#+end_src

def _as_int(x):
    return int(x.item()) if hasattr(x, "item") else int(x)


def infer_dpa_conditions(inputs):
    """Infer AC/AD/BC/BD labels from cue-channel means."""
    t1_on, t1_off = _as_int(N_STIM_ON[0]), _as_int(N_STIM_OFF[0])
    t2_on, t2_off = _as_int(N_STIM_ON[1]), _as_int(N_STIM_OFF[1])

    s1 = inputs[:, t1_on:t1_off, :2].mean(dim=1)
    s2 = inputs[:, t2_on:t2_off, 2:4].mean(dim=1)

    is_A = s1[:, 0] > s1[:, 1]
    is_C = s2[:, 0] > s2[:, 1]

    return np.array([
        ("A" if a else "B") + ("C" if c else "D")
        for a, c in zip(is_A.cpu().numpy(), is_C.cpu().numpy())
    ])


def internal_readout_matrix(model):
    """
    Return P such that kappa = h @ P.T.

    For M = [m_1, ..., m_R],
        kappa_k = h dot m_k / ||m_k||^2.
    """
    M = model.m
    col_norm2 = torch.sum(M**2, dim=0).clamp_min(1e-12)
    return M.T / col_norm2[:, None]


def h_from_internal_kappa(model, kappa_grid, ridge=1e-6):
    """
    Construct h in span(M) whose internal M-readouts equal kappa_grid.

    Coordinate:
        kappa = D^{-1} M^T h,
        D = diag(||m_k||^2).

    For non-orthogonal M:
        h = M a,
        a = (M^T M)^(-1) D kappa.
    """
    M = model.m
    _, R = M.shape

    G = M.T @ M
    D = torch.diag(torch.diag(G))
    G_reg = G + ridge * torch.eye(R, device=M.device, dtype=M.dtype)

    a_T = torch.linalg.solve(G_reg, D @ kappa_grid.T)
    a = a_T.T

    return a @ M.T


def internal_latent_vector_field(model, kappa_grid, ff_input=None, ridge=1e-6):
    """
    Vector field in internal M-readout coordinates.

    Recurrent-input dynamics:
        dh/dt = -h + M psi(h, x)

    where
        psi = N^T tanh(h + W_i x + b_i) / H.

    Internal coordinate:
        kappa_k = h dot m_k / ||m_k||^2.
    """
    device = next(model.parameters()).device
    dtype = next(model.parameters()).dtype

    kappa_grid = kappa_grid.to(device=device, dtype=dtype)
    Pn, _ = kappa_grid.shape
    H = model.m.shape[0]

    if ff_input is None:
        ff_input = torch.zeros(model.wi.in_features, device=device, dtype=dtype)
    else:
        ff_input = torch.as_tensor(ff_input, device=device, dtype=dtype)

    ff_input = ff_input[None, :].expand(Pn, -1)

    M = model.m
    N = model.n

    h = h_from_internal_kappa(model, kappa_grid, ridge=ridge)
    rates = model.nonlinearity(model.wi(ff_input) + h)

    psi = rates @ N / H
    h_drive = psi @ M.T

    P_int = internal_readout_matrix(model)
    kappa_drive = h_drive @ P_int.T

    d_kappa = kappa_drive - kappa_grid

    return d_kappa, kappa_drive


@torch.no_grad()
def run_model_with_internal_readouts(model, ff_inputs, rec_init="zero"):
    """
    Run model and return internal M-readout trajectories.

    kappa_k(t) = h_t dot m_k / ||m_k||^2.

    Returns
    -------
    kappa : np.ndarray, shape (B, T, R)
    readout : np.ndarray, shape (B, T) or (B, T, O)
    """
    was_training = model.training
    model.eval()

    device = next(model.parameters()).device
    dtype = next(model.parameters()).dtype

    x = ff_inputs.to(device=device, dtype=dtype)
    B, T, _ = x.shape
    H, _ = model.m.shape

    if rec_init == "zero":
        rec_inputs = torch.zeros(B, H, device=device, dtype=dtype)
    elif rec_init == "random":
        rec_inputs = torch.randn(B, H, device=device, dtype=dtype)
    else:
        raise ValueError("rec_init must be 'zero' or 'random'.")

    rates = model.nonlinearity(model.wi(x[:, 0]) + rec_inputs)
    P_int = internal_readout_matrix(model)

    kappa_list = []
    readout_list = []

    for t in range(T):
        rates, rec_inputs = model.update_dynamics(x[:, t], rec_inputs, rates)

        kappa_t = rec_inputs @ P_int.T
        readout_t = model.get_readout(rates, rec_inputs)

        kappa_list.append(kappa_t)
        readout_list.append(readout_t)

    kappa = torch.stack(kappa_list, dim=1)
    readout = torch.stack(readout_list, dim=1)

    if was_training:
        model.train()

    return kappa.cpu().numpy(), readout.squeeze(-1).cpu().numpy()


def compute_internal_field_for_input(model, K1, K2, ff_input=None):
    """Compute vector field on a 2D internal-coordinate grid."""
    kappa_grid_np = np.stack([K1.ravel(), K2.ravel()], axis=1)
    kappa_grid = torch.tensor(kappa_grid_np, dtype=torch.float32)

    with torch.no_grad():
        dK, _ = internal_latent_vector_field(
            model,
            kappa_grid,
            ff_input=ff_input,
        )

    dK_np = dK.cpu().numpy()
    U = dK_np[:, 0].reshape(K1.shape)
    V = dK_np[:, 1].reshape(K2.shape)
    speed = np.sqrt(U**2 + V**2)

    return U, V, speed


def pick_one_trial_per_condition(labels, condition_order=("AC", "AD", "BC", "BD")):
    """Return first available trial index for each condition."""
    idx_by_condition = {}

    for cond in condition_order:
        idx = np.flatnonzero(labels == cond)
        if len(idx) == 0:
            raise RuntimeError(f"No trials found for condition {cond}.")
        idx_by_condition[cond] = int(idx[0])

    return idx_by_condition


def make_grid_from_trajectories(
    traj,
    idx_by_condition,
    condition_order=("AC", "AD", "BC", "BD"),
    pad=0.75,
    expand=2.0,
    min_lim=5.0,
    n_grid=121,
):
    """Build square grid around selected trajectories."""
    selected = np.stack([traj[idx_by_condition[c]] for c in condition_order], axis=0)
    flat = selected.reshape(-1, 2)

    k1_min, k2_min = flat.min(axis=0) - pad
    k1_max, k2_max = flat.max(axis=0) + pad

    base_lim = np.max(np.abs([k1_min, k1_max, k2_min, k2_max]))
    lim = max(expand * base_lim, min_lim)

    k1 = np.linspace(-lim, lim, n_grid)
    k2 = np.linspace(-lim, lim, n_grid)
    K1, K2 = np.meshgrid(k1, k2)

    return K1, K2, lim

def find_fixed_points_2d(
    model,
    K1,
    K2,
    U,
    V,
    ff_input=None,
    n_candidates=120,
    residual_tol=1e-5,
    unique_tol=5e-2,
):
    """
    Find fixed points of the 2D projected latent dynamics.

    Roots satisfy

        d kappa^M / dt = 0.

    Uses low-speed grid points as initial guesses and refines with scipy.root.

    Returns
    -------
    fixed_points : np.ndarray, shape (n_fixed_points, 2)
        Rows are [kappa_1, kappa_2].
    """
    speed = np.sqrt(U**2 + V**2)

    k1_min, k1_max = K1.min(), K1.max()
    k2_min, k2_max = K2.min(), K2.max()

    n_candidates = min(n_candidates, speed.size)

    candidate_idx = np.argpartition(
        speed.ravel(),
        n_candidates - 1,
    )[:n_candidates]

    guesses = np.column_stack([
        K1.ravel()[candidate_idx],
        K2.ravel()[candidate_idx],
    ])

    def field_np(kappa_np):
        kappa = torch.tensor(
            kappa_np[None, :],
            dtype=torch.float32,
        )

        with torch.no_grad():
            dK, _ = internal_latent_vector_field(
                model,
                kappa,
                ff_input=ff_input,
            )

        return dK.cpu().numpy().reshape(-1)

    fixed_points = []

    for guess in guesses:
        sol = root(field_np, guess, method="hybr")

        if not sol.success:
            continue

        fp = sol.x
        residual = np.linalg.norm(field_np(fp))

        if residual > residual_tol:
            continue

        # Keep only fixed points inside plotted region.
        if not (k1_min <= fp[0] <= k1_max and k2_min <= fp[1] <= k2_max):
            continue

        # Deduplicate nearby roots.
        is_new = all(
            np.linalg.norm(fp - old_fp) > unique_tol
            for old_fp in fixed_points
        )

        if is_new:
            fixed_points.append(fp)

    if len(fixed_points) == 0:
        return np.empty((0, 2))

    return np.stack(fixed_points, axis=0)

def find_fixed_points_2d(
    model,
    K1,
    K2,
    U,
    V,
    ff_input=None,
    n_candidates=120,
    residual_tol=1e-5,
    unique_tol=5e-2,
):
    """
    Find fixed points of the 2D projected latent dynamics.

    Roots satisfy

        d kappa^M / dt = 0.

    Uses low-speed grid points as initial guesses and refines with scipy.root.

    Returns
    -------
    fixed_points : np.ndarray, shape (n_fixed_points, 2)
        Rows are [kappa_1, kappa_2].
    """
    speed = np.sqrt(U**2 + V**2)

    k1_min, k1_max = K1.min(), K1.max()
    k2_min, k2_max = K2.min(), K2.max()

    n_candidates = min(n_candidates, speed.size)

    candidate_idx = np.argpartition(
        speed.ravel(),
        n_candidates - 1,
    )[:n_candidates]

    guesses = np.column_stack([
        K1.ravel()[candidate_idx],
        K2.ravel()[candidate_idx],
    ])

    def field_np(kappa_np):
        kappa = torch.tensor(
            kappa_np[None, :],
            dtype=torch.float32,
        )

        with torch.no_grad():
            dK, _ = internal_latent_vector_field(
                model,
                kappa,
                ff_input=ff_input,
                # ff_input=ff_input,
            )

        return dK.cpu().numpy().reshape(-1)

    fixed_points = []

    for guess in guesses:
        sol = root(field_np, guess, method="hybr")

        if not sol.success:
            continue

        fp = sol.x
        residual = np.linalg.norm(field_np(fp))

        if residual > residual_tol:
            continue

        # Keep only fixed points inside plotted region.
        if not (k1_min <= fp[0] <= k1_max and k2_min <= fp[1] <= k2_max):
            continue

        # Deduplicate nearby roots.
        is_new = all(
            np.linalg.norm(fp - old_fp) > unique_tol
            for old_fp in fixed_points
        )

        if is_new:
            fixed_points.append(fp)

    if len(fixed_points) == 0:
        return np.empty((0, 2))

    return np.stack(fixed_points, axis=0)

*** Derivation of the phase-portrait flow

We analyze the dynamics in the recurrent-input variable \(h_t \in \mathbb{R}^H\), where \(H\) is the hidden size. In the code, the low-rank recurrent matrix is

\[J = \frac{1}{H} M N^\top,\]

with

\[M = \begin{bmatrix} m_1 & m_2 \end{bmatrix} \in \mathbb{R}^{H \times R}, \qquad N = \begin{bmatrix} n_1 & n_2 \end{bmatrix} \in \mathbb{R}^{H \times R}, \qquad R=2.\]

For a fixed external input \(x \in \mathbb{R}^{d_{\mathrm{in}}}\), the rate vector is

\[r(h,x)=\phi(h + W_i x + b_i),\]

where \(\phi = \tanh\), \(W_i\) and \(b_i\) are the input-layer weights and bias.

The discrete recurrent-input update in the model is approximately

\[h_{t+1}=(1-\alpha)h_t+\alpha \frac{1}{H} M N^\top r_t.\]

Equivalently, using

\[\psi(h,x)=\frac{1}{H}N^\top \phi(h + W_i x + b_i)\in \mathbb{R}^R,\]

the recurrent drive is

\[h_{\mathrm{drive}}(h,x)=M \psi(h,x).\]

Thus

\[
h_{t+1}
=
(1-\alpha)h_t + \alpha h_{\mathrm{drive}}(h_t,x).
\]

For phase portraits we use the continuous-time vector field corresponding to this Euler update:

\[
\dot h
=
-h + M\psi(h,x).
\]

The factor \(\alpha\) only rescales time, so it does not change fixed points or streamline directions.

**** Internal \(M\)-readout coordinates

We plot dynamics in the internal coordinates

\[\kappa^M_k=\frac{m_k^\top h}{\|m_k\|^2},\qquadk=1,\dots,R.\]

Define

\[
D
=
\operatorname{diag}
\left(
\|m_1\|^2,\dots,\|m_R\|^2
\right)
\in \mathbb{R}^{R \times R}.
\]

Then the internal coordinate vector is

\[\kappa^M=D^{-1}M^\top h.\]

This is what the code computes using

#+begin_src python
P_int = M.T / torch.sum(M**2, dim=0)[:, None]
kappa = h @ P_int.T
#+end_src

#+RESULTS:

because \(h\) is represented as a row vector in PyTorch.

**** Why we need to reconstruct \(h\) from \(\kappa^M\)

A phase portrait requires evaluating the vector field at grid points \(\kappa^M = (\kappa^M_1,\kappa^M_2)\). But the network dynamics are defined in \(h\)-space, not directly in \(\kappa^M\)-space.

If \(M\)'s columns were orthogonal, one could use

\[
h = M\kappa^M.
\]

But after training the columns of \(M\) need not be orthogonal. Therefore, \(h=M\kappa^M\) would not generally satisfy

\[
D^{-1}M^\top h = \kappa^M.
\]

Instead, for each desired grid point \(\kappa^M\), we construct an \(h\) in the span of \(M\):

\[
h = M a.
\]

We require

\[
\kappa^M
=
D^{-1}M^\top h
=
D^{-1}M^\top M a.
\]

Let

\[
G = M^\top M.
\]

Then

\[
\kappa^M
=
D^{-1}G a.
\]

Solving for \(a\),

\[
a = G^{-1}D\kappa^M.
\]

Therefore the recurrent input corresponding to an internal-coordinate grid point is

\[h(\kappa^M)=M G^{-1}D\kappa^M.\]

The code implements this in row-vector convention as

#+begin_src python
G = M.T @ M
D = torch.diag(torch.diag(G))
a_T = torch.linalg.solve(G_reg, D @ kappa_grid.T)
a = a_T.T
h = a @ M.T
#+end_src

The small ridge term in \(G_{\mathrm{reg}}\) is only for numerical stability.

**** Flow in internal coordinates

Starting from

\[
\dot h
=
-h + M\psi(h,x),
\]

we project both sides into internal coordinates:

\[
\dot \kappa^M
=
D^{-1}M^\top \dot h.
\]

Substituting the \(h\)-dynamics,

\[
\dot \kappa^M
=
D^{-1}M^\top
\left[
-h + M\psi(h,x)
\right].
\]

Using \(\kappa^M = D^{-1}M^\top h\),

\[
D^{-1}M^\top h = \kappa^M.
\]

Therefore,

\[
\dot \kappa^M = -\kappa^M + D^{-1}M^\top M\psi(h,x).
\]

Since \(G=M^\top M\),

\[
\boxed{\dot \kappa^M = -\kappa^M + D^{-1}G\psi(h(\kappa^M),x)}
\]

with

\[
\psi(h(\kappa^M),x)
=
\frac{1}{H}
N^\top
\tanh
\left[
h(\kappa^M) + W_i x + b_i
\right].
\]

This is the flow plotted in the phase portraits.

If \(M\)'s columns are orthogonal, then \(G=D\), so

\[
D^{-1}G = I,
\]

and the flow simplifies to

\[
\dot \kappa^M = -\kappa^M + \psi(h(\kappa^M),x).
\]

For non-orthogonal \(M\), the factor \(D^{-1}G\) mixes the low-rank drive components.

**** Autonomous and input-driven flows

The autonomous phase portrait uses zero input:

\[x = 0.\]

Thus

\[
\dot \kappa^M_{\mathrm{auto}}=-\kappa^M+D^{-1}G\psi(h(\kappa^M),0).
\]

The fixed-input phase portraits use constant cue inputs:

\[x_A = (1,0,0,0), \qquad x_B = (0,1,0,0), \qquad x_C = (0,0,1,0), \qquad x_D = (0,0,0,1).\]

F_1(\kappa^M;x) \\
F_2(\kappa^M;x)

The red nullcline is

Intersections of the two nullclines are fixed points:

F(\kappa^M;x)=0.

The heatmap shows the flow speed

\[\|F(\kappa^M;x)\|=\sqrt{F_1(\kappa^M;x)^2+F_2(\kappa^M;x)^2}.\]

# ---- Dual-task phase-portrait routines ----

import numpy as np
import torch
import matplotlib.pyplot as plt
import matplotlib.lines as mlines
from scipy.optimize import root


def internal_readout_matrix(model):
    """
    Return P such that kappa = h @ P.T.

    kappa_k = h dot m_k / ||m_k||^2.
    """
    M = model.m
    col_norm2 = torch.sum(M**2, dim=0).clamp_min(1e-12)
    return M.T / col_norm2[:, None]


def h_from_internal_kappa(model, kappa_grid, ridge=1e-6):
    """
    Reconstruct h in span(M) whose internal readout is kappa_grid.

    kappa = D^{-1} M^T h
    h = M a
    a = G^{-1} D kappa
    """
    M = model.m
    _, R = M.shape

    G = M.T @ M
    D = torch.diag(torch.diag(G))
    eye = torch.eye(R, device=M.device, dtype=M.dtype)
    G_reg = G + ridge * eye

    a_T = torch.linalg.solve(G_reg, D @ kappa_grid.T)
    a = a_T.T

    h = a @ M.T
    return h


def internal_latent_vector_field(model, kappa_grid, ff_input=None, ridge=1e-6):
    """
    Compute vector field in internal M-readout coordinates.

    d kappa / dt = -kappa + D^{-1} M^T M psi(h, x)

    where psi = N^T tanh(h + W_i x + b_i) / H.
    """
    device = next(model.parameters()).device
    dtype = next(model.parameters()).dtype

    kappa_grid = kappa_grid.to(device=device, dtype=dtype)
    Pn, R = kappa_grid.shape
    H = model.m.shape[0]

    if ff_input is None:
        ff_input = torch.zeros(model.wi.in_features, device=device, dtype=dtype)
    else:
        ff_input = torch.as_tensor(ff_input, device=device, dtype=dtype)

    if ff_input.numel() != model.wi.in_features:
        raise ValueError(
            f"ff_input has length {ff_input.numel()}, "
            f"but model expects {model.wi.in_features} inputs."
        )

    ff_input = ff_input.reshape(1, -1).expand(Pn, -1)

    M = model.m
    N = model.n

    h = h_from_internal_kappa(model, kappa_grid, ridge=ridge)
    rates = model.nonlinearity(model.wi(ff_input) + h)

    psi = rates @ N / H
    h_drive = psi @ M.T

    P_int = internal_readout_matrix(model)
    kappa_drive = h_drive @ P_int.T

    d_kappa = kappa_drive - kappa_grid

    return d_kappa, kappa_drive


@torch.no_grad()
def run_model_with_internal_readouts(model, ff_inputs, rec_init="zero"):
    """
    Run model and return internal M-readout trajectories.

    Returns
    -------
    kappa : ndarray, shape (B, T, R)
    readout : ndarray, shape (B, T)
    """
    was_training = model.training
    model.eval()

    device = next(model.parameters()).device
    dtype = next(model.parameters()).dtype

    x = ff_inputs.to(device=device, dtype=dtype)
    B, T, _ = x.shape
    H, R = model.m.shape

    if rec_init == "zero":
        rec_inputs = torch.zeros(B, H, device=device, dtype=dtype)
    elif rec_init == "random":
        rec_inputs = torch.randn(B, H, device=device, dtype=dtype)
    else:
        raise ValueError("rec_init must be 'zero' or 'random'.")

    rates = model.nonlinearity(model.wi(x[:, 0]) + rec_inputs)
    P_int = internal_readout_matrix(model)

    kappa_list = []
    readout_list = []

    for t in range(T):
        rates, rec_inputs = model.update_dynamics(x[:, t], rec_inputs, rates)

        kappa_t = rec_inputs @ P_int.T
        readout_t = model.get_readout(rates, rec_inputs)

        kappa_list.append(kappa_t)
        readout_list.append(readout_t)

    kappa = torch.stack(kappa_list, dim=1)
    readout = torch.stack(readout_list, dim=1)

    if was_training:
        model.train()

    return kappa.cpu().numpy(), readout.cpu().numpy()


def compute_internal_field_for_input(model, K1, K2, ff_input=None):
    """Compute vector field on a 2D internal-coordinate grid."""
    kappa_grid_np = np.stack([K1.ravel(), K2.ravel()], axis=1)
    kappa_grid = torch.tensor(kappa_grid_np, dtype=torch.float32)

    with torch.no_grad():
        dK, _ = internal_latent_vector_field(
            model,
            kappa_grid,
            ff_input=ff_input,
        )

    dK_np = dK.cpu().numpy()
    U = dK_np[:, 0].reshape(K1.shape)
    V = dK_np[:, 1].reshape(K2.shape)
    speed = np.sqrt(U**2 + V**2)

    return U, V, speed


def make_grid_from_trajectories(
    traj,
    idx_by_condition,
    condition_order,
    pad=0.75,
    expand=2.0,
    min_lim=5.0,
    n_grid=121,
):
    """Build square grid around selected trajectories."""
    selected = np.stack(
        [traj[idx_by_condition[c], :, :2] for c in condition_order],
        axis=0,
    )
    flat = selected.reshape(-1, 2)

    k1_min, k2_min = flat.min(axis=0) - pad
    k1_max, k2_max = flat.max(axis=0) + pad

    base_lim = np.max(np.abs([k1_min, k1_max, k2_min, k2_max]))
    lim = max(expand * base_lim, min_lim)

    k1 = np.linspace(-lim, lim, n_grid)
    k2 = np.linspace(-lim, lim, n_grid)

def find_fixed_points_2d(
    model,
    K1,
    K2,
    U,
    V,
    ff_input=None,
    n_candidates=160,
    residual_tol=1e-5,
    unique_tol=5e-2,
):
    """Find fixed points of the 2D projected latent vector field."""
    speed = np.sqrt(U**2 + V**2)

    k1_min, k1_max = K1.min(), K1.max()
    k2_min, k2_max = K2.min(), K2.max()

    n_candidates = min(n_candidates, speed.size)
    candidate_idx = np.argpartition(speed.ravel(), n_candidates - 1)[:n_candidates]

    guesses = np.column_stack([
        K1.ravel()[candidate_idx],
        K2.ravel()[candidate_idx],
    ])

    def field_np(kappa_np):
        kappa = torch.tensor(kappa_np[None, :], dtype=torch.float32)

        with torch.no_grad():
            dK, _ = internal_latent_vector_field(
                model,
                kappa,
                ff_input=ff_input,
            )

        return dK.cpu().numpy().reshape(-1)

    fixed_points = []

    for guess in guesses:
        sol = root(field_np, guess, method="hybr")

        if not sol.success:
            continue

        fp = sol.x
        residual = np.linalg.norm(field_np(fp))

        if residual > residual_tol:
            continue

        if not (k1_min <= fp[0] <= k1_max and k2_min <= fp[1] <= k2_max):
            continue

        is_new = all(
            np.linalg.norm(fp - old_fp) > unique_tol
            for old_fp in fixed_points
        )

        if is_new:
            fixed_points.append(fp)

    if len(fixed_points) == 0:
        return np.empty((0, 2))

    return np.stack(fixed_points, axis=0)

def field_np_2d(model, kappa_np, ff_input=None):
    """Evaluate 2D internal-coordinate vector field at one or many points."""
    kappa_np = np.asarray(kappa_np, dtype=np.float32)

    if kappa_np.ndim == 1:
        kappa_np = kappa_np[None, :]

    kappa = torch.tensor(kappa_np, dtype=torch.float32)

        dK, _ = internal_latent_vector_field(
            model,
            kappa,

def finite_difference_jacobian_2d(model, fp, ff_input=None, eps=1e-4):
    """Numerical Jacobian of the 2D kappa vector field."""
    fp = np.asarray(fp, dtype=np.float64)
    J = np.zeros((2, 2), dtype=np.float64)

    for j in range(2):
        step = np.zeros(2)
        step[j] = eps

        f_plus = field_np_2d(model, fp + step, ff_input=ff_input).reshape(2)
        f_minus = field_np_2d(model, fp - step, ff_input=ff_input).reshape(2)

        J[:, j] = (f_plus - f_minus) / (2 * eps)

    return J


def is_stable_fixed_point_2d(model, fp, ff_input=None, stability_tol=1e-4):
    """
    Stable for continuous-time flow if all real eigenvalues of Jacobian are negative.
    """
    J = finite_difference_jacobian_2d(model, fp, ff_input=ff_input)
    eigvals = np.linalg.eigvals(J)
    max_real = np.max(np.real(eigvals))

    return max_real < -stability_tol, eigvals


def cluster_points(points, tol=0.15):
    """Greedy clustering of nearby 2D points."""
    points = np.asarray(points)

    if len(points) == 0:
        return np.empty((0, 2))

    clusters = []

    for p in points:
        assigned = False

        for cluster in clusters:
            if np.linalg.norm(p - cluster["center"]) < tol:
                cluster["points"].append(p)
                cluster["center"] = np.mean(cluster["points"], axis=0)
                assigned = True
                break

        if not assigned:
            clusters.append({
                "center": p.copy(),
                "points": [p.copy()],
            })

    return np.stack([c["center"] for c in clusters], axis=0)


def find_stable_fixed_points_2d(
    model,
    K1,
    K2,
    ff_input=None,
    n_starts_per_axis=31,
    n_steps=2500,
    dt=0.05,
    speed_tol=1e-4,
    root_residual_tol=1e-5,
    cluster_tol=0.20,
    stability_tol=1e-4,
    margin=0.05,
):
    """
    Find stable attracting fixed points only.

    Procedure:
    1. Start many initial conditions on the plotting grid.
    2. Integrate the 2D flow forward.
    3. Keep endpoints with low speed.
    4. Cluster endpoints.
    5. Refine each cluster center with scipy.root.
    6. Keep only stable fixed points.
    """
    k1_min, k1_max = float(K1.min()), float(K1.max())
    k2_min, k2_max = float(K2.min()), float(K2.max())

    k1 = np.linspace(k1_min, k1_max, n_starts_per_axis)
    k2 = np.linspace(k2_min, k2_max, n_starts_per_axis)
    S1, S2 = np.meshgrid(k1, k2)

    starts = np.stack([S1.ravel(), S2.ravel()], axis=1).astype(np.float32)
    z = starts.copy()

    # Batched Euler integration.
    for _ in range(n_steps):
        dz = field_np_2d(model, z, ff_input=ff_input)
        z = z + dt * dz

        # Keep trajectories inside a slightly expanded box.
        z[:, 0] = np.clip(z[:, 0], k1_min - margin, k1_max + margin)
        z[:, 1] = np.clip(z[:, 1], k2_min - margin, k2_max + margin)

    final_speed = np.linalg.norm(
        field_np_2d(model, z, ff_input=ff_input),
        axis=1,
    )

    converged = z[final_speed < speed_tol]

    # If threshold is too strict, use the slowest endpoints.
    if len(converged) == 0:
        slow_idx = np.argsort(final_speed)[:50]
        converged = z[slow_idx]

    candidates = cluster_points(converged, tol=cluster_tol)

    refined = []

    def field_for_root(kappa_np):
        return field_np_2d(model, kappa_np, ff_input=ff_input).reshape(-1)

    for cand in candidates:
        sol = root(field_for_root, cand, method="hybr")

        if not sol.success:
            continue

        fp = sol.x
        residual = np.linalg.norm(field_for_root(fp))

        if residual > root_residual_tol:
            continue

        if not (k1_min <= fp[0] <= k1_max and k2_min <= fp[1] <= k2_max):
            continue

        stable, eigvals = is_stable_fixed_point_2d(
            model,
            fp,
            ff_input=ff_input,
            stability_tol=stability_tol,
        )

        if not stable:
            continue

        refined.append(fp)

    stable_fps = cluster_points(refined, tol=cluster_tol)

    return stable_fps

def field_np_2d(model, kappa_np, ff_input=None):
    kappa_np = np.asarray(kappa_np, dtype=np.float32)

    if kappa_np.ndim == 1:
        kappa_np = kappa_np[None, :]

    kappa = torch.tensor(kappa_np, dtype=torch.float32)

    with torch.no_grad():
        dK, _ = internal_latent_vector_field(
            model,
            kappa,
            ff_input=ff_input,
        )

    return dK.cpu().numpy()


def cluster_points_with_counts(points, tol=0.25):
    points = np.asarray(points)

    clusters = []

    for p in points:
        assigned = False

        for c in clusters:
            if np.linalg.norm(p - c["center"]) < tol:
                c["points"].append(p)
                c["center"] = np.mean(c["points"], axis=0)
                assigned = True
                break

        if not assigned:
            clusters.append({
                "center": p.copy(),
                "points": [p.copy()],
            })

    centers = np.stack([c["center"] for c in clusters], axis=0)
    counts = np.array([len(c["points"]) for c in clusters])

    order = np.argsort(counts)[::-1]

    return centers[order], counts[order]


def find_task_relevant_fixed_points(
    model,
    starts,
    ff_input=None,
    n_expected=3,
    n_steps=4000,
    dt=0.05,
    cluster_tol=0.30,
    residual_tol=1e-5,
):
    """
    Find fixed points reached by task trajectories.

    This avoids scanning all nullcline intersections and keeps only the
    attractors reached from actual trial endpoints.
    """
    z = np.asarray(starts, dtype=np.float32).copy()

    # Relax under chosen flow
    for _ in range(n_steps):
        dz = field_np_2d(model, z, ff_input=ff_input)
        z = z + dt * dz

    centers, counts = cluster_points_with_counts(z, tol=cluster_tol)

    # Keep largest basins
    centers = centers[:n_expected]

    refined = []

    def f_root(x):
        return field_np_2d(model, x, ff_input=ff_input).reshape(-1)

    for c in centers:
        sol = root(f_root, c, method="hybr")

        if not sol.success:
            continue

        fp = sol.x
        residual = np.linalg.norm(f_root(fp))

        if residual < residual_tol:
            refined.append(fp)

    if len(refined) == 0:
        return np.empty((0, 2))

    refined, _ = cluster_points_with_counts(np.stack(refined), tol=cluster_tol)

    return refined[:n_expected]
