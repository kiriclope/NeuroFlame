* 🧠 Dynamical RNNs

** 🧠 1. All to all RNN

Let:
- \( x_t \): feedforward input at time step \( t \)
- \( r_t \): rate at time \( t \)
- \( y_t \): output at time \( t \)

\[
r_t = (1 - \frac{\delta t}{\tau}) r_{t-1} + \frac{\Delta t}{\tau} \tanh(W_{ih} x_t + b_{ih} + W_{hh} r_{t-1})
\]

\[
y_t = \sigma(W_{ho} r_t + b_o)
\]

#+begin_src ipython
class BioRNNModel(nn.Module):
    def __init__(self, input_size, hidden_size, output_size, dt=0.1, tau=1.0):
        super().__init__()
        # input layer
        self.wi = nn.Linear(input_size, hidden_size)
        # recurrent layer
        self.wrec = nn.Parameter(torch.Tensor(hidden_size, hidden_size))
        nn.init.xavier_uniform_(self.wrec)
        # output layer
        self.wo = nn.Linear(hidden_size, output_size)
        # non linearity
        self.nonlinearity = torch.tanh
        # time constant
        self.DT_TAU = dt / tau

    def update_dynamics(self, ff_inputs, rates):
        # updating the recurrent inputs and rates
        rec_inputs = rates @ self.wrec.T # / rates.shape[-1]
        nonlinear = self.nonlinearity(self.wi(ff_inputs) + rec_inputs)
        rates = (1-self.DT_TAU) * rates + self.DT_TAU * nonlinear

        return rates, rec_inputs


    def forward(self, ff_inputs):
        # ff_input is (n_trials, seq_len, input_size)
        # rates and rec_inputs are (n_trials, hidden_size)
        rec_inputs = torch.randn(ff_inputs.shape[0], self.wrec.shape[0])
        rates = self.nonlinearity(self.wi(ff_inputs[:, -1]) + rec_inputs)

        readout_list = []

        for step in range(ff_inputs.shape[1]):
            rates, rec_inputs = self.update_dynamics(ff_inputs[:, step], rates)
            readout = self.wo(rates)
            readout_list.append(readout)

        readout_list = torch.stack(readout_list, dim=1)

        return readout_list

** 🧠 2. Low Rank RNN

The Vanilla RNN updates its hidden state based on the previous hidden state and the current input:


Let:
- \( x_t \): feedforward input at time step \( t \)
- \( h_t \): recurrent input at time \( t \)
- \( r_t \): rate at time \( t \)
- \( y_t \): output at time \( t \)

\[
h_t = (1- \alpha) h_{t-1} + \alpha m n.T r_{t-1}
\]

\[
r_t = \tanh(W_{ih} x_t + b_{ih} + h_{t-1})
\]

\[
y_t = \sigma(W_{ho} r_t + b_o)
\]


#+begin_src ipython
class LowRankModel(nn.Module):
    def __init__(self, input_size, hidden_size, output_size, rank=1, alpha=0.01, device='cuda'):
        super().__init__()
        # input layer
        self.wi = nn.Linear(input_size, hidden_size)

        # low rank vectors
        self.m = nn.Parameter(torch.randn((hidden_size, rank)))
        self.n = nn.Parameter(torch.randn((hidden_size, rank)))

        # output layer

        if output_size>0:
            self.wo = nn.Linear(hidden_size, output_size)
        else:
            self.wo = None

        # non linearity
        self.nonlinearity = torch.tanh

        self.device = device
        # time constants
        self.alpha = alpha

    def get_readout(self, rates):
        if self.wo is not None:
            return self.wo(rates)
        return rates @ self.n[:, -1] / rates.shape[-1]

    def update_dynamics(self, ff_inputs, rec_inputs, rates):
        # updating the recurrent inputs
        hidden = rates @ (self.m @ self.n.T) / rates.shape[-1]
        rec_inputs = (1.0 - self.alpha) * rec_inputs + self.alpha * hidden
        # updating the rates
        rates = self.nonlinearity(self.wi(ff_inputs) + rec_inputs)

        return rates, rec_inputs


    def forward(self, ff_inputs, ret_rates=False):
        # rates and rec_inputs are (n_trials, hidden_size)
        rec_inputs = torch.randn(ff_inputs.shape[0], self.m.shape[0])
        rates = self.nonlinearity(self.wi(ff_inputs[:, 0]) + rec_inputs)

        readout_list = []
        rates_list = []

        for step in range(ff_inputs.shape[1]):
            rates, rec_inputs = self.update_dynamics(ff_inputs[:, step], rec_inputs, rates)
            readout = self.get_readout(rates)

            rates_list.append(rates)
            readout_list.append(readout)

        rates_list = torch.stack(rates_list, dim=1)
        readout_list = torch.stack(readout_list, dim=1)

        if ret_rates:
            return readout_list, rates_list

        return readout_list

def generate_gng_trials(n_trials, seq_len, input_size=2, noise=0.1):
    """Generates one trial of Delayed Right/No-Right."""
    inputs = noise * torch.randn((n_trials, seq_len, input_size))
    targets = torch.zeros((n_trials, seq_len))
    mask = torch.zeros((1, seq_len))

    # Decide if it's a Right (1) or Left (0) trial
    is_go = torch.rand(n_trials) > 0.5
    # Apply input stimulus early in sequence
    inputs[is_go, stim_onset:stim_offset, 0] += 1.0
    inputs[~is_go, stim_onset:stim_offset, 0] -= 1.0

    inputs[:, response_onset:, 1] += 1.0

    # Output only during Right time
    targets[is_go, response_onset:] += 1.0
    targets[~is_go, response_onset:] -= 1.0

    mask[0, response_onset:] += 1.0

    return inputs, targets, mask

def optimization(model, X, Y, criterion, optimizer, mask=1, epochs=30, batch_size=32, verbose=1):

    loss_list = []
    for epoch in range(epochs):
        permutation = torch.randperm(X.size(0))
        total_loss = 0
        for i in range(0, X.size(0), batch_size):
            indices = permutation[i:i+batch_size]
            batch_x, batch_y = X[indices], Y[indices]

            output = model(batch_x).squeeze(-1)

            loss = torch.sum(criterion(output, batch_y) * mask) / torch.sum(mask) / 2
            loss += torch.sum(criterion(output, batch_y*0) * (1-mask)) / torch.sum(1-mask) / 2

            loss.backward()
            optimizer.step()
            optimizer.zero_grad()
            total_loss += loss.item()

        loss_list.append(total_loss)

        if (verbose) and (((epoch+1) % 5==0) or (epoch==epochs-1) or (epoch==0)):
            print(f'Epoch {epoch+1}/{epochs}, Training Loss: {loss:.4f}')

    return loss_list
#+end_src

from scipy.interpolate import griddata

from sklearn.cluster import KMeans


def create_mesh(x, y, size=100):
    x_min, x_max = x.min() - 0.5, x.max() + 0.5
    y_min, y_max = y.min() - 0.5, y.max() + 0.5

    dx = np.gradient(x, axis=1)
    dy = np.gradient(y, axis=1)

    xg = np.arange(size) * (x_max - x_min) / (size - 1) + x_min
    yg = np.arange(size) * (y_max - y_min) / (size - 1) + y_min

    xi, yi = np.meshgrid(xg, yg)

    points = np.vstack((x.flatten(), y.flatten())).T

    ui = griddata(points, dx.flatten(), (xi, yi), method='linear', fill_value=np.nan)
    vi = griddata(points, dy.flatten(), (xi, yi), method='linear', fill_value=np.nan)

    return xg, yg, xi, yi, ui, vi


def get_overlaps_from_rates(rates, model):
    n = model.n.detach().cpu()
    N = n.shape[0]

    n1 = n[:, 0].clone()
    n2 = n[:, 1].clone()
    n2 = n2 - (n2 @ n1) * n1 / (n1 @ n1)

    vec = torch.stack((n1, n2), dim=1)
    overlaps = rates.cpu() @ vec / N

    return overlaps.numpy()

def create_grid_initial_conditions(model, grid_range=3, grid_size=20):
    """Create rates on an equally spaced grid in (kappa1, kappa2).

    Set rates = kappa1 * v1 + kappa2 * v2 where v1, v2 are chosen so
    that rates @ n_orth / N = kappa exactly. Uses pseudoinverse of the
    orthogonalized n vectors, then clamps to [-1, 1].
    """
    n = model.n.detach().cpu().float()
    N = n.shape[0]

    n1 = n[:, 0].clone()
    n2 = n[:, 1].clone()
    n2 = n2 - (n2 @ n1) * n1 / (n1 @ n1)
    V = torch.stack((n1, n2), dim=1)  # (N, 2)

    V_pinv = torch.linalg.pinv(V)  # (2, N)

    k1 = torch.linspace(-grid_range, grid_range, grid_size)
    k2 = torch.linspace(-grid_range, grid_range, grid_size)
    K1, K2 = torch.meshgrid(k1, k2, indexing='ij')

    kappa_target = torch.stack((K1.flatten(), K2.flatten()), dim=1)  # (n_points, 2)

    # rates such that rates @ V / N = kappa_target
    rates = kappa_target @ (N * V_pinv)  # (n_points, N)
    rates = torch.clamp(rates, -1.0, 1.0)

    # Consistent rec_inputs: h = m n^T r / N (what recurrence would produce)
    rec_inputs = rates @ (model.m @ model.n.T).detach().cpu() / N

    # Verify
    kappa_check = rates @ V / N
    print(f'Max kappa error: {(kappa_check - kappa_target).abs().max().item():.6f}')
    print(f'Rates range: [{rates.min().item():.3f}, {rates.max().item():.3f}]')

    return rates, rec_inputs, K1, K2

def run_dynamics(model, rates_init, rec_inputs_init, input_type='none', n_steps=50):
    n_points = rates_init.shape[0]
    input_size = model.wi.in_features

    ff_input = torch.zeros(n_points, input_size)

    if input_type == 'go':
        ff_input[:, 0] = 1.0
    elif input_type == 'nogo':
        ff_input[:, 0] = -1.0
    elif input_type == 'response':
        ff_input[:, 1] = 1.0
    elif input_type == 'go_response':
        ff_input[:, 0] = 1.0
        ff_input[:, 1] = 1.0
    elif input_type == 'nogo_response':
        ff_input[:, 0] = -1.0
        ff_input[:, 1] = 1.0

    rates = rates_init.clone()
    rec_inputs = rec_inputs_init.clone()

    rates_list = [rates.detach().clone()]

def create_grid_from_rec_inputs(model, grid_range=3, grid_size=20):
    """Create initial conditions on a grid in rec_input space: h = α*n1 + β*n2."""
    n = model.n.detach().cpu().float()
    N = n.shape[0]

    # Orthogonalize for visualization
    n1 = n[:, 0].clone()
    n2 = n[:, 1].clone()
    n2_orth = n2 - (n2 @ n1) * n1 / (n1 @ n1)

    k1 = torch.linspace(-grid_range, grid_range, grid_size)
    k2 = torch.linspace(-grid_range, grid_range, grid_size)
    K1, K2 = torch.meshgrid(k1, k2, indexing='ij')

    # rec_inputs = alpha * n1 + beta * n2_orth  (in N-dimensional space)
    alphas = K1.flatten()  # (n_points,)
    betas = K2.flatten()

    rec_inputs = alphas.unsqueeze(1) * n1.unsqueeze(0) + betas.unsqueeze(1) * n2_orth.unsqueeze(0)
    # (n_points, N)

    # Rates from rec_inputs (no ff input)
    rates = torch.tanh(rec_inputs)

    # Verify overlaps
    V = torch.stack((n1, n2_orth), dim=1)
    kappa_check = rates @ V / N
    print(f'Rates range: [{rates.min().item():.3f}, {rates.max().item():.3f}]')
    print(f'Kappa range: [{kappa_check.min().item():.3f}, {kappa_check.max().item():.3f}]')

    return rates, rec_inputs, K1, K2


def run_dynamics_bypass(model, rates_init, rec_inputs_init, ff_bypass=None, n_steps_input=50, n_steps_free=50):
    """Run dynamics, optionally injecting ff input directly into rec_inputs (bypassing wi),
    then run free (autonomous) dynamics.

    ff_bypass: (n_points, N) tensor added to rec_inputs at each input step, or None.
    """
    rates = rates_init.clone()
    rec_inputs = rec_inputs_init.clone()
    N = rates.shape[-1]

    input_size = model.wi.in_features
    ff_zero = torch.zeros(rates.shape[0], input_size)

    rates_list = [rates.detach().clone()]

    with torch.no_grad():
        # Phase 1: input ON (bypass wi, add directly to rec_inputs)
        for step in range(n_steps_input):
            hidden = rates @ (model.m @ model.n.T).detach() / N
            rec_inputs = (1.0 - model.alpha) * rec_inputs + model.alpha * hidden
            if ff_bypass is not None:
                rec_inputs = rec_inputs + model.alpha * ff_bypass
            rates = torch.tanh(rec_inputs)  # no wi contribution
            rates_list.append(rates.detach().clone())

        # Phase 2: autonomous (no input)
        for step in range(n_steps_free):
            rates, rec_inputs = model.update_dynamics(ff_zero, rec_inputs, rates)
            rates_list.append(rates.detach().clone())

    return torch.stack(rates_list, dim=1)

def find_fixed_points(model, rates_init, rec_inputs_init, input_type='none',
                      n_steps=500, speed_thr=1e-4, n_clusters=3):
    """Find fixed points by running dynamics to convergence under a specific input."""
    n = model.n.detach().cpu().float()
    N = n.shape[0]
    n1 = n[:, 0].clone()
    n2 = n[:, 1].clone()
    n2 = n2 - (n2 @ n1) * n1 / (n1 @ n1)
    V = torch.stack((n1, n2), dim=1)

    rates = rates_init.clone()
    rec_inputs = rec_inputs_init.clone()
    input_size = model.wi.in_features
    ff_input = torch.zeros(rates.shape[0], input_size)

    if input_type == 'go':
        ff_input[:, 0] = 1.0
    elif input_type == 'nogo':
        ff_input[:, 0] = -1.0
    elif input_type == 'response':
        ff_input[:, 1] = 1.0
    elif input_type == 'go_response':
        ff_input[:, 0] = 1.0; ff_input[:, 1] = 1.0
    elif input_type == 'nogo_response':
        ff_input[:, 0] = -1.0; ff_input[:, 1] = 1.0

    with torch.no_grad():
        for _ in range(n_steps):
            rates_old = rates.clone()
            rates, rec_inputs = model.update_dynamics(ff_input, rec_inputs, rates)

    # Speed at convergence
    kappa_final = (rates @ V / N).numpy()
    kappa_old = (rates_old @ V / N).numpy()
    speed = np.sqrt(((kappa_final - kappa_old)**2).sum(axis=1))

    # Keep only converged points
    mask = speed < speed_thr
    if mask.sum() == 0:
        mask = speed < np.percentile(speed, 5)

    pts = kappa_final[mask]
    nc = min(n_clusters, len(pts))
    if nc > 0 and len(pts) > 0:
        km = KMeans(n_clusters=nc, random_state=0, n_init=10).fit(pts)
        return km.cluster_centers_
    return np.array([])

*** Why are the flows different when created from the inputs or from the rates? These should be the same

The flows look different because the two grid methods produce *different initial =rec_inputs=* for the same overlap coordinates, and the dynamics depend on *both* =rates= and =rec_inputs= independently.

**** The core issue

Your dynamical system has *two state variables*: =rates= and =rec_inputs=. The update rule is:

#+begin_src python
rec_inputs = (1 - α) * rec_inputs + α * (rates @ M @ N.T / N)
rates = tanh(wi(ff_input) + rec_inputs)
#+end_src

So the next state depends on the *current =rec_inputs=*, not just the current =rates=. Two points with identical =rates= (and thus identical overlaps κ) but different =rec_inputs= will evolve differently.

**** What each method does

*Method 1: =create_grid_initial_conditions=* (grid on rates)
- Sets =rates= via pseudoinverse so that =rates @ V / N = κ_target=
- Sets =rec_inputs = rates @ (m @ n.T) / N= — the *steady-state recurrent input* assuming no feedforward drive

*Method 2: =create_grid_from_rec_inputs=* (grid on rec_inputs)
- Sets =rec_inputs = α₁·n₁ + α₂·n₂_orth= on a uniform grid
- Sets =rates = tanh(rec_inputs)= — consistent with zero feedforward input

These give *different (rates, rec_inputs) pairs* even at the same κ values because:

1. *Method 1*: =rec_inputs= is determined by =rates= through the recurrent weight matrix — it's what the recurrence /would produce/, but =rates= themselves were constructed algebraically and aren't necessarily consistent with =rates = tanh(wi(0) + rec_inputs)=.

2. *Method 2*: =rates = tanh(rec_inputs)= is self-consistent for zero ff input, but =rec_inputs= lives only in the span of ={n₁, n₂}=, whereas in Method 1 it lives in the span of =m @ n.T= applied to rates (which can have components in many directions).

**** The fix

To get consistent flows, you need to initialize from points on the *correct manifold* — where both state variables are consistent with each other. The cleanest approach:

1. *Run autonomous dynamics to convergence* from many initial conditions to find the slow manifold
2. *Sample initial conditions from that manifold*
3. Then compute single-step flows

Alternatively, if you want a grid in κ-space, pick one method and stick with it, but understand that the flow field is really a projection of a higher-dimensional (rates, rec_inputs) system onto 2D, and the projection depends on which 2D slice of the full state space you're sampling.

*** I tought that to get a flow field, I needed to start the network on the grid and let it evolve to a fixed point. Then compute the speed of the trajectories

You're absolutely right about the procedure — the issue isn't the method (start on grid → evolve → compute velocity), it's that *the grid defines a 2D slice of a much higher-dimensional state space*, and the two methods pick /different slices/.

**** The key point

Your state space is *(rates, rec_inputs)* — that's 2×N = 256 dimensions. The overlap κ = (κ₁, κ₂) is a 2D projection. Many different (rates, rec_inputs) configurations project to the *same* (κ₁, κ₂) point but have *different dynamics*.

When you create a grid in κ-space, you must choose /how to lift/ each 2D point back to the full 256D state. The two methods make different choices → different full states → different dynamics → different flows.

**** What to do

The most principled approach: *initialize from actual network trajectories*, not from an artificial grid. Your trial data already lives on the correct manifold:

#+begin_src python
# Use rates from actual trials as initial conditions
# These are self-consistent (rates, rec_inputs) pairs
inputs_grid, targets_grid, _ = generate_gng_trials(2000, seq_len, input_size, noise=0.5)
_, rates_all = model(inputs_grid, ret_rates=True)
#+end_src

Then pick a specific time point, use those (rates, rec_inputs) as initial conditions, and compute the flow from there. The trajectories will live on the manifold the network actually uses.

Alternatively, if you want the grid approach, just *pick one method and use it consistently* — the flows will be self-consistent within that slice. The "correct" one depends on what question you're asking. Method 2 (grid on rec_inputs, rates = tanh(rec_inputs)) is arguably more natural since it satisfies the instantaneous rate equation.

*** but still you are just running one step instead of the full trajectory to the fixed points

#+begin_src python

This computes the /instantaneous velocity/ (single-step displacement), which is perfectly valid for a flow field — it shows the direction and speed of the dynamics at each point. You don't need to run to the fixed point for this.

**** Two different things

1. *Flow field (velocity field)*: Run 1 step (or a few), compute displacement. This shows /where the dynamics push you/ from each point. This is what =plot_flow_raw= does and it's correct.

2. *Trajectories to fixed points*: Run many steps, watch where each initial condition ends up. This is what =run_dynamics= with large =n_steps= does, and what =plot_flow_grid= uses with the interpolated mesh.

Both are valid visualizations. The single-step flow field is actually the more standard way to show a phase portrait — it's the vector field $\dot{\kappa} = f(\kappa)$.


**** what is the othere method

The other method (=plot_flow_grid= / =create_mesh=) does something different:

1. Runs /full trajectories/ (=n_steps=100=) from each initial condition
2. Computes velocities /along the entire trajectory/ via =np.gradient=
3. /Interpolates/ all those velocities onto a regular mesh using =griddata=

#+begin_src python
# In create_mesh:
dx = np.gradient(x, axis=1)  # velocity along trajectory
dy = np.gradient(y, axis=1)
# Then interpolate ALL (x,y,dx,dy) points from ALL timesteps onto a grid
points = np.vstack((x.flatten(), y.flatten())).T
ui = griddata(points, dx.flatten(), (xi, yi), method='linear')
#+end_src

***** Why it looks different

- It mixes velocities from /different times and locations/ along the trajectories. A trajectory that passes through point (κ₁, κ₂) early (moving fast) and another that passes through the same point late (moving slow) both contribute — and =griddata= averages them.
- This gives a /smoothed-out average flow/ across the visited manifold, rather than the instantaneous velocity from a specific initial state.

***** Summary

| Method | What it computes | Pros | Cons |
|---|---|---|---|
| =plot_flow_raw= (1 step) | Instantaneous velocity at each grid point | Clean, exact, standard phase portrait | Depends on choice of initial (rates, rec_inputs) |
| =plot_flow_grid= (many steps + interpolation) | Averaged velocity from trajectories passing through each region | Samples the actual manifold | Mixes velocities from different times, can be noisy/misleading |

The single-step method (=plot_flow_raw=) is the more standard and interpretable one for phase portraits. The trajectory-based method is useful for seeing where the dynamics /actually go/, but the interpolated velocity field can be hard to interpret.
